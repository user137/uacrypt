1//! `crypto_pwhash` equivalent (`docs/dstu-crypto-project.md` "Mapping onto the libsodium API",
2//! `docs/TASKS.md` T-71, `docs/DECISIONS.md` D-03/D-49/D-50) - plain Argon2id, the one deliberately non-DSTU
3//! component (no Ukrainian standard covers password hashing). Wraps the `argon2` crate
4//! (`RustCrypto/password-hashes`, vetted in D-49) with libsodium's own `crypto_pwhash_str`/
5//! `crypto_pwhash_str_verify` shape: a self-describing PHC string that embeds algorithm, version,
6//! salt, and parameters, so `verify_password` needs nothing but the password and that string back.
7//!
8//! Every parameter choice here is cited to libsodium's own `crypto_pwhash_argon2id` C source
9//! (`docs/DECISIONS.md` D-50), not invented: only Argon2id (no algorithm knob), a fixed 1-lane
10//! parallelism (`pwhash_argon2id.c`'s own `argon2id_hash_encoded(..., (uint32_t) 1U, ...)` call,
11//! not a knob either), a 16-byte salt (`crypto_pwhash_argon2id_SALTBYTES`), a 32-byte hash
12//! (`STR_HASHBYTES`), and three named strength presets mirroring
13//! `OPSLIMIT`/`MEMLIMIT_{INTERACTIVE,MODERATE,SENSITIVE}` exactly - no raw `m_cost`/`t_cost` knob
14//! exposed, per D-47's "libsodium API shape, no misconfigurable knobs" criterion. `verify_password`
15//! reads its parameters from an untrusted string, so it accepts only a bounded set (argon2id or
16//! libsodium's argon2i, `v=19`, costs up to [`Strength::Sensitive`] - `docs/DECISIONS.md` D-222).
17//!
18//! # Example
19//!
20//! For hashing passwords before storing them (never a general-purpose hash - deliberately slow and
21//! memory-hard so guessing many candidate passwords against a stolen hash is expensive).
22//! [`Strength::Interactive`] is used below so this example runs quickly; a real login system would
23//! usually want [`Strength::Moderate`] or [`Strength::Sensitive`] instead (both take real seconds
24//! and hundreds of MiB, deliberately, so not run here).
25//!
26//! ```rust
27//! use dstu_core::crypto_pwhash::{hash_password, verify_password, Strength};
28//!
29//! let stored_hash = hash_password(b"correct horse battery staple", Strength::Interactive)
30//!     .expect("OS CSPRNG should not fail");
31//!
32//! assert!(verify_password(b"correct horse battery staple", &stored_hash));
33//! assert!(!verify_password(b"wrong guess", &stored_hash));
34//! ```
35

41/// Argon2id cost preset - mirrors libsodium's `crypto_pwhash_argon2id` `OPSLIMIT`/`MEMLIMIT`
42/// named constants (`crypto_pwhash_argon2id.h`) exactly, not an independently chosen value.
43#[derive(Clone, Copy, Debug, PartialEq, Eq)]
44pub enum Strength {
45    /// `OPSLIMIT_INTERACTIVE` / `MEMLIMIT_INTERACTIVE` (t=2, 64 MiB) - online, latency-sensitive.
46    Interactive,
47    /// `OPSLIMIT_MODERATE` / `MEMLIMIT_MODERATE` (t=3, 256 MiB).
48    Moderate,
49    /// `OPSLIMIT_SENSITIVE` / `MEMLIMIT_SENSITIVE` (t=4, 1024 MiB) - highly sensitive, offline-ok.
50    Sensitive,
51}
52
53impl Strength {
54    /// `(m_cost in KiB, t_cost)` - libsodium's own `OPSLIMIT`/`MEMLIMIT_*` constants
55    /// (`crypto_pwhash_argon2id.h`), `MEMLIMIT` converted from bytes to KiB. `p_cost` is fixed at
56    /// 1 lane below, not part of this preset - libsodium hardcodes it the same way.
57    const fn m_and_t_cost(self) -> (u32, u32) {
58        match self {
59            Strength::Interactive => (65536, 2),   // 64 MiB
60            Strength::Moderate => (262_144, 3),    // 256 MiB
61            Strength::Sensitive => (1_048_576, 4), // 1024 MiB
62        }
63    }
64
65    /// Builds the actual `argon2::Params` for this preset. A `const fn`: an invalid preset would
66    /// be a compile-time error, not a runtime one - these three presets are the only values this
67    /// type can hold, so validity is closed over at compile time rather than checked per call.
68    const fn params(self) -> Params {
69        let (m_cost, t_cost) = self.m_and_t_cost();
70        match Params::new(m_cost, t_cost, 1, None) {
71            Ok(p) => p,
72            Err(_) => panic!("Strength's own hardcoded (m_cost, t_cost, 1) is always valid"),
73        }
74    }
75
76    /// This preset's `(m_cost in KiB, t_cost, p_cost)`, for a file format that records them
77    /// (`uacrypt`'s passphrase container, `docs/DECISIONS.md` D-219).
78    #[must_use]
79    pub const fn cost(self) -> (u32, u32, u32) {
80        let (m_cost, t_cost) = self.m_and_t_cost();
81        (m_cost, t_cost, 1)
82    }
83
84    /// The preset whose [`cost`](Self::cost) is exactly `(m_cost, t_cost, p_cost)`, or `None`. A
85    /// reader of untrusted parameters uses this to accept only the known presets instead of
86    /// running Argon2 with attacker-chosen memory.
87    #[must_use]
88    pub fn from_cost(m_cost: u32, t_cost: u32, p_cost: u32) -> Option<Self> {
89        [
90            Strength::Interactive,
91            Strength::Moderate,
92            Strength::Sensitive,
93        ]
94        .into_iter()
95        .find(|s| s.cost() == (m_cost, t_cost, p_cost))
96    }
97}

105/// `crypto_pwhash_str`/`_str_verify` can fail for reasons unrelated to a wrong password: this
106/// covers those.
107#[derive(Debug)]
108pub enum PwHashError {
109    /// The OS CSPRNG failed while generating a salt (see [`crate::randombytes`]).
110    Random(RandomError),
111    /// Argon2/PHC-string encoding failed. Not expected to occur in practice for the fixed-length
112    /// salt this module always generates - included because the underlying crate's API is
113    /// fallible, not because a known failure mode exists here.
114    Hash(argon2::password_hash::Error),
115    /// [`derive_key`] could not allocate the preset's Argon2 memory (256 MiB for
116    /// [`Strength::Moderate`]) - reported instead of aborting the process.
117    OutOfMemory,
118    /// [`derive_key`]'s raw Argon2 call failed. Not expected for the fixed presets, salt and
119    /// output length used here.
120    Derive(argon2::Error),
121}
122
123impl fmt::Display for PwHashError {
124    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
125        match self {
126            PwHashError::Random(e) => write!(f, "{e}"),
127            PwHashError::Hash(e) => write!(f, "Argon2 hashing failed: {e}"),
128            PwHashError::OutOfMemory => write!(f, "not enough memory for Argon2"),
129            PwHashError::Derive(e) => write!(f, "Argon2 key derivation failed: {e}"),
130        }
131    }
132}

134impl core::error::Error for PwHashError {}

148/// Hashes `password` into a self-describing PHC string (`$argon2id$v=19$m=...,t=...,p=1$<salt>$
149/// <hash>`) - libsodium's `crypto_pwhash_str` equivalent. A fresh 16-byte salt
150/// (`crypto_pwhash_argon2id_SALTBYTES`) is drawn per call via
151/// [`crate::randombytes::randombytes_buf`], never `password_hash`'s own `rand_core`-based
152/// `SaltString::generate`, so this module pulls in no `CryptoRng` dependency of its own
153/// (`docs/DECISIONS.md` D-48/D-50).
154///
155/// # Errors
156///
157/// Returns [`PwHashError::Random`] if the OS CSPRNG fails. [`PwHashError::Hash`] is not expected
158/// for the fixed 16-byte salt generated here.
159pub fn hash_password(password: &[u8], strength: Strength) -> Result<String, PwHashError> {
160    let mut salt_bytes = [0u8; 16];
161    randombytes_buf(&mut salt_bytes)?;
162    let salt = SaltString::encode_b64(&salt_bytes)?;
163    let argon2 = Argon2::new(Algorithm::Argon2id, Version::V0x13, strength.params());
164    let hash = argon2.hash_password(password, &salt)?;
165    Ok(hash.to_string())
166}

168/// Largest `m_cost` (KiB) [`verify_password`] accepts: the [`Strength::Sensitive`] preset.
169const VERIFY_MAX_M_COST: u32 = 1_048_576;
170
171/// Largest `m_cost * t_cost` [`verify_password`] accepts: the work of [`Strength::Sensitive`]
172/// (1 GiB x 4), which libsodium's argon2i `SENSITIVE` preset (512 MiB x 8) also lands on exactly.
173const VERIFY_MAX_WORK: u64 = 4 * 1_048_576;
174
175/// Shortest tag [`verify_password`] accepts - libsodium's `ARGON2_MIN_OUTLEN`.
176const VERIFY_MIN_TAG_LEN: usize = 16;
177
178/// Checks everything [`verify_password`] takes from an untrusted PHC string before any memory is
179/// allocated (`docs/DECISIONS.md` D-222) and returns the Argon2 context to run.
180fn verify_context(
181    algorithm: &str,
182    version: Option<u32>,
183    (m_cost, t_cost, p_cost): (u32, u32, u32),
184    tag_len: usize,
185    salt_len: usize,
186) -> Option<Argon2<'static>> {
187    let algorithm = match algorithm {
188        "argon2id" => Algorithm::Argon2id,
189        "argon2i" => Algorithm::Argon2i,
190        _ => return None,
191    };
192    if version != Some(0x13)
193        || m_cost > VERIFY_MAX_M_COST
194        || u64::from(m_cost) * u64::from(t_cost) > VERIFY_MAX_WORK
195        || !(VERIFY_MIN_TAG_LEN..=argon2::password_hash::Output::MAX_LENGTH).contains(&tag_len)
196        || salt_len < argon2::MIN_SALT_LEN
197        // `Params::new` computes `p_cost * 8` before its own `MAX_P_COST` check: an overflow panic
198        // with overflow checks on (T-272's fuzz target found it), so reject that range first.
199        || p_cost > Params::MAX_P_COST
200    {
201        return None;
202    }
203    let params = Params::new(m_cost, t_cost, p_cost, Some(tag_len)).ok()?;
204    Some(Argon2::new(algorithm, Version::V0x13, params))
205}

207/// The `m`, `t` and `p` of a PHC string, each exactly once and nothing else.
208fn phc_costs(parsed: &PasswordHash<'_>) -> Option<(u32, u32, u32)> {
209    let (mut m, mut t, mut p) = (None, None, None);
210    for (ident, value) in parsed.params.iter() {
211        let slot = match ident.as_str() {
212            "m" => &mut m,
213            "t" => &mut t,
214            "p" => &mut p,
215            _ => return None,
216        };
217        if slot.replace(value.decimal().ok()?).is_some() {
218            return None;
219        }
220    }
221    Some((m?, t?, p?))
222}

224/// Verifies `password` against a PHC string produced by [`hash_password`] - libsodium's
225/// `crypto_pwhash_str_verify` equivalent. Returns `false` for both a wrong password and a
226/// malformed/unparseable hash string, mirroring libsodium's own single pass/fail return (nothing
227/// for a caller to mishandle by branching differently on the two failure cases).
228///
229/// The string is untrusted, so only what libsodium itself verifies is accepted, within this
230/// module's own costs (`docs/DECISIONS.md` D-222): `argon2id` or `argon2i`, `v=19`, exactly the
231/// `m`/`t`/`p` parameters, `m` at most the [`Strength::Sensitive`] preset's 1 GiB, `m * t` at most
232/// that preset's work, a salt of at least 8 bytes and a tag of 16 to 64 bytes. Anything else is
233/// `false` before any memory is allocated, and a machine that cannot allocate an accepted `m` gets
234/// `false` rather than an aborted process.
235#[must_use]
236pub fn verify_password(password: &[u8], hash: &str) -> bool {
237    use subtle::ConstantTimeEq;
238    use zeroize::Zeroize;
239
240    let Ok(parsed) = PasswordHash::new(hash) else {
241        return false;
242    };
243    let (Some(salt), Some(expected)) = (parsed.salt, parsed.hash) else {
244        return false;
245    };
246    let Some(costs) = phc_costs(&parsed) else {
247        return false;
248    };
249    let mut salt_buf = [0u8; argon2::password_hash::Salt::MAX_LENGTH];
250    let Ok(salt) = salt.decode_b64(&mut salt_buf) else {
251        return false;
252    };
253    let Some(argon2) = verify_context(
254        parsed.algorithm.as_str(),
255        parsed.version,
256        costs,
257        expected.len(),
258        salt.len(),
259    ) else {
260        return false;
261    };
262
263    let block_count = argon2.params().block_count();
264    let mut blocks = Vec::new();
265    if blocks.try_reserve_exact(block_count).is_err() {
266        return false;
267    }
268    blocks.resize(block_count, argon2::Block::default());
269
270    let mut tag_buf = zeroize::Zeroizing::new([0u8; argon2::password_hash::Output::MAX_LENGTH]);
271    let tag = &mut tag_buf[..expected.len()];
272    let result = argon2.hash_password_into_with_memory(password, salt, tag, &mut blocks);
273    blocks.iter_mut().for_each(Zeroize::zeroize);
274    result.is_ok() && bool::from(tag.ct_eq(expected.as_bytes()))
275}

277/// Derives a 32-byte key from `password` and `salt` - libsodium's `crypto_pwhash` equivalent
278/// (Argon2id v1.3, one lane, no secret, no associated data), byte-identical to libsodium for the
279/// same preset (`tests/vectors/pwhash/derive_key.json`, `docs/DECISIONS.md` D-219). The caller
280/// stores the salt (and the preset's [`Strength::cost`]) next to the ciphertext; a fresh random
281/// salt per encryption is the caller's job.
282///
283/// The Argon2 memory is reserved with `try_reserve_exact`, so a machine without the preset's
284/// memory gets [`PwHashError::OutOfMemory`] rather than an aborted process, and it is zeroized
285/// before being freed.
286///
287/// ```rust
288/// use dstu_core::crypto_pwhash::{derive_key, Strength, SALT_BYTES};
289///
290/// let salt = [7u8; SALT_BYTES]; // store this next to the ciphertext; random in real use
291/// let key = derive_key(b"correct horse battery staple", &salt, Strength::Interactive)?;
292/// assert_eq!(key.len(), 32);
293/// # Ok::<(), dstu_core::crypto_pwhash::PwHashError>(())
294/// ```
295///
296/// # Errors
297///
298/// [`PwHashError::OutOfMemory`] if the preset's memory cannot be allocated;
299/// [`PwHashError::Derive`] is not expected for the fixed parameters used here.
300pub fn derive_key(
301    password: &[u8],
302    salt: &[u8; SALT_BYTES],
303    strength: Strength,
304) -> Result<zeroize::Zeroizing<[u8; KEY_BYTES]>, PwHashError> {
305    use zeroize::Zeroize;
306
307    let params = strength.params();
308    let block_count = params.block_count();
309    let argon2 = Argon2::new(Algorithm::Argon2id, Version::V0x13, params);
310
311    let mut blocks = Vec::new();
312    blocks
313        .try_reserve_exact(block_count)
314        .map_err(|_| PwHashError::OutOfMemory)?;
315    blocks.resize(block_count, argon2::Block::default());
316
317    let mut key = zeroize::Zeroizing::new([0u8; KEY_BYTES]);
318    let result =
319        argon2.hash_password_into_with_memory(password, salt, key.as_mut_slice(), &mut blocks);
320    blocks.iter_mut().for_each(Zeroize::zeroize);
321    result.map_err(PwHashError::Derive)?;
322    Ok(key)
323}

330    /// Confirms the `argon2` dependency itself is spec-correct before trusting it through this
331    /// module's wrapper - RFC 9106 (IETF, primary source) Appendix A's Argon2id test vector,
332    /// cited verbatim. Deliberately bypasses `hash_password`/PHC-string encoding: the vector's own
333    /// `p=4` doesn't match this module's fixed `p=1` (libsodium's own choice, see the module doc),
334    /// so a raw `Argon2` context is built directly against the vector's exact parameters instead.
335    #[test]
336    fn argon2_dependency_matches_rfc9106_argon2id_vector() -> Result<(), argon2::Error> {
337        let password = [0x01u8; 32];
338        let salt = [0x02u8; 16];
339        let secret = [0x03u8; 8];
340        let associated_data = AssociatedData::new(&[0x04u8; 12])?;
341
342        let mut builder = argon2::ParamsBuilder::new();
343        builder
344            .m_cost(32)
345            .t_cost(3)
346            .p_cost(4)
347            .data(associated_data)
348            .output_len(32);
349        let params = builder.build()?;
350
351        let argon2 = Argon2::new_with_secret(&secret, Algorithm::Argon2id, Version::V0x13, params)?;
352        let mut out = [0u8; 32];
353        argon2.hash_password_into(&password, &salt, &mut out)?;
354
355        assert_eq!(
356            out,
357            [
358                0x0d, 0x64, 0x0d, 0xf5, 0x8d, 0x78, 0x76, 0x6c, 0x08, 0xc0, 0x37, 0xa3, 0x4a, 0x8b,
359                0x53, 0xc9, 0xd0, 0x1e, 0xf0, 0x45, 0x2d, 0x75, 0xb6, 0x5e, 0xb5, 0x25, 0x20, 0xe9,
360                0x6b, 0x01, 0xe6, 0x59,
361            ]
362        );
363        Ok(())
364    }

366    /// The bounds of `verify_context` at each edge, without running Argon2: a real hash at the
367    /// memory/work bound would cost 1 GiB and seconds, so the vector file cannot reach them
368    /// (`docs/DECISIONS.md` D-222).
369    #[test]
370    fn verify_context_accepts_exactly_up_to_each_bound() {
371        let ok = |alg, ver, costs, tag, salt| verify_context(alg, ver, costs, tag, salt).is_some();
372        let v19 = Some(0x13);
373
374        assert!(ok("argon2id", v19, Strength::Sensitive.cost(), 32, 16));
375        assert!(
376            ok("argon2i", v19, (524_288, 8, 1), 32, 16),
377            "libsodium argon2i SENSITIVE"
378        );
379        assert!(!ok("argon2id", v19, (1_048_577, 1, 1), 32, 16));
380        assert!(!ok("argon2id", v19, (1_048_576, 5, 1), 32, 16));
381        assert!(ok("argon2id", v19, (65_536, 64, 1), 32, 16));
382        assert!(!ok("argon2id", v19, (65_536, 65, 1), 32, 16));
383        assert!(
384            !ok("argon2id", v19, (1_048_576, u32::MAX, 1), 32, 16),
385            "u64 product"
386        );
387        assert!(!ok("argon2id", v19, (8, u32::MAX, 1), 32, 16));
388
389        assert!(!ok("argon2id", v19, (8, 1, 1), 15, 16));
390        assert!(ok("argon2id", v19, (8, 1, 1), 16, 16));
391        assert!(ok("argon2id", v19, (8, 1, 1), 64, 16));
392        assert!(!ok("argon2id", v19, (8, 1, 1), 65, 16));
393        assert!(!ok("argon2id", v19, (8, 1, 1), 32, 7));
394        assert!(ok("argon2id", v19, (8, 1, 1), 32, 8));
395
396        assert!(
397            ok("argon2id", v19, (1_048_576, 1, 64), 32, 16),
398            "no cap on lanes"
399        );
400        assert!(!ok("argon2id", v19, (8, 1, 2), 32, 16), "RFC 9106: m >= 8p");
401        assert!(!ok("argon2id", v19, (8, 0, 1), 32, 16));
402        assert!(!ok("argon2id", v19, (8, 1, 0), 32, 16));
403        assert!(
404            !ok("argon2id", v19, (8, 1, u32::MAX), 32, 16),
405            "argon2's own 8p overflows"
406        );
407        assert!(!ok("argon2d", v19, (8, 1, 1), 32, 16));
408        assert!(!ok("argon2id", Some(0x10), (8, 1, 1), 32, 16));
409        assert!(!ok("argon2id", None, (8, 1, 1), 32, 16));
410    }
