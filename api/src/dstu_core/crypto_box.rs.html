1//! `crypto_box` equivalent (`docs/dstu-crypto-project.md` "Mapping onto the libsodium API",
2//! `docs/TASKS.md` T-178) - public-key encryption over `hazmat::dstu9041` (`l(p)=256`, E256/1
3//! only, T-177).
4//!
5//! # Hybrid via KDF - why, not just "how"
6//!
7//! `hazmat::dstu9041`'s `l(p)=256` variant caps a single ciphertext's payload at `L_MAX_P` = 200
8//! bits (25 bytes, Table 1) - far below this project's existing 32-byte symmetric keys
9//! (`crypto_secretbox::SecretKey`, `crypto_secretstream::Key`). This is not a defect: every
10//! asymmetric-encryption standard of this shape (RSA-OAEP, ECIES, RSA-KEM) is a **KEM**, meant to
11//! wrap a short secret, not to encrypt bulk data directly - OpenSSL's own `EVP_Seal*`/`EVP_Open*`
12//! ("digital envelope") and libsodium's `crypto_box_seal` both follow exactly this shape: the
13//! asymmetric step only ever establishes key material, a symmetric cipher does the actual work.
14//! `seal` draws a fresh random 25-byte (200-bit, `L_MAX_P` exactly) seed, wraps it to the
15//! recipient with `hazmat::dstu9041::encryption::encrypt`, then derives a 32-byte
16//! `crypto_secretstream::Key` as `Kupyna256Kmac(key = seed, message = b"cryptbox" || version ||
17//! kem_ciphertext || recipient_public_key)` (embedding the 25-byte seed into the low-order bytes
18//! of a zero-padded 32-byte buffer - `crypto_sign::derive_nonce`'s own "an embedding, not a
19//! truncation, no information lost" precedent). Binding the KEM ciphertext and the recipient key
20//! into the derivation follows RFC 9180 §4.1's DHKEM `kem_context = concat(enc, pkRm)` (T-248,
21//! `docs/DECISIONS.md` D-201); before 0.4.0 the key came from the seed alone.
22//! `crypto_secretstream` then encrypts the actual message, of any length, in one `Tag::Final`
23//! chunk (`seal`/`open` are one-shot, matching `crypto_secretbox`'s own `Vec<u8>` convention - a
24//! later genuinely multi-chunk `seal_stream`/`open_stream` pair could reuse this same KEM-prefix
25//! format without changing it).
26//!
27//! Wire format: `version (1 byte, currently 2) || dstu9041_ciphertext (128 bytes) ||
28//! secretstream_header (32 bytes) || ciphertext (message.len() bytes) || tag (16 bytes)`. The
29//! version byte (T-232/D-202) makes a blob from another format fail with
30//! [`OpenError::UnsupportedVersion`] rather than an unexplained authentication failure.
31//!
32//! # `PublicKey` is 32 bytes - the curve point's `x`-coordinate only
33//!
34//! Not `x || y` (64 bytes). This is safe by an explicit group-theory argument, not an assumption:
35//! this curve's negation is `-(x,y) = (x,-y)` (the swapped-Edwards form, `docs/pseudocode/
36//! dstu9041.md`), so `x` never distinguishes a point `Q` from its negation `-Q`, and `x_T = x_{-T}`
37//! holds for any point `T` on this curve. Since `k*(-Q) = -(k*Q)` for any scalar `k`, the two
38//! possible reconstructions of `Q` from just `x_Q` give the *same* `kappa = x_{epsilon*Q}` on
39//! `seal`'s own encrypt step, regardless of which square-root branch
40//! [`crate::hazmat::dstu9041::curve256::point_from_x`] happens to return - see that function's own
41//! doc comment, and `tests/dstu9041_curve.rs`'s `point_from_x_gives_same_kappa_regardless_of_sqrt_branch`
42//! for the arithmetic proof. `PublicKey::from_bytes` runs the exact same reconstruction gauntlet
43//! `hazmat::dstu9041::encryption::decrypt` already runs (reject `x in {0,1,p-1}`, reject
44//! `x^2=a*d^-1`, `euler_criterion` before `sqrt`, subgroup check) via that shared helper - not a
45//! second, independently-maintained copy of a security-critical check.
46//!
47//! # Error collapsing
48//!
49//! [`OpenError`] deliberately does not distinguish a KEM failure from a secretstream tag failure
50//! from a recovered-but-wrong-length seed - same padding-oracle-avoidance posture as
51//! `hazmat::dstu9041::encryption::DecryptError` (D-56/D-63 precedent). Only [`OpenError::Truncated`]
52//! (a public wire-length check, no secret-dependent data involved) and
53//! [`OpenError::UnsupportedVersion`] (a public version byte, checked before any secret is used)
54//! stay separate variants.
55//!
56//! # Provenance
57//!
58//! This composite construction (KEM + KDF + secretstream) is not itself DSTU-specified - like
59//! `crypto_secretstream` (D-68), there is no vector oracle for it, ever; verified by
60//! property/tamper/misuse tests only. `hazmat::dstu9041::encryption` itself remains verified
61//! against the standard's own worked example (T-177).
62//!
63//! # Example
64//!
65//! ```rust
66//! use dstu_core::crypto_box::{seal, open, SecretKey};
67//!
68//! # if cfg!(miri) { return; } // several 256-iteration curve256::Point::scalar_multiply calls
69//! # // (keygen, KEM encrypt/decrypt) - minutes each under Miri's interpreter, same reasoning as
70//! # // crypto_sign's own doctest guard; type-checked normally, just not executed there.
71//! let secret = SecretKey::generate().expect("OS CSPRNG should not fail");
72//! let public = secret.public_key(); // safe to share/publish
73//!
74//! let sealed = seal(b"a message for the public key's holder only", &public)
75//!     .expect("OS CSPRNG should not fail");
76//! let opened = open(&sealed, &secret).expect("authentic ciphertext under the matching key");
77//! assert_eq!(opened, b"a message for the public key's holder only");
78//!
79//! // Tampering with the sealed blob (KEM prefix, header, ciphertext, or tag) is detected.
80//! let mut tampered = sealed.clone();
81//! let last = tampered.len() - 1;
82//! tampered[last] ^= 1;
83//! assert!(open(&tampered, &secret).is_err());
84//! ```

98const SEED_LEN: usize = L_MAX_P / 8;
99const KEM_CIPHERTEXT_LEN: usize = 128;
100const HEADER_LEN: usize = 32;
101const TAG_LEN: usize = 16;
102/// Wire-format version, the sealed blob's first byte (T-232/T-248; 0.3.x blobs had none).
103const FORMAT_VERSION: u8 = 2;
104/// `version || kem_ciphertext` - everything in front of the secretstream header.
105const PREFIX_LEN: usize = 1 + KEM_CIPHERTEXT_LEN;
106const PUBLIC_KEY_LEN: usize = 32;
107/// Domain-separation label for the stream-key derivation - distinct from `crypto_box512`'s.
108const KDF_LABEL: &[u8; 8] = b"cryptbox";
109const KDF_MESSAGE_LEN: usize = KDF_LABEL.len() + 1 + KEM_CIPHERTEXT_LEN + PUBLIC_KEY_LEN;
110
111/// Derives the secretstream key as `Kupyna256Kmac(key = seed, message = label || version ||
112/// kem_ciphertext || recipient_public_key)`, binding the KEM ciphertext and the recipient into
113/// the key - RFC 9180 §4.1's DHKEM `kem_context = concat(enc, pkRm)` (T-248, `docs/DECISIONS.md`
114/// D-201). A plain `Kupyna256Kdf::derive_subkey` can't carry this: its context is a fixed 8 bytes.
115fn stream_key(
116    seed: &[u8; 32],
117    kem_ciphertext: &[u8; KEM_CIPHERTEXT_LEN],
118    recipient: &[u8; PUBLIC_KEY_LEN],
119) -> Key {
120    let mut message = [0u8; KDF_MESSAGE_LEN];
121    let (label, rest) = message.split_at_mut(KDF_LABEL.len());
122    label.copy_from_slice(KDF_LABEL);
123    let (version, rest) = rest.split_at_mut(1);
124    version[0] = FORMAT_VERSION;
125    let (kem, pk) = rest.split_at_mut(KEM_CIPHERTEXT_LEN);
126    kem.copy_from_slice(kem_ciphertext);
127    pk.copy_from_slice(recipient);
128    let Ok(mut key_bytes) = Kupyna256Kmac::mac(seed, &message) else {
129        unreachable!("seed is always exactly 32 bytes, Kupyna256Kmac's own mac_len")
130    };
131    let key = Key::from_bytes(key_bytes);
132    key_bytes.zeroize();
133    key
134}

158/// `crypto_box` can fail while opening for reasons beyond a wrong key.
159#[derive(Debug)]
160pub enum OpenError {
161    /// `sealed` is shorter than a version byte, a KEM ciphertext, a secretstream header and a tag
162    /// (177 bytes) - too short to have ever been produced by [`seal`].
163    Truncated,
164    /// `sealed` starts with a format version byte this build does not read - a blob from an older
165    /// or newer wire format. The byte is public, so this reveals nothing about the key.
166    UnsupportedVersion,
167    /// Any late-stage failure: wrong secret key, a tampered KEM prefix/header/ciphertext/tag, or a
168    /// recovered seed whose bit length isn't exactly `L_MAX_P` - deliberately collapsed, see the
169    /// module doc's "Error collapsing" section.
170    InvalidCiphertext,
171}
172
173impl fmt::Display for OpenError {
174    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
175        match self {
176            OpenError::Truncated => write!(f, "input too short to contain a sealed message"),
177            OpenError::InvalidCiphertext => write!(f, "authentication failed"),
178            OpenError::UnsupportedVersion => write!(f, "unsupported sealed-box format version"),
179        }
180    }
181}

185/// Rejection-samples a scalar in `{2, ..., n-2}` (`is_valid_scalar`) - shared by [`SecretKey::generate`]
186/// (a long-term key) and [`seal`] (a fresh ephemeral key per call). Mirrors
187/// `crypto_sign::SigningKey::generate`'s own pattern: zeroize each rejected candidate immediately,
188/// never a modulo reduction (`n` is not a power of two - that would bias small residues).
189fn random_valid_scalar() -> Result<[u8; 32], RandomError> {
190    loop {
191        let mut candidate = [0u8; 32];
192        randombytes_buf(&mut candidate)?;
193        if is_valid_scalar(&candidate) {
194            return Ok(candidate);
195        }
196        candidate.zeroize();
197    }
198}

200/// Embeds a 25-byte seed into the low-order bytes of a zero-padded 32-byte buffer - see the module
201/// doc's "Hybrid via KDF" section for why this, not a fresh KDF variant, is the right fix for the
202/// length mismatch.
203fn embed_seed(seed: &[u8; SEED_LEN]) -> [u8; 32] {
204    let mut embedded = [0u8; 32];
205    embedded[32 - SEED_LEN..].copy_from_slice(seed);
206    embedded
207}

209/// A `crypto_box` private key - the DSTU 9041 scalar `e`.
210pub struct SecretKey([u8; 32]);
211
212impl Drop for SecretKey {
213    fn drop(&mut self) {
214        self.0.zeroize();
215    }
216}
217
218impl SecretKey {
219    /// Builds a secret key from a big-endian 32-byte scalar. Returns `None` if it's outside the
220    /// valid range `{2, ..., n-2}` (`hazmat::dstu9041::curve256::is_valid_scalar`).
221    #[must_use]
222    pub fn from_bytes(e: &[u8; 32]) -> Option<Self> {
223        if is_valid_scalar(e) {
224            Some(SecretKey(*e))
225        } else {
226            None
227        }
228    }
229
230    /// Generates a fresh secret key from the OS CSPRNG - libsodium's `crypto_box_keypair()`
231    /// equivalent (its public-key half is [`Self::public_key`]).
232    ///
233    /// # Errors
234    ///
235    /// Returns [`RandomError`] if the OS CSPRNG fails while drawing a candidate.
236    #[cfg(any(feature = "std", feature = "getrandom"))]
237    pub fn generate() -> Result<Self, RandomError> {
238        random_valid_scalar().map(SecretKey)
239    }
240
241    /// Returns `e`'s big-endian 32-byte encoding, so a generated key can be persisted and later
242    /// reloaded via [`Self::from_bytes`]. The caller becomes responsible for zeroizing the
243    /// returned array once done with it.
244    #[must_use]
245    pub fn to_bytes(&self) -> [u8; 32] {
246        self.0
247    }
248
249    #[must_use]
250    pub fn public_key(&self) -> PublicKey {
251        PublicKey(base_point().scalar_multiply(&self.0))
252    }
253}

255/// A `crypto_box` public key - a curve point's `x`-coordinate only (32 bytes), see the module
256/// doc's own section on why this compression is safe.
257#[derive(Clone, Copy)]
258pub struct PublicKey(Point);
259
260impl PublicKey {
261    /// Reconstructs a public key from its compressed 32-byte `x`-coordinate encoding. Returns
262    /// `None` if `bytes` isn't a valid field element, or doesn't reconstruct to a point inside the
263    /// base point's own prime-order subgroup (`curve256::point_from_x`'s own rejection gauntlet).
264    #[must_use]
265    pub fn from_bytes(bytes: &[u8; 32]) -> Option<Self> {
266        from_candidate_bytes(bytes)
267            .and_then(point_from_x)
268            .map(PublicKey)
269    }
270
271    #[must_use]
272    pub fn to_bytes(&self) -> [u8; 32] {
273        self.0.x.to_be_bytes()
274    }
275}

277/// Encrypts `message` (any length) to `recipient`, drawing a fresh random seed and ephemeral key
278/// internally - see the module doc's "Hybrid via KDF" section.
279///
280/// # Errors
281///
282/// Returns [`SealError::Random`] if the OS CSPRNG fails - the only way this can fail.
283pub fn seal(message: &[u8], recipient: &PublicKey) -> Result<Vec<u8>, SealError> {
284    let mut seed = [0u8; SEED_LEN];
285    randombytes_buf(&mut seed)?;
286
287    let mut epsilon = random_valid_scalar()?;
288    // Unreachable: `seed` is always exactly SEED_LEN bytes at message_bits=L_MAX_P (can't trigger
289    // `InvalidMessage`), and `epsilon` is already validated by `random_valid_scalar` (can't
290    // trigger `InvalidEphemeralKey`).
291    let Ok(kem_ciphertext) = dstu9041_encrypt(&seed, L_MAX_P, recipient.0, &epsilon) else {
292        unreachable!("seed/epsilon are always valid by construction")
293    };
294    epsilon.zeroize();
295
296    let mut embedded = embed_seed(&seed);
297    seed.zeroize();
298    let key = stream_key(&embedded, &kem_ciphertext, &recipient.to_bytes());
299    embedded.zeroize();
300
301    let (mut push, header) = match PushState::init(&key) {
302        Ok(ok) => ok,
303        Err(SecretstreamError::Random(e)) => return Err(SealError::Random(e)),
304        // `PushState::init`'s only documented failure mode is the OS CSPRNG call for its header -
305        // see `crypto_secretstream::SecretstreamError`'s own doc comment.
306        Err(_) => unreachable!("PushState::init only ever fails via SecretstreamError::Random"),
307    };
308
309    let mut ciphertext = vec![0u8; message.len()];
310    let Ok(tag) = push.push(Tag::Final, message, &mut ciphertext) else {
311        unreachable!(
312            "ciphertext.len() == message.len() by construction; stream freshly initialized, \
313             never finalized yet"
314        )
315    };
316
317    let mut out = Vec::with_capacity(PREFIX_LEN + HEADER_LEN + ciphertext.len() + TAG_LEN);
318    out.push(FORMAT_VERSION);
319    out.extend_from_slice(&kem_ciphertext);
320    out.extend_from_slice(&header);
321    out.extend_from_slice(&ciphertext);
322    out.extend_from_slice(&tag);
323    Ok(out)
324}

326/// Decrypts `sealed` (as produced by [`seal`]) under `secret`.
327///
328/// # Errors
329///
330/// Returns [`OpenError::Truncated`] if `sealed` is shorter than the minimum possible length,
331/// [`OpenError::UnsupportedVersion`] if its first byte is not this build's format version, or
332/// [`OpenError::InvalidCiphertext`] for any other failure - see the module doc's "Error collapsing"
333/// section.
334pub fn open(sealed: &[u8], secret: &SecretKey) -> Result<Vec<u8>, OpenError> {
335    const MIN_LEN: usize = PREFIX_LEN + HEADER_LEN + TAG_LEN;
336    if sealed.len() < MIN_LEN {
337        return Err(OpenError::Truncated);
338    }
339    if sealed[0] != FORMAT_VERSION {
340        return Err(OpenError::UnsupportedVersion);
341    }
342
343    let mut kem_ciphertext = [0u8; KEM_CIPHERTEXT_LEN];
344    kem_ciphertext.copy_from_slice(&sealed[1..PREFIX_LEN]);
345    let mut header = [0u8; HEADER_LEN];
346    header.copy_from_slice(&sealed[PREFIX_LEN..PREFIX_LEN + HEADER_LEN]);
347    let ciphertext_len = sealed.len() - MIN_LEN;
348    let ciphertext_start = PREFIX_LEN + HEADER_LEN;
349    let ciphertext = &sealed[ciphertext_start..ciphertext_start + ciphertext_len];
350    let tag = &sealed[ciphertext_start + ciphertext_len..];
351
352    let (mut seed_padded, bit_len) =
353        dstu9041_decrypt(&kem_ciphertext, &secret.0).map_err(|_| OpenError::InvalidCiphertext)?;
354    // Defense in depth: an honestly-sealed ciphertext always has bit_len == L_MAX_P (the hash
355    // check inside `decrypt` already makes forging a different-but-valid bit_length as hard as a
356    // Kupyna-256 preimage), but this is never trusted blindly.
357    if bit_len != L_MAX_P {
358        seed_padded.zeroize();
359        return Err(OpenError::InvalidCiphertext);
360    }
361
362    let mut embedded = embed_seed(&seed_padded);
363    seed_padded.zeroize();
364    let key = stream_key(&embedded, &kem_ciphertext, &secret.public_key().to_bytes());
365    embedded.zeroize();
366
367    let mut pull = PullState::init(&key, &header);
368    let mut plaintext = vec![0u8; ciphertext_len];
369    pull.pull(Tag::Final.to_byte(), ciphertext, tag, &mut plaintext)
370        .map_err(|_| OpenError::InvalidCiphertext)?;
371
372    Ok(plaintext)
373}

375#[cfg(test)]
376mod tests {
377    use super::*;
378
379    /// T-248: the KEM ciphertext (first, middle and last byte), the recipient key (first and last
380    /// byte) and the seed all feed the stream key - the property a round-trip test cannot show.
381    #[test]
382    fn stream_key_binds_kem_ciphertext_recipient_and_seed() {
383        let seed = [0x11u8; 32];
384        let kem = [0x22u8; KEM_CIPHERTEXT_LEN];
385        let pk = [0x33u8; PUBLIC_KEY_LEN];
386        let base = *stream_key(&seed, &kem, &pk).as_bytes();
387
388        for i in [0, KEM_CIPHERTEXT_LEN / 2, KEM_CIPHERTEXT_LEN - 1] {
389            let mut kem2 = kem;
390            kem2[i] ^= 1;
391            assert_ne!(
392                *stream_key(&seed, &kem2, &pk).as_bytes(),
393                base,
394                "kem byte {i}"
395            );
396        }
397        for i in [0, PUBLIC_KEY_LEN - 1] {
398            let mut pk2 = pk;
399            pk2[i] ^= 1;
400            assert_ne!(
401                *stream_key(&seed, &kem, &pk2).as_bytes(),
402                base,
403                "pk byte {i}"
404            );
405        }
406        let mut seed2 = seed;
407        seed2[0] ^= 1;
408        assert_ne!(*stream_key(&seed2, &kem, &pk).as_bytes(), base);
409        assert_eq!(*stream_key(&seed, &kem, &pk).as_bytes(), base);
410    }
