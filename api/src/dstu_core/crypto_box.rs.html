1//! `crypto_box` equivalent (`docs/dstu-crypto-project.md` "Mapping onto the libsodium API",
2//! `docs/TASKS.md` T-178) - public-key encryption over `hazmat::dstu9041` (`l(p)=256`, E256/1
3//! only, T-177).
4//!
5//! # Hybrid via KDF - why, not just "how"
6//!
7//! `hazmat::dstu9041`'s `l(p)=256` variant caps a single ciphertext's payload at `L_MAX_P` = 200
8//! bits (25 bytes, Table 1) - far below this project's existing 32-byte symmetric keys
9//! (`crypto_secretbox::SecretKey`, `crypto_secretstream::Key`). This is not a defect: every
10//! asymmetric-encryption standard of this shape (RSA-OAEP, ECIES, RSA-KEM) is a **KEM**, meant to
11//! wrap a short secret, not to encrypt bulk data directly - OpenSSL's own `EVP_Seal*`/`EVP_Open*`
12//! ("digital envelope") and libsodium's `crypto_box_seal` both follow exactly this shape: the
13//! asymmetric step only ever establishes key material, a symmetric cipher does the actual work.
14//! `seal` draws a fresh random 25-byte (200-bit, `L_MAX_P` exactly) seed, wraps it to the
15//! recipient with `hazmat::dstu9041::encryption::encrypt`, then derives a 32-byte
16//! `crypto_secretstream::Key` as `Kupyna256Kmac(key = seed, message = b"cryptbox" || version ||
17//! kem_ciphertext || recipient_public_key)` (embedding the 25-byte seed into the low-order bytes
18//! of a zero-padded 32-byte buffer - `crypto_sign::derive_nonce`'s own "an embedding, not a
19//! truncation, no information lost" precedent). Binding the KEM ciphertext and the recipient key
20//! into the derivation follows RFC 9180 §4.1's DHKEM `kem_context = concat(enc, pkRm)` (T-248,
21//! `docs/DECISIONS.md` D-201); before 0.4.0 the key came from the seed alone.
22//! `crypto_secretstream` then encrypts the actual message, of any length, as a D-208 stream file
23//! (T-265): [`seal_stream`]/[`open_stream`] hand out the stream-file codec's
24//! [`FileEncryptor`]/[`FileDecryptor`], so a message never has to fit in memory; [`seal`]/[`open`]
25//! are the same format in one call.
26//!
27//! Wire format (version 3, D-239): `version (1 byte, 3) || dstu9041_ciphertext (128 bytes) ||
28//! <stream file: version (2) || header (32) || records>`, each record `tag (1) || length (u32 LE)
29//! || ciphertext || tag (16)` over at most 8192 plaintext bytes - the same container shape as
30//! `uacrypt`'s passphrase files (D-219). The version byte (T-232/D-202) makes a blob from another
31//! format fail with [`OpenError::UnsupportedVersion`] rather than an unexplained authentication
32//! failure; it is also bound into the stream key. The one-chunk version 2 that 0.4-0.6 wrote is
33//! read only by [`open_v2_legacy`].
34//!
35//! # `PublicKey` is 32 bytes - the curve point's `x`-coordinate only
36//!
37//! Not `x || y` (64 bytes). This is safe by an explicit group-theory argument, not an assumption:
38//! this curve's negation is `-(x,y) = (x,-y)` (the swapped-Edwards form, `docs/pseudocode/
39//! dstu9041.md`), so `x` never distinguishes a point `Q` from its negation `-Q`, and `x_T = x_{-T}`
40//! holds for any point `T` on this curve. Since `k*(-Q) = -(k*Q)` for any scalar `k`, the two
41//! possible reconstructions of `Q` from just `x_Q` give the *same* `kappa = x_{epsilon*Q}` on
42//! `seal`'s own encrypt step, regardless of which square-root branch
43//! [`crate::hazmat::dstu9041::curve256::point_from_x`] happens to return - see that function's own
44//! doc comment, and `tests/dstu9041_curve.rs`'s `point_from_x_gives_same_kappa_regardless_of_sqrt_branch`
45//! for the arithmetic proof. `PublicKey::from_bytes` runs the exact same reconstruction gauntlet
46//! `hazmat::dstu9041::encryption::decrypt` already runs (reject `x in {0,1,p-1}`, reject
47//! `x^2=a*d^-1`, `euler_criterion` before `sqrt`, subgroup check) via that shared helper - not a
48//! second, independently-maintained copy of a security-critical check.
49//!
50//! # Error collapsing
51//!
52//! [`OpenError`] deliberately does not distinguish a KEM failure from a secretstream tag failure
53//! from a recovered-but-wrong-length seed - same padding-oracle-avoidance posture as
54//! `hazmat::dstu9041::encryption::DecryptError` (D-56/D-63 precedent). Only [`OpenError::Truncated`]
55//! (a public wire-length check, no secret-dependent data involved) and
56//! [`OpenError::UnsupportedVersion`] (a public version byte, checked before any secret is used)
57//! stay separate variants.
58//!
59//! # Provenance
60//!
61//! This composite construction (KEM + KDF + secretstream) is not itself DSTU-specified - like
62//! `crypto_secretstream` (D-68), there is no vector oracle for it, ever; verified by
63//! property/tamper/misuse tests only. `hazmat::dstu9041::encryption` itself remains verified
64//! against the standard's own worked example (T-177).
65//!
66//! # Example
67//!
68//! ```rust
69//! use dstu_core::crypto_box::{seal, open, SecretKey};
70//!
71//! # if cfg!(miri) { return; } // several 256-iteration curve256::Point::scalar_multiply calls
72//! # // (keygen, KEM encrypt/decrypt) - minutes each under Miri's interpreter, same reasoning as
73//! # // crypto_sign's own doctest guard; type-checked normally, just not executed there.
74//! let secret = SecretKey::generate().expect("OS CSPRNG should not fail");
75//! let public = secret.public_key(); // safe to share/publish
76//!
77//! let sealed = seal(b"a message for the public key's holder only", &public)
78//!     .expect("OS CSPRNG should not fail");
79//! let opened = open(&sealed, &secret).expect("authentic ciphertext under the matching key");
80//! assert_eq!(opened, b"a message for the public key's holder only");
81//!
82//! // Tampering with the sealed blob (KEM prefix, header, ciphertext, or tag) is detected.
83//! let mut tampered = sealed.clone();
84//! let last = tampered.len() - 1;
85//! tampered[last] ^= 1;
86//! assert!(open(&tampered, &secret).is_err());
87//! ```

104const SEED_LEN: usize = L_MAX_P / 8;
105const KEM_CIPHERTEXT_LEN: usize = 128;
106const HEADER_LEN: usize = 32;
107const TAG_LEN: usize = 16;
108/// Wire-format version, the sealed blob's first byte (T-265, D-239: a D-208 stream file follows
109/// the KEM ciphertext).
110pub const FORMAT_VERSION: u8 = 3;
111/// The one-chunk format 0.4-0.6 wrote (T-232/T-248), read only by [`open_v2_legacy`].
112pub const LEGACY_FORMAT_VERSION: u8 = 2;
113/// `version || kem_ciphertext` - everything in front of the stream file.
114const PREFIX_LEN: usize = 1 + KEM_CIPHERTEXT_LEN;
115/// The bytes [`open_stream`] takes: the version byte and the KEM ciphertext.
116pub const OPEN_STREAM_PREFIX_BYTES: usize = PREFIX_LEN;
117/// The bytes [`seal_stream`] returns to write first: [`OPEN_STREAM_PREFIX_BYTES`], then the stream
118/// file's own version byte and header.
119pub const SEAL_STREAM_PREFIX_BYTES: usize = PREFIX_LEN + FILE_PREFIX_BYTES;
120/// The shortest sealed message: the prefix and one empty `Final` record.
121const MIN_SEALED_LEN: usize = SEAL_STREAM_PREFIX_BYTES + RECORD_OVERHEAD_BYTES;
122const PUBLIC_KEY_LEN: usize = 32;
123/// Domain-separation label for the stream-key derivation - distinct from `crypto_box512`'s.
124const KDF_LABEL: &[u8; 8] = b"cryptbox";
125const KDF_MESSAGE_LEN: usize = KDF_LABEL.len() + 1 + KEM_CIPHERTEXT_LEN + PUBLIC_KEY_LEN;
126
127/// Derives the secretstream key as `Kupyna256Kmac(key = seed, message = label || version ||
128/// kem_ciphertext || recipient_public_key)`, binding the KEM ciphertext and the recipient into
129/// the key - RFC 9180 §4.1's DHKEM `kem_context = concat(enc, pkRm)` (T-248, `docs/DECISIONS.md`
130/// D-201). A plain `Kupyna256Kdf::derive_subkey` can't carry this: its context is a fixed 8 bytes.
131/// `version` is the wire format's, so a v2 and a v3 blob never share a stream key (D-239).
132fn stream_key(
133    seed: &[u8; 32],
134    kem_ciphertext: &[u8; KEM_CIPHERTEXT_LEN],
135    recipient: &[u8; PUBLIC_KEY_LEN],
136    version: u8,
137) -> Key {
138    let mut message = [0u8; KDF_MESSAGE_LEN];
139    let (label, rest) = message.split_at_mut(KDF_LABEL.len());
140    label.copy_from_slice(KDF_LABEL);
141    let (version_byte, rest) = rest.split_at_mut(1);
142    version_byte[0] = version;
143    let (kem, pk) = rest.split_at_mut(KEM_CIPHERTEXT_LEN);
144    kem.copy_from_slice(kem_ciphertext);
145    pk.copy_from_slice(recipient);
146    let Ok(mut key_bytes) = Kupyna256Kmac::mac(seed, &message) else {
147        unreachable!("seed is always exactly 32 bytes, Kupyna256Kmac's own mac_len")
148    };
149    let key = Key::from_bytes(key_bytes);
150    key_bytes.zeroize();
151    key
152}

176/// `crypto_box` can fail while opening for reasons beyond a wrong key.
177#[derive(Debug)]
178pub enum OpenError {
179    /// `sealed` is shorter than the prefix and one empty record (183 bytes; 177 for
180    /// [`open_v2_legacy`]), or its stream ends before a verified `Final` record - it was cut.
181    Truncated,
182    /// `sealed` starts with a format version byte this build does not read - a blob from an older
183    /// or newer wire format. The byte is public, so this reveals nothing about the key.
184    UnsupportedVersion,
185    /// Any late-stage failure: wrong secret key, a tampered KEM prefix/header/ciphertext/tag, or a
186    /// recovered seed whose bit length isn't exactly `L_MAX_P` - deliberately collapsed, see the
187    /// module doc's "Error collapsing" section.
188    InvalidCiphertext,
189}
190
191impl fmt::Display for OpenError {
192    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
193        match self {
194            OpenError::Truncated => write!(f, "input too short to contain a sealed message"),
195            OpenError::InvalidCiphertext => write!(f, "authentication failed"),
196            OpenError::UnsupportedVersion => write!(f, "unsupported sealed-box format version"),
197        }
198    }
199}

203/// Rejection-samples a scalar in `{2, ..., n-2}` (`is_valid_scalar`) - shared by [`SecretKey::generate`]
204/// (a long-term key) and [`seal`] (a fresh ephemeral key per call). Mirrors
205/// `crypto_sign::SigningKey::generate`'s own pattern: zeroize each rejected candidate immediately,
206/// never a modulo reduction (`n` is not a power of two - that would bias small residues).
207fn random_valid_scalar() -> Result<[u8; 32], RandomError> {
208    loop {
209        let mut candidate = [0u8; 32];
210        randombytes_buf(&mut candidate)?;
211        if is_valid_scalar(&candidate) {
212            return Ok(candidate);
213        }
214        candidate.zeroize();
215    }
216}

218/// Embeds a 25-byte seed into the low-order bytes of a zero-padded 32-byte buffer - see the module
219/// doc's "Hybrid via KDF" section for why this, not a fresh KDF variant, is the right fix for the
220/// length mismatch.
221fn embed_seed(seed: &[u8; SEED_LEN]) -> [u8; 32] {
222    let mut embedded = [0u8; 32];
223    embedded[32 - SEED_LEN..].copy_from_slice(seed);
224    embedded
225}

227/// A `crypto_box` private key - the DSTU 9041 scalar `e`.
228pub struct SecretKey {
229    e: [u8; 32],
230    /// `e * P`, computed once here: `open` binds the recipient public key into the KDF
231    /// (D-201), and a full ladder per call was 28-30% of `open` (T-287).
232    public: PublicKey,
233}
234
235impl Drop for SecretKey {
236    fn drop(&mut self) {
237        self.e.zeroize();
238    }
239}
240
241impl SecretKey {
242    fn from_valid_scalar(e: [u8; 32]) -> Self {
243        let public = PublicKey(base_point().scalar_multiply(&e));
244        SecretKey { e, public }
245    }
246
247    /// Builds a secret key from a big-endian 32-byte scalar. Returns `None` if it's outside the
248    /// valid range `{2, ..., n-2}` (`hazmat::dstu9041::curve256::is_valid_scalar`).
249    #[must_use]
250    pub fn from_bytes(e: &[u8; 32]) -> Option<Self> {
251        if is_valid_scalar(e) {
252            Some(Self::from_valid_scalar(*e))
253        } else {
254            None
255        }
256    }
257
258    /// Generates a fresh secret key from the OS CSPRNG - libsodium's `crypto_box_keypair()`
259    /// equivalent (its public-key half is [`Self::public_key`]).
260    ///
261    /// # Errors
262    ///
263    /// Returns [`RandomError`] if the OS CSPRNG fails while drawing a candidate.
264    #[cfg(any(feature = "std", feature = "getrandom"))]
265    pub fn generate() -> Result<Self, RandomError> {
266        random_valid_scalar().map(Self::from_valid_scalar)
267    }
268
269    /// Returns `e`'s big-endian 32-byte encoding, so a generated key can be persisted and later
270    /// reloaded via [`Self::from_bytes`]. The caller becomes responsible for zeroizing the
271    /// returned array once done with it.
272    #[must_use]
273    pub fn to_bytes(&self) -> [u8; 32] {
274        self.e
275    }
276
277    #[must_use]
278    pub fn public_key(&self) -> PublicKey {
279        self.public
280    }
281}

283/// A `crypto_box` public key - a curve point's `x`-coordinate only (32 bytes), see the module
284/// doc's own section on why this compression is safe.
285#[derive(Clone, Copy)]
286pub struct PublicKey(Point);
287
288impl PublicKey {
289    /// Reconstructs a public key from its compressed 32-byte `x`-coordinate encoding. Returns
290    /// `None` if `bytes` isn't a valid field element, or doesn't reconstruct to a point inside the
291    /// base point's own prime-order subgroup (`curve256::point_from_x`'s own rejection gauntlet).
292    #[must_use]
293    pub fn from_bytes(bytes: &[u8; 32]) -> Option<Self> {
294        from_candidate_bytes(bytes)
295            .and_then(point_from_x)
296            .map(PublicKey)
297    }
298
299    #[must_use]
300    pub fn to_bytes(&self) -> [u8; 32] {
301        self.0.x.to_be_bytes()
302    }
303}

305/// The KEM half of every sealing entry point: a fresh random seed wrapped to `recipient` under a
306/// fresh ephemeral key, and the stream key derived from it for wire format `version` - see the
307/// module doc's "Hybrid via KDF" section.
308fn encapsulate(
309    recipient: &PublicKey,
310    version: u8,
311) -> Result<([u8; KEM_CIPHERTEXT_LEN], Key), RandomError> {
312    let mut seed = [0u8; SEED_LEN];
313    randombytes_buf(&mut seed)?;
314    let mut epsilon = match random_valid_scalar() {
315        Ok(epsilon) => epsilon,
316        Err(e) => {
317            seed.zeroize();
318            return Err(e);
319        }
320    };
321    // Unreachable: `seed` is always exactly SEED_LEN bytes at message_bits=L_MAX_P (can't trigger
322    // `InvalidMessage`), and `epsilon` is already validated by `random_valid_scalar` (can't
323    // trigger `InvalidEphemeralKey`).
324    let Ok(kem_ciphertext) = dstu9041_encrypt(&seed, L_MAX_P, recipient.0, &epsilon) else {
325        unreachable!("seed/epsilon are always valid by construction")
326    };
327    epsilon.zeroize();
328
329    let mut embedded = embed_seed(&seed);
330    seed.zeroize();
331    let key = stream_key(&embedded, &kem_ciphertext, &recipient.to_bytes(), version);
332    embedded.zeroize();
333    Ok((kem_ciphertext, key))
334}

336/// The KEM half of every opening entry point: recovers the seed under `secret` and derives the
337/// stream key for wire format `version`. Every failure is [`OpenError::InvalidCiphertext`].
338fn decapsulate(
339    kem_ciphertext: &[u8; KEM_CIPHERTEXT_LEN],
340    secret: &SecretKey,
341    version: u8,
342) -> Result<Key, OpenError> {
343    let (mut seed_padded, bit_len) =
344        dstu9041_decrypt(kem_ciphertext, &secret.e).map_err(|_| OpenError::InvalidCiphertext)?;
345    // Defense in depth: an honestly-sealed ciphertext always has bit_len == L_MAX_P (the hash
346    // check inside `decrypt` already makes forging a different-but-valid bit_length as hard as a
347    // Kupyna-256 preimage), but this is never trusted blindly.
348    if bit_len != L_MAX_P {
349        seed_padded.zeroize();
350        return Err(OpenError::InvalidCiphertext);
351    }
352
353    let mut embedded = embed_seed(&seed_padded);
354    seed_padded.zeroize();
355    let key = stream_key(
356        &embedded,
357        kem_ciphertext,
358        &secret.public_key().to_bytes(),
359        version,
360    );
361    embedded.zeroize();
362    Ok(key)
363}

365/// Starts sealing a message of any length to `recipient` as a stream (T-265): returns the
366/// [`FileEncryptor`] for its records and the [`SEAL_STREAM_PREFIX_BYTES`]-byte prefix to write
367/// first. Drive the encryptor as the stream-file codec documents (every full
368/// [`CHUNK_BYTES`] chunk with `seal_chunk`, the last `0..=CHUNK_BYTES` bytes with `seal_final`);
369/// the bytes are then exactly what [`seal`] returns, and [`open`]/[`open_stream`] read them. The
370/// stream key never leaves the library.
371///
372/// # Errors
373///
374/// Returns [`SealError::Random`] if the OS CSPRNG fails - the only way this can fail.
375pub fn seal_stream(
376    recipient: &PublicKey,
377) -> Result<(FileEncryptor, [u8; SEAL_STREAM_PREFIX_BYTES]), SealError> {
378    let (kem_ciphertext, key) = encapsulate(recipient, FORMAT_VERSION)?;
379    let (enc, file_prefix) = match FileEncryptor::init(&key) {
380        Ok(ok) => ok,
381        Err(FileError::Random(e)) => return Err(SealError::Random(e)),
382        // `FileEncryptor::init`'s only failure mode is the OS CSPRNG call for its header.
383        Err(_) => unreachable!("FileEncryptor::init only ever fails via FileError::Random"),
384    };
385    let mut prefix = [0u8; SEAL_STREAM_PREFIX_BYTES];
386    prefix[0] = FORMAT_VERSION;
387    prefix[1..PREFIX_LEN].copy_from_slice(&kem_ciphertext);
388    prefix[PREFIX_LEN..].copy_from_slice(&file_prefix);
389    Ok((enc, prefix))
390}

392/// Starts opening a message sealed by [`seal`] or [`seal_stream`] (T-265): `prefix` is its first
393/// [`OPEN_STREAM_PREFIX_BYTES`] bytes. Feed the returned [`FileDecryptor`] everything after them,
394/// as the stream-file codec documents: a record's plaintext comes out only once it verifies, the
395/// `Final` record's only from `finish`, once the input is known to end there.
396///
397/// # Errors
398///
399/// Returns [`OpenError::UnsupportedVersion`] if `prefix` does not start with this format's version
400/// byte (checked before `secret` is used), or [`OpenError::InvalidCiphertext`] if the KEM
401/// ciphertext does not open under `secret`.
402pub fn open_stream(
403    prefix: &[u8; OPEN_STREAM_PREFIX_BYTES],
404    secret: &SecretKey,
405) -> Result<FileDecryptor, OpenError> {
406    if prefix[0] != FORMAT_VERSION {
407        return Err(OpenError::UnsupportedVersion);
408    }
409    let mut kem_ciphertext = [0u8; KEM_CIPHERTEXT_LEN];
410    kem_ciphertext.copy_from_slice(&prefix[1..]);
411    let key = decapsulate(&kem_ciphertext, secret, FORMAT_VERSION)?;
412    Ok(FileDecryptor::new(&key))
413}

415/// The exact length [`seal`] returns for a `message_len`-byte message, or `None` if it overflows
416/// `usize`: the prefix, the message, and one record overhead per started [`CHUNK_BYTES`] chunk -
417/// at least one, since an empty message is one empty `Final` record.
418#[must_use]
419pub fn sealed_len(message_len: usize) -> Option<usize> {
420    let records = message_len.div_ceil(CHUNK_BYTES).max(1);
421    records
422        .checked_mul(RECORD_OVERHEAD_BYTES)?
423        .checked_add(message_len)?
424        .checked_add(SEAL_STREAM_PREFIX_BYTES)
425}

427/// Encrypts `message` (any length) to `recipient`, drawing a fresh random seed and ephemeral key
428/// internally - see the module doc's "Hybrid via KDF" section. The in-memory form of
429/// [`seal_stream`]; the result is [`sealed_len`]`(message.len())` bytes.
430///
431/// # Errors
432///
433/// Returns [`SealError::Random`] if the OS CSPRNG fails - the only way this can fail.
434pub fn seal(message: &[u8], recipient: &PublicKey) -> Result<Vec<u8>, SealError> {
435    let (mut enc, prefix) = seal_stream(recipient)?;
436    let mut out = Vec::with_capacity(sealed_len(message.len()).unwrap_or(0));
437    out.extend_from_slice(&prefix);
438    if seal_all(&mut enc, message, &mut out).is_err() {
439        unreachable!(
440            "a fresh encryptor over an in-memory message: never finalized, chunks exact by \
441             construction, far below the record counter's limit"
442        )
443    }
444    Ok(out)
445}

447/// Decrypts `sealed` (as produced by [`seal`] or [`seal_stream`]) under `secret` - the in-memory
448/// form of [`open_stream`]. Partially decrypted plaintext is zeroized before an error returns.
449///
450/// # Errors
451///
452/// Returns [`OpenError::UnsupportedVersion`] if `sealed`'s first byte is not this build's format
453/// version (a blob from 0.4-0.6 included - see [`open_v2_legacy`]), [`OpenError::Truncated`] if
454/// it is shorter than the minimum possible length or its stream ends before a verified `Final`
455/// record, or [`OpenError::InvalidCiphertext`] for any other failure - see the module doc's
456/// "Error collapsing" section.
457pub fn open(sealed: &[u8], secret: &SecretKey) -> Result<Vec<u8>, OpenError> {
458    let Some(&version) = sealed.first() else {
459        return Err(OpenError::Truncated);
460    };
461    if version != FORMAT_VERSION {
462        return Err(OpenError::UnsupportedVersion);
463    }
464    if sealed.len() < MIN_SEALED_LEN {
465        return Err(OpenError::Truncated);
466    }
467    let (prefix, file) = sealed.split_at(PREFIX_LEN);
468    let Ok(prefix) = <&[u8; PREFIX_LEN]>::try_from(prefix) else {
469        return Err(OpenError::Truncated);
470    };
471    let mut dec = open_stream(prefix, secret)?;
472    open_all(&mut dec, file).map_err(|e| match e {
473        FileError::Truncated => OpenError::Truncated,
474        _ => OpenError::InvalidCiphertext,
475    })
476}

478/// Decrypts a blob in the one-chunk format 0.4-0.6 wrote (version byte 2: `version || kem
479/// ciphertext (128) || secretstream header (32) || ciphertext || tag (16)`). It exists for
480/// `uacrypt box-open`, which still opens files sealed before T-265 (owner decision 2026-10-15,
481/// D-239); it is not memory-bounded and not wired into the C ABI or any binding. [`seal`] never
482/// writes this format.
483///
484/// # Errors
485///
486/// Returns [`OpenError::Truncated`] if `sealed` is shorter than 177 bytes,
487/// [`OpenError::UnsupportedVersion`] if its first byte is not 2, or
488/// [`OpenError::InvalidCiphertext`] for any other failure.
489pub fn open_v2_legacy(sealed: &[u8], secret: &SecretKey) -> Result<Vec<u8>, OpenError> {
490    const MIN_LEN: usize = PREFIX_LEN + HEADER_LEN + TAG_LEN;
491    if sealed.len() < MIN_LEN {
492        return Err(OpenError::Truncated);
493    }
494    if sealed[0] != LEGACY_FORMAT_VERSION {
495        return Err(OpenError::UnsupportedVersion);
496    }
497
498    let mut kem_ciphertext = [0u8; KEM_CIPHERTEXT_LEN];
499    kem_ciphertext.copy_from_slice(&sealed[1..PREFIX_LEN]);
500    let mut header = [0u8; HEADER_LEN];
501    header.copy_from_slice(&sealed[PREFIX_LEN..PREFIX_LEN + HEADER_LEN]);
502    let ciphertext_len = sealed.len() - MIN_LEN;
503    let ciphertext_start = PREFIX_LEN + HEADER_LEN;
504    let ciphertext = &sealed[ciphertext_start..ciphertext_start + ciphertext_len];
505    let tag = &sealed[ciphertext_start + ciphertext_len..];
506
507    let key = decapsulate(&kem_ciphertext, secret, LEGACY_FORMAT_VERSION)?;
508    let mut pull = PullState::init(&key, &header);
509    let mut plaintext = vec![0u8; ciphertext_len];
510    pull.pull(Tag::Final.to_byte(), ciphertext, tag, &mut plaintext)
511        .map_err(|_| OpenError::InvalidCiphertext)?;
512
513    Ok(plaintext)
514}

540    /// T-248: the KEM ciphertext (first, middle and last byte), the recipient key (first and last
541    /// byte) and the seed all feed the stream key - the property a round-trip test cannot show.
542    #[test]
543    fn stream_key_binds_kem_ciphertext_recipient_and_seed() {
544        let seed = [0x11u8; 32];
545        let kem = [0x22u8; KEM_CIPHERTEXT_LEN];
546        let pk = [0x33u8; PUBLIC_KEY_LEN];
547        let base = *stream_key(&seed, &kem, &pk, FORMAT_VERSION).as_bytes();
548
549        for i in [0, KEM_CIPHERTEXT_LEN / 2, KEM_CIPHERTEXT_LEN - 1] {
550            let mut kem2 = kem;
551            kem2[i] ^= 1;
552            assert_ne!(
553                *stream_key(&seed, &kem2, &pk, FORMAT_VERSION).as_bytes(),
554                base,
555                "kem byte {i}"
556            );
557        }
558        for i in [0, PUBLIC_KEY_LEN - 1] {
559            let mut pk2 = pk;
560            pk2[i] ^= 1;
561            assert_ne!(
562                *stream_key(&seed, &kem, &pk2, FORMAT_VERSION).as_bytes(),
563                base,
564                "pk byte {i}"
565            );
566        }
567        let mut seed2 = seed;
568        seed2[0] ^= 1;
569        assert_ne!(
570            *stream_key(&seed2, &kem, &pk, FORMAT_VERSION).as_bytes(),
571            base
572        );
573        assert_eq!(
574            *stream_key(&seed, &kem, &pk, FORMAT_VERSION).as_bytes(),
575            base
576        );
577    }
