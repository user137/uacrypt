1//! Kupyna-based KMAC - DSTU 7564:2014's own MAC mode (`crypto_auth`/`crypto_onetimeauth`
2//! equivalent, `docs/dstu-crypto-project.md` "Mapping onto the libsodium API").
3//!
4//! **Provisional, not confirmed against the primary DSTU 7564:2014 text** - the paper this project
5//! otherwise treats as its highest-trust Kupyna source (`docs/papers/Kupyna.pdf`) states this MAC
6//! mode exists in the standard but does not itself describe it. Ported from two independent
7//! reference implementations instead: `oracles/uapki/library/uapkic/src/dstu7564.c`
8//! (`dstu7564_init_kmac`/`dstu7564_update_kmac`/`dstu7564_final_kmac`, whose own comment states
9//! the construction directly - `HMAC(M,K) = H(PAD(K) || PAD(M) || (~K))`) and
10//! `oracles/bouncycastle-java/.../macs/DSTU7564Mac.java` (an independent Java implementation, not
11//! a port of the C above). Both implementations' self-test vectors agree byte-for-byte across all
12//! three MAC sizes - see `crates/dstu-core/tests/vectors/kupyna-kmac/*.json` and `docs/DECISIONS.md`
13//! D-44 for why this is stronger evidence than Strumok's or Kalyna-CCM's equivalent caveats.
14//! Full construction cited step-by-step in `docs/pseudocode/kupyna-kmac.md`.
15
16use super::kupyna::{kupyna_padding, KupynaCore};
17use subtle::ConstantTimeEq;
18use zeroize::Zeroize;

20#[derive(Debug, Clone, Copy, PartialEq, Eq)]
21pub enum KmacError {
22    /// Both oracles' test data uses `key.len() == mac_len` in every case, and UAPKI's C source
23    /// hard-enforces it (`CHECK_PARAM(key_buf_len == mac_len)`) - matched here rather than
24    /// building Bouncy Castle's more permissive (but untested-by-any-vector) arbitrary-key-length
25    /// path. See `docs/pseudocode/kupyna-kmac.md`.
26    WrongKeyLength,
27    TagMismatch,
28}
29
30/// Shared construction for all three MAC sizes - see `docs/pseudocode/kupyna-kmac.md` for the
31/// algorithm and citations.
32fn kmac_generic(
33    key: &[u8],
34    message: &[u8],
35    columns: usize,
36    rounds: usize,
37    last_row_shift: usize,
38    output_bytes: usize,
39) -> Result<[u8; 64], KmacError> {
40    if key.len() != output_bytes {
41        return Err(KmacError::WrongKeyLength);
42    }
43
44    let mut core = KupynaCore::new(columns, rounds, last_row_shift);
45    let block_bytes = core.block_bytes();
46
47    // PAD(K): always exactly one block for all three MAC sizes (key.len() + 13 <= block_bytes
48    // holds for 32+13<=64, 48+13<=128, 64+13<=128 - not a coincidence, the same margin the
49    // official vectors themselves rely on).
50    #[allow(clippy::cast_possible_truncation)] // key.len() is 32/48/64, always << u64::MAX
51    let (padded_key, padded_key_len) = kupyna_padding(key, (key.len() as u64) * 8, block_bytes);
52    core.update(&padded_key[..padded_key_len]);
53
54    // M, raw - contributes to `core`'s own running `total_len` normally, same as any other
55    // streamed message.
56    core.update(message);
57
58    // PAD(M)'s padding suffix, using M's *own* byte length (not `key.len() + message.len()`) as
59    // the length field - `message.len()`'s tail may already be sitting in `core`'s internal
60    // buffer from the `update(message)` call above, so only the *new* suffix bytes (from the
61    // already-buffered length onward) are fed in, or `update` would double-count them.
62    #[allow(clippy::cast_possible_truncation)] // message.len() here is always << u64::MAX
63    let (pad_m, pad_m_len) =
64        kupyna_padding(core.buffered(), (message.len() as u64) * 8, block_bytes);
65    let already_buffered = core.buffered().len();
66    core.update(&pad_m[already_buffered..pad_m_len]);
67
68    // ~K, raw - the outermost `finalize` below applies the *true* total-length padding over
69    // everything fed to `core` so far (PAD(K)'s one block + M + PAD(M)'s suffix + ~K), exactly
70    // matching both oracles' final `doFinal`/`dstu7564_final_kmac` step.
71    let mut inverted_key = [0u8; 64];
72    for (dst, &src) in inverted_key[..key.len()].iter_mut().zip(key) {
73        *dst = !src;
74    }
75    core.update(&inverted_key[..key.len()]);
76    inverted_key.zeroize();
77
78    Ok(core.finalize(output_bytes))
79}

81macro_rules! kmac_variant {
82    ($name:ident, $columns:literal, $rounds:literal, $last_row_shift:literal, $mac_len:literal) => {
83        pub struct $name;
84
85        impl $name {
86            /// Computes the MAC of `message` under `key` (`key` must be exactly `$mac_len` bytes).
87            ///
88            /// # Errors
89            ///
90            /// Returns `Err(KmacError::WrongKeyLength)` if `key.len() != $mac_len`.
91            pub fn mac(key: &[u8], message: &[u8]) -> Result<[u8; $mac_len], KmacError> {
92                let full =
93                    kmac_generic(key, message, $columns, $rounds, $last_row_shift, $mac_len)?;
94                let mut out = [0u8; $mac_len];
95                out.copy_from_slice(&full[..$mac_len]);
96                Ok(out)
97            }
98
99            /// Recomputes the MAC and compares it against `expected` in constant time
100            /// (`subtle::ConstantTimeEq`, per `docs/SECURITY.md`'s hard constraint on secret
101            /// comparisons - a MAC verification is exactly this category).
102            ///
103            /// # Errors
104            ///
105            /// Returns `Err(KmacError::WrongKeyLength)` if `key.len() != $mac_len`, or
106            /// `Err(KmacError::TagMismatch)` if the recomputed MAC doesn't match `expected`.
107            pub fn verify(
108                key: &[u8],
109                message: &[u8],
110                expected: &[u8; $mac_len],
111            ) -> Result<(), KmacError> {
112                let mac = Self::mac(key, message)?;
113                if mac.ct_eq(expected).into() {
114                    Ok(())
115                } else {
116                    Err(KmacError::TagMismatch)
117                }
118            }
119        }
120    };
121}

124kmac_variant!(Kupyna384Kmac, 16, 14, 11, 48);
125kmac_variant!(Kupyna512Kmac, 16, 14, 11, 64);
