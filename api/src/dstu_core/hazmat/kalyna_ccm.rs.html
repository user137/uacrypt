1//! Kalyna-CCM: DSTU 7624:2014 mode of operation "вироблення імітовставки і гамування" (§13).
2//!
3//! **Follows the standard's text** - DSTU 7624:2014, draft edition (T-234, `docs/DECISIONS.md`
4//! D-205), checked against all five annex В.9 examples. First ported from
5//! `oracles/uapki/library/uapkic/src/dstu7624.c` (`dstu7624_init_ccm`, `ccm_padd`,
6//! `dstu7624_encrypt_ccm`/`dstu7624_decrypt_ccm`, D-41), which matches the annex examples but not the
7//! text elsewhere. Three places were corrected to §13.2:
8//! - the flag byte's top bit marks a non-empty AAD (table 13.2), not a non-empty plaintext;
9//! - with no AAD the CBC-MAC starts from `T(G1)` alone (the draft literally says "B = G1", read as a
10//!   drafting slip - see D-205);
11//! - `G2` is `lambda_o` in `N_B` bytes LE plus exactly enough zeros to block-align `G1 || G2 || O`,
12//!   which can spill into another block.
13//!
14//! An empty plaintext is rejected ([`CcmError::EmptyPlaintext`]): §13.1 requires `|M| >= 8` bits.
15//! UAPKI and Bouncy Castle 1.85 differ from this module on some inputs - see
16//! `docs/COMPATIBILITY.md`.
17//!
18//! This module is a standalone hazmat-level primitive. `crypto_secretbox` was first built on it
19//! (T-37, D-51) and moved to Kalyna-GCM in D-63; no `crypto_*` module uses it now.
20//!
21//! # Hard length limit - sourced, not chosen
22//!
23//! `dstu7624.c`'s `ccm_padd` wrote the plaintext and AAD lengths as a single byte each, so the
24//! port capped **both at 255 bytes**. The standard itself writes them as `N_B`-byte LE fields (this
25//! module now does too, D-205), so the cap is a scope limit kept from the port, not a property of
26//! the standard. [`MAX_PLAINTEXT_LEN`]/[`MAX_AAD_LEN`] enforce it; longer input is rejected, never
27//! truncated.
28//!
29//! # Nonce
30//!
31//! The nonce is a full block-size buffer (matching the vectors, which supply a full-block IV even
32//! though `G1` only takes a `block_len - ccm_nb - 1`-byte prefix of it - the remaining bytes still
33//! feed the CTR keystream, and it is the *full* block that seeds it via `Gamma::new`). §13.1 requires
34//! only that **prefix** to be unique per key (11 bytes for a 128-bit block), so the uniqueness
35//! margin to plan for is the prefix's, not the full block's (D-205). **Strategy resolved, `docs/DECISIONS.md` D-40/`docs/TASKS.md`
36//! T-82**: this module always treats the nonce as caller-supplied, never generating one itself -
37//! `no_std` callers may have no OS CSPRNG to call. The recommended pattern for any caller that
38//! *does* have one is a fresh, independently-random full-block nonce per message under a given
39//! key (a libsodium-style wide random nonce, not a TLS-1.3-style stateful counter - a counter's
40//! uniqueness guarantee needs durable cross-reboot state this project can't assume for its
41//! embedded targets). For the narrowest variants (128-bit block) a random nonce's 88-bit prefix
42//! gives a birthday bound of roughly 2^44 messages under one key, not the 2^48 D-40 first stated
43//! for the full 128-bit block (D-205). `increment_counter` carries across the whole block, where
44//! §13.3 increments only its low half; the two differ only after 2^64 blocks, unreachable under the
45//! 255-byte cap. `uacrypt kalyna-ccm encrypt` (the CLI
46//! layer) implements this recommendation by generating the nonce itself via `getrandom` rather
47//! than accepting one - the five `(ccm_nb, q)` pairs used here remain exactly what the cross-oracle
48//! vectors confirm, not a new choice made by this module.

53const MAX_BLOCK: usize = 64;
54/// Sourced limit - see the module doc comment's "Hard length limit" section.
55pub const MAX_PLAINTEXT_LEN: usize = 255;
56/// Sourced limit - see the module doc comment's "Hard length limit" section.
57pub const MAX_AAD_LEN: usize = 255;
58
59/// Largest `N_B` (length-field width) any variant uses.
60const MAX_CCM_NB: usize = 8;
61/// `G1 || G2 || O` at its largest: `G1` is one block, `G2` is `N_B` length bytes plus at most
62/// `block_len - 1` zero bytes (§13.2), then up to `MAX_AAD_LEN` bytes of AAD. 512/512 with a 255-byte
63/// AAD needs exactly 64 + 65 + 255 = 384 bytes.
64const H_BUF_LEN: usize = MAX_BLOCK + (MAX_CCM_NB + MAX_BLOCK - 1) + MAX_AAD_LEN;
65/// Plaintext plus at most one block of `padding`'s 0x80-then-zeros pad.
66const P_BUF_LEN: usize = MAX_PLAINTEXT_LEN + MAX_BLOCK;
67
68#[derive(Debug, Clone, Copy, PartialEq, Eq)]
69pub enum CcmError {
70    /// §13.1 requires `|M| = 8r` with `r >= 1`: an empty plaintext is outside the standard's domain.
71    EmptyPlaintext,
72    PlaintextTooLong,
73    AadTooLong,
74    TagMismatch,
75}

77/// `gamma_gen` (`dstu7624.c:2730`): little-endian increment-with-carry over the first `block_len`
78/// bytes only - byte 0 is least-significant, matching the oracle's own indexing.
79fn increment_counter(counter: &mut [u8], block_len: usize) {
80    for byte in counter.iter_mut().take(block_len) {
81        *byte = byte.wrapping_add(1);
82        if *byte != 0 {
83            return;
84        }
85    }
86}

88#[allow(clippy::cast_possible_truncation)] // q is always one of {8,16,32,48,64}, never truncated
89fn tag_length_code(q: usize) -> u8 {
90    match q {
91        8 => 2,
92        16 => 3,
93        32 => 4,
94        48 => 5,
95        64 => 6,
96        _ => 0,
97    }
98}

100/// DSTU 7624:2014 (draft) §13.2 (T-234, `docs/DECISIONS.md` D-205): builds `G1` (table 13.1) and,
101/// when the AAD `O` is non-empty, `G2 || O`, and runs the CBC-MAC (repeated XOR-then-encrypt) over
102/// them and the Б.2-padded plaintext. Returns a `MAX_BLOCK`-byte buffer whose first `q` bytes are
103/// the raw (unmasked) tag.
104#[allow(clippy::too_many_arguments)]
105fn compute_tag(
106    encrypt_block: &dyn Fn(&[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK],
107    block_len: usize,
108    ccm_nb: usize,
109    q: usize,
110    nonce: &[u8],
111    aad: &[u8],
112    plaintext: &[u8],
113) -> [u8; MAX_BLOCK] {
114    let prefix_len = block_len - ccm_nb - 1;
115
116    // G1 = L(S) || |M|/8 as N_B bytes LE || flags (tables 13.1/13.2).
117    let mut g1 = [0u8; MAX_BLOCK];
118    g1[..prefix_len].copy_from_slice(&nonce[..prefix_len]);
119    let m_len = (plaintext.len() as u64).to_le_bytes();
120    g1[prefix_len..prefix_len + ccm_nb].copy_from_slice(&m_len[..ccm_nb]);
121    let mut flags = if aad.is_empty() { 0u8 } else { 0x80 };
122    flags |= tag_length_code(q) << 4;
123    #[allow(clippy::cast_possible_truncation)] // ccm_nb is always one of {4,6,8}, fits easily
124    {
125        flags |= (ccm_nb - 1) as u8;
126    }
127    g1[block_len - 1] = flags;
128
129    // |O| > 0: G1 || G2 || O with G2 = lambda_o (N_B bytes LE) || 0^z, z chosen so the whole is
130    // block-aligned. |O| = 0: just G1, so B = T(G1) (the text's literal "B = G1" read as a drafting
131    // slip - see D-205).
132    let mut h_buf = [0u8; H_BUF_LEN];
133    h_buf[..block_len].copy_from_slice(&g1[..block_len]);
134    let mut h_len = block_len;
135    if !aad.is_empty() {
136        let o_len = (aad.len() as u64).to_le_bytes();
137        h_buf[h_len..h_len + ccm_nb].copy_from_slice(&o_len[..ccm_nb]);
138        let zeros = (2 * block_len - aad.len() % block_len - ccm_nb) % block_len;
139        h_len += ccm_nb + zeros;
140        h_buf[h_len..h_len + aad.len()].copy_from_slice(aad);
141        h_len += aad.len();
142    }
143
144    // `chunks_exact` would silently drop a remainder; the G2 formula makes one impossible.
145    debug_assert_eq!(h_len % block_len, 0);
146    let mut b = [0u8; MAX_BLOCK];
147    for chunk in h_buf[..h_len].chunks_exact(block_len) {
148        for (acc, byte) in b.iter_mut().zip(chunk) {
149            *acc ^= byte;
150        }
151        b = encrypt_block(&b);
152    }
153
154    let mut p_buf = [0u8; P_BUF_LEN];
155    p_buf[..plaintext.len()].copy_from_slice(plaintext);
156    let mut p_len = plaintext.len();
157    if !p_len.is_multiple_of(block_len) {
158        p_buf[p_len] = 0x80;
159        p_len += block_len - (p_len % block_len);
160    }
161    for chunk in p_buf[..p_len].chunks_exact(block_len) {
162        for (acc, byte) in b.iter_mut().zip(chunk) {
163            *acc ^= byte;
164        }
165        b = encrypt_block(&b);
166    }
167
168    b
169}

171/// `dstu7624_init_ctr`/`encrypt_ctr` (`dstu7624.c:4397`/`2739`): the running CTR keystream state.
172/// The first block computed at construction (`E_K(nonce)`) is never itself used as keystream - it
173/// only seeds the counter that gets incremented before the first real keystream block is derived
174/// (`used = block_len` forces regeneration on the very first [`Self::apply`] call) - this doubled
175/// indirection is a property of the source, transcribed as-is rather than "simplified" to
176/// textbook CTR.
177struct Gamma {
178    counter: [u8; MAX_BLOCK],
179    keystream: [u8; MAX_BLOCK],
180    used: usize,
181    block_len: usize,
182}
183
184impl Gamma {
185    fn new(
186        block_len: usize,
187        nonce_block: &[u8; MAX_BLOCK],
188        encrypt_block: &dyn Fn(&[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK],
189    ) -> Self {
190        let seed = encrypt_block(nonce_block);
191        Self {
192            counter: seed,
193            keystream: seed,
194            used: block_len,
195            block_len,
196        }
197    }
198
199    /// XORs `buf` with the keystream in place, continuing from wherever the previous call left
200    /// off - callers rely on this to mask the tag with the keystream bytes immediately following
201    /// whatever the plaintext/ciphertext call consumed, not a fresh block.
202    fn apply(
203        &mut self,
204        buf: &mut [u8],
205        encrypt_block: &dyn Fn(&[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK],
206    ) {
207        let block_len = self.block_len;
208        let mut offset = self.used;
209        let mut data_off = 0usize;
210
211        if offset != 0 {
212            while offset < block_len && data_off < buf.len() {
213                buf[data_off] ^= self.keystream[offset];
214                data_off += 1;
215                offset += 1;
216            }
217            if offset == block_len {
218                increment_counter(&mut self.counter, block_len);
219                self.keystream = encrypt_block(&self.counter);
220                offset = 0;
221            }
222        }
223
224        while data_off + block_len <= buf.len() {
225            for i in 0..block_len {
226                buf[data_off + i] ^= self.keystream[i];
227            }
228            data_off += block_len;
229            increment_counter(&mut self.counter, block_len);
230            self.keystream = encrypt_block(&self.counter);
231        }
232
233        while data_off < buf.len() {
234            buf[data_off] ^= self.keystream[offset];
235            data_off += 1;
236            offset += 1;
237        }
238
239        self.used = offset;
240    }
241}

243/// Shared core behind every variant's `seal_in_place` (see `kalyna_ccm_variant!`). Encrypts `buf`
244/// in place and returns the `q`-byte masked tag (first `q` bytes of the returned buffer).
245#[allow(clippy::too_many_arguments)]
246fn seal_core(
247    encrypt_block: &dyn Fn(&[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK],
248    block_len: usize,
249    ccm_nb: usize,
250    q: usize,
251    nonce: &[u8],
252    aad: &[u8],
253    buf: &mut [u8],
254) -> Result<[u8; MAX_BLOCK], CcmError> {
255    if buf.is_empty() {
256        return Err(CcmError::EmptyPlaintext);
257    }
258    if buf.len() > MAX_PLAINTEXT_LEN {
259        return Err(CcmError::PlaintextTooLong);
260    }
261    if aad.len() > MAX_AAD_LEN {
262        return Err(CcmError::AadTooLong);
263    }
264
265    let raw_tag = compute_tag(encrypt_block, block_len, ccm_nb, q, nonce, aad, buf);
266
267    let mut nonce_block = [0u8; MAX_BLOCK];
268    nonce_block[..block_len].copy_from_slice(&nonce[..block_len]);
269    let mut gamma = Gamma::new(block_len, &nonce_block, encrypt_block);
270
271    gamma.apply(buf, encrypt_block);
272
273    let mut tag_buf = [0u8; MAX_BLOCK];
274    tag_buf[..q].copy_from_slice(&raw_tag[..q]);
275    gamma.apply(&mut tag_buf[..q], encrypt_block);
276
277    Ok(tag_buf)
278}

280/// Shared core behind every variant's `open_in_place`. Decrypts `buf` in place (tentatively),
281/// recomputes the tag over the recovered plaintext, and only leaves the recovered plaintext in
282/// `buf` if the recomputed tag matches - on mismatch, `buf` is zeroed before returning `Err`, so a
283/// caller can never observe unverified plaintext even transiently (`CLAUDE.md`'s "no secret
284/// material" discipline, generalized to "no unverified plaintext" for AEAD).
285#[allow(clippy::too_many_arguments)]
286fn open_core(
287    encrypt_block: &dyn Fn(&[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK],
288    block_len: usize,
289    ccm_nb: usize,
290    q: usize,
291    nonce: &[u8],
292    aad: &[u8],
293    buf: &mut [u8],
294    tag: &[u8],
295) -> Result<(), CcmError> {
296    if buf.is_empty() {
297        return Err(CcmError::EmptyPlaintext);
298    }
299    if buf.len() > MAX_PLAINTEXT_LEN {
300        return Err(CcmError::PlaintextTooLong);
301    }
302    if aad.len() > MAX_AAD_LEN {
303        return Err(CcmError::AadTooLong);
304    }
305
306    let mut nonce_block = [0u8; MAX_BLOCK];
307    nonce_block[..block_len].copy_from_slice(&nonce[..block_len]);
308    let mut gamma = Gamma::new(block_len, &nonce_block, encrypt_block);
309
310    gamma.apply(buf, encrypt_block);
311
312    let mut recovered_tag = [0u8; MAX_BLOCK];
313    recovered_tag[..q].copy_from_slice(&tag[..q]);
314    gamma.apply(&mut recovered_tag[..q], encrypt_block);
315
316    let expected_tag = compute_tag(encrypt_block, block_len, ccm_nb, q, nonce, aad, buf);
317
318    let ok: bool = recovered_tag[..q].ct_eq(&expected_tag[..q]).into();
319    if ok {
320        Ok(())
321    } else {
322        buf.zeroize();
323        Err(CcmError::TagMismatch)
324    }
325}

327macro_rules! kalyna_ccm_variant {
328    ($name:ident, $expanded:ident, $key_bytes:literal, $block_bytes:literal, $ccm_nb:literal, $q:literal) => {
329        #[doc = concat!(
330            "CCM mode over [`super::kalyna::", stringify!($expanded), "`] - see the module doc ",
331            "comment for the construction citation and its standard alignment (D-205)."
332        )]
333        pub struct $name {
334            key: super::kalyna::$expanded,
335        }
336
337        impl $name {
338            #[must_use]
339            pub fn new(key: &[u8; $key_bytes]) -> Self {
340                Self {
341                    key: super::kalyna::$expanded::new(key),
342                }
343            }
344
345            fn encrypt_block_padded(&self, block: &[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK] {
346                let mut input = [0u8; $block_bytes];
347                input.copy_from_slice(&block[..$block_bytes]);
348                let out = self.key.encrypt_block(&input);
349                let mut padded = [0u8; MAX_BLOCK];
350                padded[..$block_bytes].copy_from_slice(&out);
351                padded
352            }
353
354            /// Encrypts `buf` in place (plaintext -> ciphertext) and returns the masked
355            /// authentication tag. `nonce` must never repeat under the same key - see the module
356            /// doc comment's "Nonce" section for the recommended generation strategy
357            /// (`docs/DECISIONS.md` D-40) and this variant's safe per-key message-count margin.
358            ///
359            /// # Errors
360            ///
361            /// Returns `Err` if `buf` or `aad` exceed the sourced length limit (see the module
362            /// doc comment's "Hard length limit" section) - never silently truncates.
363            pub fn seal_in_place(
364                &self,
365                nonce: &[u8; $block_bytes],
366                aad: &[u8],
367                buf: &mut [u8],
368            ) -> Result<[u8; $q], CcmError> {
369                let encrypt_block = |b: &[u8; MAX_BLOCK]| self.encrypt_block_padded(b);
370                let tag = seal_core(
371                    &encrypt_block,
372                    $block_bytes,
373                    $ccm_nb,
374                    $q,
375                    nonce,
376                    aad,
377                    buf,
378                )?;
379                let mut out = [0u8; $q];
380                out.copy_from_slice(&tag[..$q]);
381                Ok(out)
382            }
383
384            /// Decrypts `buf` in place (ciphertext -> plaintext) only if `tag` verifies. On
385            /// failure, `buf` is zeroed before returning `Err` - the caller can never observe
386            /// unverified plaintext.
387            ///
388            /// # Errors
389            ///
390            /// Returns `Err(CcmError::TagMismatch)` if authentication fails, or a length error
391            /// under the same conditions as [`Self::seal_in_place`].
392            pub fn open_in_place(
393                &self,
394                nonce: &[u8; $block_bytes],
395                aad: &[u8],
396                buf: &mut [u8],
397                tag: &[u8; $q],
398            ) -> Result<(), CcmError> {
399                let encrypt_block = |b: &[u8; MAX_BLOCK]| self.encrypt_block_padded(b);
400                open_core(
401                    &encrypt_block,
402                    $block_bytes,
403                    $ccm_nb,
404                    $q,
405                    nonce,
406                    aad,
407                    buf,
408                    tag,
409                )
410            }
411        }
412    };
413}
