1//! Kalyna-CCM: a provisional, Kalyna-alone authenticated mode of operation (DSTU 7624:2014 CCM).
2//!
3//! **Provisional, not confirmed against the primary DSTU 7624:2014 text** - the same posture as
4//! Strumok's UAPKI-attributed vectors (`docs/DECISIONS.md` D-15). Ported directly from
5//! `oracles/uapki/library/uapkic/src/dstu7624.c` (`dstu7624_init_ccm` at line 4139, `ccm_padd` at
6//! line 2621, `dstu7624_encrypt_ccm`/`dstu7624_decrypt_ccm` at lines 2792/2849), cross-checked
7//! against `oracles/bouncycastle-java`'s `DSTU7624Test.java` CCM vectors byte-for-byte (BC's own
8//! `KCCMBlockCipher` Java source is not present in this project's sparse vendored checkout, so the
9//! cross-check is against BC's *vector outputs* only, not a second reading of BC's construction
10//! code - a materially weaker claim than "read both implementations", stated explicitly here per
11//! `CLAUDE.md`'s citation discipline). See `docs/DECISIONS.md` D-05 (revised) and D-41 for the full
12//! citation and the reasoning for choosing this construction over encrypt-then-MAC.
13//!
14//! This module is a standalone hazmat-level primitive, not `crypto_secretbox` itself - see
15//! `dstu_core::crypto_secretbox` (`docs/TASKS.md` T-37, `docs/DECISIONS.md` D-51) for the high-level wrapper
16//! built on top of it (a single fixed variant, `Kalyna256_256Ccm`, with an internally-generated
17//! nonce and a combined output format) - both still inherit this module's own not-primary-text-
18//! confirmed status.
19//!
20//! # Hard length limit - sourced, not chosen
21//!
22//! `ccm_padd`'s authentication header encodes both the plaintext length and the AAD length as a
23//! single byte each (`G1[tmp] = (uint8_t) p_data_len`, `G2[0] = (uint8_t) a_data_len`) - so this
24//! exact construction, as extracted, only correctly authenticates messages where **both plaintext
25//! and AAD are at most 255 bytes**. This is a property of the source, not a design choice made
26//! here; [`MAX_PLAINTEXT_LEN`]/[`MAX_AAD_LEN`] enforce it and [`seal`]/[`open`] reject anything
27//! longer rather than silently truncating the length field. It is also, concretely, why this is a
28//! *short-message* mode.
29//!
30//! # Nonce
31//!
32//! The nonce is a full block-size buffer (matching the vectors, which supply a full-block IV even
33//! though `ccm_padd` only consumes a `block_len - ccm_nb - 1`-byte prefix of it for the
34//! authentication header - the remaining bytes still feed the CTR keystream, and it is the *full*
35//! block that seeds it via `Gamma::new`). **Strategy resolved, `docs/DECISIONS.md` D-40/`docs/TASKS.md`
36//! T-82**: this module always treats the nonce as caller-supplied, never generating one itself -
37//! `no_std` callers may have no OS CSPRNG to call. The recommended pattern for any caller that
38//! *does* have one is a fresh, independently-random full-block nonce per message under a given
39//! key (a libsodium-style wide random nonce, not a TLS-1.3-style stateful counter - a counter's
40//! uniqueness guarantee needs durable cross-reboot state this project can't assume for its
41//! embedded targets). The narrowest variants (128-bit block) have a 128-bit nonce, safe for
42//! roughly 2^48 messages under one key by the birthday bound - see D-40 for the full margin and
43//! the caveat about `increment_counter` carrying over the full block width, which makes that bound
44//! contingent on this module's own 255-byte plaintext cap. `uacrypt kalyna-ccm encrypt` (the CLI
45//! layer) implements this recommendation by generating the nonce itself via `getrandom` rather
46//! than accepting one - the five `(ccm_nb, q)` pairs used here remain exactly what the cross-oracle
47//! vectors confirm, not a new choice made by this module.
48

52const MAX_BLOCK: usize = 64;
53/// Sourced limit - see the module doc comment's "Hard length limit" section.
54pub const MAX_PLAINTEXT_LEN: usize = 255;
55/// Sourced limit - see the module doc comment's "Hard length limit" section.
56pub const MAX_AAD_LEN: usize = 255;
57
58/// `ccm_padd`'s two fixed header blocks (`G1`, `G2`), zero-padded so `G2`'s slice length rounds
59/// its length byte + AAD up to a block boundary, plus room for `MAX_AAD_LEN` bytes of AAD.
60const H_BUF_LEN: usize = 2 * MAX_BLOCK + MAX_AAD_LEN;
61/// Plaintext plus at most one block of `padding`'s 0x80-then-zeros pad.
62const P_BUF_LEN: usize = MAX_PLAINTEXT_LEN + MAX_BLOCK;
63
64#[derive(Debug, Clone, Copy, PartialEq, Eq)]
65pub enum CcmError {
66    PlaintextTooLong,
67    AadTooLong,
68    TagMismatch,
69}

71/// `gamma_gen` (`dstu7624.c:2730`): little-endian increment-with-carry over the first `block_len`
72/// bytes only - byte 0 is least-significant, matching the oracle's own indexing.
73fn increment_counter(counter: &mut [u8], block_len: usize) {
74    for byte in counter.iter_mut().take(block_len) {
75        *byte = byte.wrapping_add(1);
76        if *byte != 0 {
77            return;
78        }
79    }
80}

82#[allow(clippy::cast_possible_truncation)] // q is always one of {8,16,32,48,64}, never truncated
83fn tag_length_code(q: usize) -> u8 {
84    match q {
85        8 => 2,
86        16 => 3,
87        32 => 4,
88        48 => 5,
89        64 => 6,
90        _ => 0,
91    }
92}

94/// `ccm_padd` (`dstu7624.c:2621`): builds the CBC-MAC authentication header (`G1`/`G2`) and runs
95/// the CBC-MAC (repeated XOR-then-encrypt) over header || AAD || padded-plaintext. Returns a
96/// `MAX_BLOCK`-byte buffer whose first `q` bytes are the raw (unmasked) tag.
97#[allow(clippy::too_many_arguments)]
98fn compute_tag(
99    encrypt_block: &dyn Fn(&[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK],
100    block_len: usize,
101    ccm_nb: usize,
102    q: usize,
103    nonce: &[u8],
104    aad: &[u8],
105    plaintext: &[u8],
106) -> [u8; MAX_BLOCK] {
107    let tmp = block_len - ccm_nb - 1;
108
109    let mut g1 = [0u8; MAX_BLOCK];
110    g1[..tmp].copy_from_slice(&nonce[..tmp]);
111    #[allow(clippy::cast_possible_truncation)] // sourced limit: plaintext.len() <= 255
112    {
113        g1[tmp] = plaintext.len() as u8;
114    }
115    let mut flags = if plaintext.is_empty() { 0u8 } else { 0x80 };
116    flags |= tag_length_code(q) << 4;
117    #[allow(clippy::cast_possible_truncation)] // ccm_nb is always one of {4,6,8}, fits easily
118    {
119        flags |= (ccm_nb - 1) as u8;
120    }
121    g1[block_len - 1] = flags;
122
123    let mut g2 = [0u8; MAX_BLOCK];
124    #[allow(clippy::cast_possible_truncation)] // sourced limit: aad.len() <= 255
125    {
126        g2[0] = aad.len() as u8;
127    }
128
129    let aad_rem = aad.len() % block_len;
130    let g2_len = block_len - aad_rem;
131
132    let mut h_buf = [0u8; H_BUF_LEN];
133    let mut h_len = 0usize;
134    h_buf[h_len..h_len + block_len].copy_from_slice(&g1[..block_len]);
135    h_len += block_len;
136    h_buf[h_len..h_len + g2_len].copy_from_slice(&g2[..g2_len]);
137    h_len += g2_len;
138    h_buf[h_len..h_len + aad.len()].copy_from_slice(aad);
139    h_len += aad.len();
140
141    let mut b = [0u8; MAX_BLOCK];
142    let mut offset = 0usize;
143    while offset < h_len {
144        for i in 0..block_len {
145            b[i] ^= h_buf[offset + i];
146        }
147        b = encrypt_block(&b);
148        offset += block_len;
149    }
150
151    let mut p_buf = [0u8; P_BUF_LEN];
152    p_buf[..plaintext.len()].copy_from_slice(plaintext);
153    let mut p_len = plaintext.len();
154    if !p_len.is_multiple_of(block_len) {
155        p_buf[p_len] = 0x80;
156        p_len += block_len - (p_len % block_len);
157    }
158    offset = 0;
159    while offset < p_len {
160        for i in 0..block_len {
161            b[i] ^= p_buf[offset + i];
162        }
163        b = encrypt_block(&b);
164        offset += block_len;
165    }
166
167    b
168}

170/// `dstu7624_init_ctr`/`encrypt_ctr` (`dstu7624.c:4397`/`2739`): the running CTR keystream state.
171/// The first block computed at construction (`E_K(nonce)`) is never itself used as keystream - it
172/// only seeds the counter that gets incremented before the first real keystream block is derived
173/// (`used = block_len` forces regeneration on the very first [`Self::apply`] call) - this doubled
174/// indirection is a property of the source, transcribed as-is rather than "simplified" to
175/// textbook CTR.
176struct Gamma {
177    counter: [u8; MAX_BLOCK],
178    keystream: [u8; MAX_BLOCK],
179    used: usize,
180    block_len: usize,
181}
182
183impl Gamma {
184    fn new(
185        block_len: usize,
186        nonce_block: &[u8; MAX_BLOCK],
187        encrypt_block: &dyn Fn(&[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK],
188    ) -> Self {
189        let seed = encrypt_block(nonce_block);
190        Self {
191            counter: seed,
192            keystream: seed,
193            used: block_len,
194            block_len,
195        }
196    }
197
198    /// XORs `buf` with the keystream in place, continuing from wherever the previous call left
199    /// off - callers rely on this to mask the tag with the keystream bytes immediately following
200    /// whatever the plaintext/ciphertext call consumed, not a fresh block.
201    fn apply(
202        &mut self,
203        buf: &mut [u8],
204        encrypt_block: &dyn Fn(&[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK],
205    ) {
206        let block_len = self.block_len;
207        let mut offset = self.used;
208        let mut data_off = 0usize;
209
210        if offset != 0 {
211            while offset < block_len && data_off < buf.len() {
212                buf[data_off] ^= self.keystream[offset];
213                data_off += 1;
214                offset += 1;
215            }
216            if offset == block_len {
217                increment_counter(&mut self.counter, block_len);
218                self.keystream = encrypt_block(&self.counter);
219                offset = 0;
220            }
221        }
222
223        while data_off + block_len <= buf.len() {
224            for i in 0..block_len {
225                buf[data_off + i] ^= self.keystream[i];
226            }
227            data_off += block_len;
228            increment_counter(&mut self.counter, block_len);
229            self.keystream = encrypt_block(&self.counter);
230        }
231
232        while data_off < buf.len() {
233            buf[data_off] ^= self.keystream[offset];
234            data_off += 1;
235            offset += 1;
236        }
237
238        self.used = offset;
239    }
240}

242/// Shared core behind every variant's `seal_in_place` (see `kalyna_ccm_variant!`). Encrypts `buf`
243/// in place and returns the `q`-byte masked tag (first `q` bytes of the returned buffer).
244#[allow(clippy::too_many_arguments)]
245fn seal_core(
246    encrypt_block: &dyn Fn(&[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK],
247    block_len: usize,
248    ccm_nb: usize,
249    q: usize,
250    nonce: &[u8],
251    aad: &[u8],
252    buf: &mut [u8],
253) -> Result<[u8; MAX_BLOCK], CcmError> {
254    if buf.len() > MAX_PLAINTEXT_LEN {
255        return Err(CcmError::PlaintextTooLong);
256    }
257    if aad.len() > MAX_AAD_LEN {
258        return Err(CcmError::AadTooLong);
259    }
260
261    let raw_tag = compute_tag(encrypt_block, block_len, ccm_nb, q, nonce, aad, buf);
262
263    let mut nonce_block = [0u8; MAX_BLOCK];
264    nonce_block[..block_len].copy_from_slice(&nonce[..block_len]);
265    let mut gamma = Gamma::new(block_len, &nonce_block, encrypt_block);
266
267    gamma.apply(buf, encrypt_block);
268
269    let mut tag_buf = [0u8; MAX_BLOCK];
270    tag_buf[..q].copy_from_slice(&raw_tag[..q]);
271    gamma.apply(&mut tag_buf[..q], encrypt_block);
272
273    Ok(tag_buf)
274}

276/// Shared core behind every variant's `open_in_place`. Decrypts `buf` in place (tentatively),
277/// recomputes the tag over the recovered plaintext, and only leaves the recovered plaintext in
278/// `buf` if the recomputed tag matches - on mismatch, `buf` is zeroed before returning `Err`, so a
279/// caller can never observe unverified plaintext even transiently (`CLAUDE.md`'s "no secret
280/// material" discipline, generalized to "no unverified plaintext" for AEAD).
281#[allow(clippy::too_many_arguments)]
282fn open_core(
283    encrypt_block: &dyn Fn(&[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK],
284    block_len: usize,
285    ccm_nb: usize,
286    q: usize,
287    nonce: &[u8],
288    aad: &[u8],
289    buf: &mut [u8],
290    tag: &[u8],
291) -> Result<(), CcmError> {
292    if buf.len() > MAX_PLAINTEXT_LEN {
293        return Err(CcmError::PlaintextTooLong);
294    }
295    if aad.len() > MAX_AAD_LEN {
296        return Err(CcmError::AadTooLong);
297    }
298
299    let mut nonce_block = [0u8; MAX_BLOCK];
300    nonce_block[..block_len].copy_from_slice(&nonce[..block_len]);
301    let mut gamma = Gamma::new(block_len, &nonce_block, encrypt_block);
302
303    gamma.apply(buf, encrypt_block);
304
305    let mut recovered_tag = [0u8; MAX_BLOCK];
306    recovered_tag[..q].copy_from_slice(&tag[..q]);
307    gamma.apply(&mut recovered_tag[..q], encrypt_block);
308
309    let expected_tag = compute_tag(encrypt_block, block_len, ccm_nb, q, nonce, aad, buf);
310
311    let ok: bool = recovered_tag[..q].ct_eq(&expected_tag[..q]).into();
312    if ok {
313        Ok(())
314    } else {
315        buf.zeroize();
316        Err(CcmError::TagMismatch)
317    }
318}

320macro_rules! kalyna_ccm_variant {
321    ($name:ident, $expanded:ident, $key_bytes:literal, $block_bytes:literal, $ccm_nb:literal, $q:literal) => {
322        #[doc = concat!(
323            "CCM mode over [`super::kalyna::", stringify!($expanded), "`] - see the module doc ",
324            "comment for the construction citation and its provisional status."
325        )]
326        pub struct $name {
327            key: super::kalyna::$expanded,
328        }
329
330        impl $name {
331            #[must_use]
332            pub fn new(key: &[u8; $key_bytes]) -> Self {
333                Self {
334                    key: super::kalyna::$expanded::new(key),
335                }
336            }
337
338            fn encrypt_block_padded(&self, block: &[u8; MAX_BLOCK]) -> [u8; MAX_BLOCK] {
339                let mut input = [0u8; $block_bytes];
340                input.copy_from_slice(&block[..$block_bytes]);
341                let out = self.key.encrypt_block(&input);
342                let mut padded = [0u8; MAX_BLOCK];
343                padded[..$block_bytes].copy_from_slice(&out);
344                padded
345            }
346
347            /// Encrypts `buf` in place (plaintext -> ciphertext) and returns the masked
348            /// authentication tag. `nonce` must never repeat under the same key - see the module
349            /// doc comment's "Nonce" section for the recommended generation strategy
350            /// (`docs/DECISIONS.md` D-40) and this variant's safe per-key message-count margin.
351            ///
352            /// # Errors
353            ///
354            /// Returns `Err` if `buf` or `aad` exceed the sourced length limit (see the module
355            /// doc comment's "Hard length limit" section) - never silently truncates.
356            pub fn seal_in_place(
357                &self,
358                nonce: &[u8; $block_bytes],
359                aad: &[u8],
360                buf: &mut [u8],
361            ) -> Result<[u8; $q], CcmError> {
362                let encrypt_block = |b: &[u8; MAX_BLOCK]| self.encrypt_block_padded(b);
363                let tag = seal_core(
364                    &encrypt_block,
365                    $block_bytes,
366                    $ccm_nb,
367                    $q,
368                    nonce,
369                    aad,
370                    buf,
371                )?;
372                let mut out = [0u8; $q];
373                out.copy_from_slice(&tag[..$q]);
374                Ok(out)
375            }
376
377            /// Decrypts `buf` in place (ciphertext -> plaintext) only if `tag` verifies. On
378            /// failure, `buf` is zeroed before returning `Err` - the caller can never observe
379            /// unverified plaintext.
380            ///
381            /// # Errors
382            ///
383            /// Returns `Err(CcmError::TagMismatch)` if authentication fails, or a length error
384            /// under the same conditions as [`Self::seal_in_place`].
385            pub fn open_in_place(
386                &self,
387                nonce: &[u8; $block_bytes],
388                aad: &[u8],
389                buf: &mut [u8],
390                tag: &[u8; $q],
391            ) -> Result<(), CcmError> {
392                let encrypt_block = |b: &[u8; MAX_BLOCK]| self.encrypt_block_padded(b);
393                open_core(
394                    &encrypt_block,
395                    $block_bytes,
396                    $ccm_nb,
397                    $q,
398                    nonce,
399                    aad,
400                    buf,
401                    tag,
402                )
403            }
404        }
405    };
406}
