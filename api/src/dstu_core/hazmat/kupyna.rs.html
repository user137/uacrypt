1//! Kupyna hash function (DSTU 7564:2014).
2//!
3//! Ported from `docs/pseudocode/kupyna.md` (itself transcribed from the designers' paper,
4//! `docs/papers/Kupyna.pdf` Sections 4-6) and structurally mirrors
5//! `oracles/kupyna-reference/kupyna.c` (Roman Oliynykov) byte-matrix-for-byte-matrix, rather than
6//! the T-table-fused optimization Bouncy Castle uses — deliberately, since this port could not be
7//! compiled or run locally (no Rust toolchain available; see `.claude.local.md`), and the more
8//! literal translation carries less risk of an unverifiable transposition bug. Full citation and
9//! verification status: `docs/DECISIONS.md` D-10.
10//!
11//! Only byte-aligned messages are supported - the public API takes `&[u8]`, which cannot express
12//! a bit-level length anyway. This matches the extracted test vectors exactly (see the `note`
13//! field in `crates/dstu-core/tests/vectors/kupyna/*.json`).
14
15use super::tables::{apply_forward_matrix, forward_sbox_mds, ROWS, SBOXES};

17const MAX_COLUMNS: usize = 16;
18const MAX_BLOCK_BYTES: usize = MAX_COLUMNS * ROWS;

20/// S-box layer (kappa): `state[col][row] <- S_{row mod 4}(state[col][row])`.
21///
22/// No production code path calls this directly anymore - `sub_shift_mix` fuses this with
23/// `shift_bytes`/`mix_columns` via `tables::SBOX_MDS` (D-28). Kept as the independent reference
24/// `tests::fused_sub_shift_mix_matches_naive` checks the fused round against (same pattern as
25/// `hazmat::kalyna`'s `sub_bytes`/`shift_rows`, D-28, and `hazmat::tables`' `MDS_MATRIX`/`gf_mul`,
26/// D-27).
27#[allow(dead_code)]
28fn sub_bytes(state: &mut [[u8; ROWS]]) {
29    for column in state.iter_mut() {
30        for (row, byte) in column.iter_mut().enumerate() {
31            *byte = SBOXES[row % 4][*byte as usize];
32        }
33    }
34}

36/// Row permutation (pi): row `r` (0..=6) rotated right by `r`; the last row rotated right by
37/// `last_row_shift` (7 for Kupyna-256's l=512, 11 for Kupyna-512's l=1024).
38///
39/// No production code path calls this directly anymore - see `sub_bytes`'s doc comment (D-28).
40#[allow(dead_code)]
41fn shift_bytes(state: &mut [[u8; ROWS]], last_row_shift: usize) {
42    let columns = state.len();
43    let mut shifted = [[0u8; ROWS]; MAX_COLUMNS];
44    for row in 0..ROWS {
45        let shift = if row == ROWS - 1 { last_row_shift } else { row };
46        for col in 0..columns {
47            shifted[(col + shift) % columns][row] = state[col][row];
48        }
49    }
50    state[..columns].copy_from_slice(&shifted[..columns]);
51}

53/// Linear layer (tau): each column multiplied by the MDS matrix over GF(2^8).
54///
55/// No production code path calls this directly anymore - see `sub_bytes`'s doc comment (D-28).
56#[allow(dead_code)]
57fn mix_columns(state: &mut [[u8; ROWS]]) {
58    apply_forward_matrix(state);
59}

61/// Fused `sub_bytes -> shift_bytes -> mix_columns`, one gather-and-XOR pass over
62/// `tables::SBOX_MDS` instead of three separate passes (D-28) - see `hazmat::kalyna::encipher_
63/// round`'s doc comment for why this is valid (S-box is row-indexed, the row permutation preserves
64/// row, so the two commute) and why it needs no per-`columns` tables, only a cheap gather index.
65///
66/// No production code path calls this directly anymore as of T-134 (`docs/DECISIONS.md` D-85) -
67/// `KupynaCore::compress_block`/`finalize` dispatch to the const-generic `sub_shift_mix_n` instead.
68/// Kept as the independent reference `const_shift_mix_tests` checks the const-generic twin against,
69/// same pattern as `sub_bytes`/`shift_bytes`/`mix_columns` above.
70#[allow(dead_code)]
71fn sub_shift_mix(state: &mut [[u8; ROWS]], last_row_shift: usize) {
72    let columns = state.len();
73    debug_assert!(columns.is_power_of_two());
74    let columns_mask = columns - 1;
75    let mut result = [[0u8; ROWS]; MAX_COLUMNS];
76    for (out_col, out_word) in result[..columns].iter_mut().enumerate() {
77        let mut acc = 0u64;
78        // `row` also drives `shift`/`src_col` and is passed to `forward_sbox_mds`, not just a
79        // direct single-collection index - not a real `iter().enumerate()` candidate.
80        #[allow(clippy::needless_range_loop)]
81        for row in 0..ROWS {
82            let shift = if row == ROWS - 1 { last_row_shift } else { row };
83            let src_col = (out_col + columns - shift) & columns_mask;
84            let byte = state[src_col][row];
85            acc ^= forward_sbox_mds(row, byte);
86        }
87        *out_word = acc.to_le_bytes();
88    }
89    state[..columns].copy_from_slice(&result[..columns]);
90}

92/// Const-generic twin of `sub_shift_mix` (T-134, direct analogue of `hazmat::kalyna`'s
93/// `encipher_round_n`, `docs/DECISIONS.md` T-128/D-77) - `COLUMNS` (always 8 or 16, verified against
94/// every `KupynaCore::new`/`digest_generic`/`kmac_generic` call site) is known at compile time, so
95/// `result` is exactly `COLUMNS` wide (no `MAX_COLUMNS`-oversized zeroing) and every bounds check
96/// on `state`/`result` is eliminated by monomorphization.
97fn sub_shift_mix_n<const COLUMNS: usize>(state: &mut [[u8; ROWS]; COLUMNS], last_row_shift: usize) {
98    debug_assert!(COLUMNS.is_power_of_two());
99    let columns_mask = COLUMNS - 1;
100    let mut result = [[0u8; ROWS]; COLUMNS];
101    for (out_col, out_word) in result.iter_mut().enumerate() {
102        let mut acc = 0u64;
103        #[allow(clippy::needless_range_loop)]
104        for row in 0..ROWS {
105            let shift = if row == ROWS - 1 { last_row_shift } else { row };
106            let src_col = (out_col + COLUMNS - shift) & columns_mask;
107            let byte = state[src_col][row];
108            acc ^= forward_sbox_mds(row, byte);
109        }
110        *out_word = acc.to_le_bytes();
111    }
112    *state = result;
113}

115/// XOR-based round-constant addition (psi-xor), used by `T`/`P`. Kupyna.pdf Section 6.2.
116///
117/// No production code path calls this directly anymore as of T-134 (`docs/DECISIONS.md` D-85) - see
118/// `sub_shift_mix`'s doc comment.
119#[allow(dead_code)]
120#[allow(clippy::cast_possible_truncation)] // col < MAX_COLUMNS (16), always fits u8
121fn add_round_constant_xor(state: &mut [[u8; ROWS]], round: u8) {
122    for (col, column) in state.iter_mut().enumerate() {
123        column[0] ^= (col as u8).wrapping_mul(0x10) ^ round;
124    }
125}

127/// Const-generic twin of `add_round_constant_xor` - see `sub_shift_mix_n`.
128#[allow(clippy::cast_possible_truncation)] // col < COLUMNS <= 16, always fits u8
129fn add_round_constant_xor_n<const COLUMNS: usize>(state: &mut [[u8; ROWS]; COLUMNS], round: u8) {
130    for (col, column) in state.iter_mut().enumerate() {
131        column[0] ^= (col as u8).wrapping_mul(0x10) ^ round;
132    }
133}

135/// Modulo-2^64-add round-constant addition (psi-add), used by `T+`/`Q`. Kupyna.pdf Section 6.2.
136///
137/// No production code path calls this directly anymore as of T-134 (`docs/DECISIONS.md` D-85) - see
138/// `sub_shift_mix`'s doc comment.
139#[allow(dead_code)]
140#[allow(clippy::cast_possible_truncation)] // columns - 1 - col < MAX_COLUMNS (16), always fits u8
141fn add_round_constant_add(state: &mut [[u8; ROWS]], round: u8) {
142    let columns = state.len();
143    for (col, column) in state.iter_mut().enumerate() {
144        let top_byte = ((columns - 1 - col) as u8).wrapping_mul(0x10) ^ round;
145        let addend = u64::from_le_bytes([0xF3, 0xF0, 0xF0, 0xF0, 0xF0, 0xF0, 0xF0, top_byte]);
146        let word = u64::from_le_bytes(*column).wrapping_add(addend);
147        *column = word.to_le_bytes();
148    }
149}

151/// Const-generic twin of `add_round_constant_add` - see `sub_shift_mix_n`.
152#[allow(clippy::cast_possible_truncation)] // COLUMNS - 1 - col < COLUMNS <= 16, always fits u8
153fn add_round_constant_add_n<const COLUMNS: usize>(state: &mut [[u8; ROWS]; COLUMNS], round: u8) {
154    for (col, column) in state.iter_mut().enumerate() {
155        let top_byte = ((COLUMNS - 1 - col) as u8).wrapping_mul(0x10) ^ round;
156        let addend = u64::from_le_bytes([0xF3, 0xF0, 0xF0, 0xF0, 0xF0, 0xF0, 0xF0, top_byte]);
157        let word = u64::from_le_bytes(*column).wrapping_add(addend);
158        *column = word.to_le_bytes();
159    }
160}

162/// `T` (Kupyna.pdf Section 6.1): `rounds` iterations of xor-constant -> S-box -> shift -> mix.
163///
164/// No production code path calls this directly anymore as of T-134 (`docs/DECISIONS.md` D-85) - see
165/// `sub_shift_mix`'s doc comment.
166#[allow(dead_code)]
167#[allow(clippy::cast_possible_truncation)] // rounds is 10 or 14 (Table 1), always fits u8
168fn t_transform(state: &mut [[u8; ROWS]], rounds: usize, last_row_shift: usize) {
169    for round in 0..rounds {
170        add_round_constant_xor(state, round as u8);
171        sub_shift_mix(state, last_row_shift);
172    }
173}

214/// One compression step: `h <- T+(m) xor T(h xor m) xor h` (Kupyna.pdf Section 4).
215///
216/// No production code path calls this directly anymore as of T-134 (`docs/DECISIONS.md` D-85) - see
217/// `sub_shift_mix`'s doc comment.
218#[allow(dead_code)]
219fn compress(h: &mut [[u8; ROWS]], block: &[[u8; ROWS]], rounds: usize, last_row_shift: usize) {
220    let columns = h.len();
221    let mut t_input = [[0u8; ROWS]; MAX_COLUMNS];
222    let mut q_input = [[0u8; ROWS]; MAX_COLUMNS];
223    for col in 0..columns {
224        for row in 0..ROWS {
225            t_input[col][row] = h[col][row] ^ block[col][row];
226            q_input[col][row] = block[col][row];
227        }
228    }
229    t_transform(&mut t_input[..columns], rounds, last_row_shift);
230    t_plus_transform(&mut q_input[..columns], rounds, last_row_shift);
231    for col in 0..columns {
232        for row in 0..ROWS {
233            h[col][row] ^= t_input[col][row] ^ q_input[col][row];
234        }
235    }
236}

238/// Const-generic twin of `compress` - see `sub_shift_mix_n`. `t_input`/`q_input` are exactly
239/// `COLUMNS` wide here (not `MAX_COLUMNS`), which is the actual "2x wasted zeroing" T-134 names for
240/// Kupyna-256 (`COLUMNS=8` against a 16-wide scratch buffer) - the round-loop trip count above is
241/// only half of the fix.
242fn compress_n<const COLUMNS: usize, const ROUNDS: usize>(
243    h: &mut [[u8; ROWS]; COLUMNS],
244    block: &[[u8; ROWS]; COLUMNS],
245    last_row_shift: usize,
246) {
247    let mut t_input = [[0u8; ROWS]; COLUMNS];
248    let mut q_input = [[0u8; ROWS]; COLUMNS];
249    for col in 0..COLUMNS {
250        for row in 0..ROWS {
251            t_input[col][row] = h[col][row] ^ block[col][row];
252            q_input[col][row] = block[col][row];
253        }
254    }
255    t_transform_n::<COLUMNS, ROUNDS>(&mut t_input, last_row_shift);
256    t_plus_transform_n::<COLUMNS, ROUNDS>(&mut q_input, last_row_shift);
257    for col in 0..COLUMNS {
258        for row in 0..ROWS {
259            h[col][row] ^= t_input[col][row] ^ q_input[col][row];
260        }
261    }
262}

264/// Splits a `block_bytes`-long byte slice into `columns` column-major 8-byte words.
265///
266/// No production code path calls this directly anymore as of T-134 (`docs/DECISIONS.md` D-85) - see
267/// `sub_shift_mix`'s doc comment.
268#[allow(dead_code)]
269fn bytes_to_columns(bytes: &[u8], columns: usize) -> [[u8; ROWS]; MAX_COLUMNS] {
270    let mut out = [[0u8; ROWS]; MAX_COLUMNS];
271    for col in 0..columns {
272        out[col].copy_from_slice(&bytes[col * ROWS..col * ROWS + ROWS]);
273    }
274    out
275}

277/// Const-generic twin of `bytes_to_columns` - see `sub_shift_mix_n`.
278fn bytes_to_columns_n<const COLUMNS: usize>(bytes: &[u8]) -> [[u8; ROWS]; COLUMNS] {
279    let mut out = [[0u8; ROWS]; COLUMNS];
280    for (col, word) in out.iter_mut().enumerate() {
281        word.copy_from_slice(&bytes[col * ROWS..col * ROWS + ROWS]);
282    }
283    out

286/// Slice-to-array coercion for `KupynaCore::h`'s fixed `MAX_COLUMNS`-wide storage, needed to hand a
287/// `&mut [[u8; ROWS]; COLUMNS]` into the const-generic hot path - copies `hazmat::kalyna`'s
288/// `state_array_mut::<NB>` shape verbatim (`kalyna.rs:425-430`): `unreachable!` instead of
289/// `.unwrap()`/`.expect()` only because `lib.rs` denies both crate-wide (D-19/SECURITY.md), not
290/// because this conversion can actually fail - every call site below passes `COLUMNS` in
291/// `{8, 16}`, both `<= MAX_COLUMNS`.
292fn state_array_mut_kupyna<const COLUMNS: usize>(
293    full: &mut [[u8; ROWS]; MAX_COLUMNS],
294) -> &mut [[u8; ROWS]; COLUMNS] {
295    match (&mut full[..COLUMNS]).try_into() {
296        Ok(array) => array,
297        Err(_) => unreachable!("KupynaCore only ever constructs COLUMNS <= MAX_COLUMNS"),
298    }
299}

301/// Owned-copy counterpart of `state_array_mut_kupyna`, for `finalize`'s `t_final` scratch (which
302/// must be independent of `self.h`, not an alias into it) - same shape, same `unreachable!`
303/// rationale.
304fn h_to_array<const COLUMNS: usize>(h: &[[u8; ROWS]; MAX_COLUMNS]) -> [[u8; ROWS]; COLUMNS] {
305    match h[..COLUMNS].try_into() {
306        Ok(array) => array,
307        Err(_) => unreachable!("KupynaCore only ever constructs columns=8 or 16"),
308    }
309}

311/// Kupyna's own padding formula (`0x80` || zero bytes || 96-bit little-endian bit-length, sized so
312/// the result brings `prefix` up to a whole number of `block_bytes`-byte blocks) applied to an
313/// already-buffered `prefix` with a caller-supplied bit-length field. Shared by `KupynaCore::
314/// finalize`'s own padding and `hazmat::kupyna_kmac`'s two independent padding points (`PAD(K)`,
315/// `PAD(M)`), which each need this formula applied with a length value other than `KupynaCore`'s
316/// own running `total_len` - factored out so there is one implementation of this formula, not
317/// three. Returns the padding buffer and how many of its bytes are actually used.
318pub(crate) fn kupyna_padding(
319    prefix: &[u8],
320    msg_bits: u64,
321    block_bytes: usize,
322) -> ([u8; 2 * MAX_BLOCK_BYTES], usize) {
323    let mut tail = [0u8; 2 * MAX_BLOCK_BYTES];
324    let mut pos = prefix.len();
325    tail[..pos].copy_from_slice(prefix);
326    tail[pos] = 0x80;
327    pos += 1;
328    let used = pos + 12;
329    let zero_bytes = (block_bytes - (used % block_bytes)) % block_bytes;
330    pos += zero_bytes;
331    tail[pos..pos + 8].copy_from_slice(&msg_bits.to_le_bytes());
332    pos += 12; // upper 4 bytes of the 96-bit length field stay zero (message lengths fit in u64 bits)
333    (tail, pos)
334}

336/// Incremental compression state shared by the one-shot `digest_generic`, the public streaming
337/// `Kupyna256Hasher`/`Kupyna512Hasher` below, and `hazmat::kupyna_kmac` - `digest_generic` is now
338/// just `new` + one `update` + `finalize`, so there is exactly one implementation of the
339/// padding/length-tracking logic, not two. `pub(crate)` (not `pub`) since `kupyna_kmac` needs
340/// direct access to feed its two independent padding points into the same running compression
341/// state, a lower level of control than the public `Kupyna*Hasher` streaming API exposes.
342pub(crate) struct KupynaCore {
343    h: [[u8; ROWS]; MAX_COLUMNS],
344    /// Bytes not yet folded into `h` - always fewer than `block_bytes()`, per `update`'s invariant.
345    buffer: [u8; MAX_BLOCK_BYTES],
346    buffer_len: usize,
347    /// Total bytes ever passed to `update` - needed for the padding's message-length field, which
348    /// `finalize` cannot otherwise recover once earlier blocks have already been compressed away.
349    total_len: u64,
350    columns: usize,
351    /// Never read after construction as of T-134 (`docs/DECISIONS.md` D-85): `compress_block`/`finalize`
352    /// now select `ROUNDS` as a const generic via the `columns` match arm instead. Kept as a
353    /// stored field (not removed from `new`'s signature) to avoid rippling a breaking-signature
354    /// change into `kupyna_kmac.rs`'s `kmac_generic`/`kmac_variant!` call sites, which is out of
355    /// this task's scope per its own `advisor()`-reviewed plan (`KupynaCore` stays
356    /// runtime-parameterized end-to-end, only its hot path is const-generic).
357    #[allow(dead_code)]
358    rounds: usize,
359    last_row_shift: usize,
360}

362impl KupynaCore {
363    #[allow(clippy::cast_possible_truncation)] // block_bytes is 64 or 128, always fits u8
364    pub(crate) fn new(columns: usize, rounds: usize, last_row_shift: usize) -> Self {
365        let block_bytes = columns * ROWS;
366        let mut h = [[0u8; ROWS]; MAX_COLUMNS];
367        h[0][0] = block_bytes as u8; // official IV - see docs/pseudocode/kupyna.md "Initial value"
368        Self {
369            h,
370            buffer: [0u8; MAX_BLOCK_BYTES],
371            buffer_len: 0,
372            total_len: 0,
373            columns,
374            rounds,
375            last_row_shift,
376        }
377    }
378
379    pub(crate) fn block_bytes(&self) -> usize {
380        self.columns * ROWS
381    }
382
383    /// The currently-buffered, not-yet-compressed tail of everything fed to `update` so far -
384    /// `kupyna_kmac` needs this to correctly compute `PAD(M)`'s length-field padding relative to
385    /// what's actually still buffered, the same way `finalize` below does for its own padding.
386    pub(crate) fn buffered(&self) -> &[u8] {
387        &self.buffer[..self.buffer_len]
388    }
389
390    /// Dispatches into the const-generic hot path (T-134, `docs/DECISIONS.md` D-85) - `self.columns` is
391    /// always 8 or 16 (verified against every construction call site: `Kupyna256Hasher`/
392    /// `digest_generic`/`kmac_generic` for Kupyna256/Kupyna384/Kupyna512), so this match is a
393    /// one-time-per-block branch on a value that's constant for the lifetime of a given `self`, not
394    /// a per-round cost - see this task's `advisor()`-reviewed plan for why `KupynaCore` itself
395    /// stays runtime-parameterized rather than becoming const-generic end-to-end.
396    fn compress_block(&mut self, block: &[u8]) {
397        match self.columns {
398            8 => {
399                let columns_buf = bytes_to_columns_n::<8>(block);
400                let h = state_array_mut_kupyna::<8>(&mut self.h);
401                compress_n::<8, 10>(h, &columns_buf, self.last_row_shift);
402            }
403            16 => {
404                let columns_buf = bytes_to_columns_n::<16>(block);
405                let h = state_array_mut_kupyna::<16>(&mut self.h);
406                compress_n::<16, 14>(h, &columns_buf, self.last_row_shift);
407            }
408            _ => unreachable!("KupynaCore::new only ever constructed with columns=8 or 16"),
409        }
410    }

412    #[allow(clippy::cast_possible_truncation)] // data.len() here is always << u64::MAX
413    pub(crate) fn update(&mut self, mut data: &[u8]) {
414        self.total_len += data.len() as u64;
415        let block_bytes = self.block_bytes();
416
417        if self.buffer_len > 0 {
418            let take = (block_bytes - self.buffer_len).min(data.len());
419            self.buffer[self.buffer_len..self.buffer_len + take].copy_from_slice(&data[..take]);
420            self.buffer_len += take;
421            data = &data[take..];
422            if self.buffer_len < block_bytes {
423                // Didn't reach a full block - the only way `take` was smaller than the space
424                // available is that `data` ran out, so there is nothing left to process this
425                // call. Returning here (rather than falling through to the remainder-writing
426                // code below) matters: that code unconditionally overwrites `buffer_len` from
427                // `data`'s remainder, which would wipe out this still-partial fill.
428                debug_assert!(data.is_empty());
429                return;
430            }
431            let block = self.buffer;
432            self.compress_block(&block[..block_bytes]);
433            self.buffer_len = 0;
434        }
435
436        let mut full_blocks = data.chunks_exact(block_bytes);
437        for block in &mut full_blocks {
438            self.compress_block(block);
439        }
440        let remainder = full_blocks.remainder();
441        self.buffer[..remainder.len()].copy_from_slice(remainder);
442        self.buffer_len = remainder.len();
443    }

445    /// Consumes `self`: pads the buffered tail with the remaining message-length data, runs the
446    /// output transformation, and returns a 64-byte buffer - callers truncate to `output_bytes`,
447    /// same convention `digest_generic` used to return directly.
448    pub(crate) fn finalize(mut self, output_bytes: usize) -> [u8; 64] {
449        let block_bytes = self.block_bytes();
450
451        // Padding: buffered tail || 0x80 || zero bytes || 96-bit little-endian length, sized to
452        // fill whole block(s) - equivalent to the spec's bit-level `d = (-N-97) mod l` formula for
453        // byte-aligned N (see docs/DECISIONS.md D-10 for the derivation).
454        let msg_bits: u64 = self.total_len * 8;
455        let (tail, pos) = kupyna_padding(&self.buffer[..self.buffer_len], msg_bits, block_bytes);
456
457        let tail_blocks = pos / block_bytes;
458        for i in 0..tail_blocks {
459            self.compress_block(&tail[i * block_bytes..(i + 1) * block_bytes]);
460        }
461
462        // Output transformation: H = R_n(T(h_k) xor h_k) (Kupyna.pdf Section 4). Dispatches into
463        // the const-generic `t_transform_n` (T-134, `docs/DECISIONS.md` D-85) - same rationale as
464        // `compress_block` above: this runs once per `finalize`, not once per round, but for
465        // single-block messages it's a comparable share of total work to one `compress_block`
466        // call, so it gets the same treatment rather than being left on the slow path.
467        //
468        // Two arrays in lockstep by the same index (`self.h`/`t_final`) - same documented false
469        // positive as `state_array_mut_kupyna`'s callers, D-39's established pattern.
470        #[allow(clippy::needless_range_loop)]
471        match self.columns {
472            8 => {
473                let mut t_final = h_to_array::<8>(&self.h);
474                t_transform_n::<8, 10>(&mut t_final, self.last_row_shift);
475                for col in 0..8 {
476                    for row in 0..ROWS {
477                        self.h[col][row] ^= t_final[col][row];
478                    }
479                }
480            }
481            16 => {
482                let mut t_final = h_to_array::<16>(&self.h);
483                t_transform_n::<16, 14>(&mut t_final, self.last_row_shift);
484                for col in 0..16 {
485                    for row in 0..ROWS {
486                        self.h[col][row] ^= t_final[col][row];
487                    }
488                }
489            }
490            _ => unreachable!("KupynaCore only ever constructs columns=8 or 16"),
491        }
492
493        // Truncate to the `output_bytes` most-significant bytes of the column-major byte stream
494        // (mirrors oracles/kupyna-reference/kupyna.c `Trunc`: copies from `state + nbytes -
495        // hash_nbytes`).
496        let mut flat = [0u8; MAX_BLOCK_BYTES];
497        for col in 0..self.columns {
498            flat[col * ROWS..(col + 1) * ROWS].copy_from_slice(&self.h[col]);
499        }
500        let mut out = [0u8; 64];
501        out[..output_bytes].copy_from_slice(&flat[block_bytes - output_bytes..block_bytes]);
502        out
503    }
504}

506/// Shared implementation for both Kupyna variants' one-shot `digest()` - now just `KupynaCore`'s
507/// `new`/`update`/`finalize` with the whole message in one `update` call.
508fn digest_generic(
509    message: &[u8],
510    columns: usize,
511    rounds: usize,
512    last_row_shift: usize,
513    output_bytes: usize,
514) -> [u8; 64] {
515    let mut core = KupynaCore::new(columns, rounds, last_row_shift);
516    core.update(message);
517    core.finalize(output_bytes)
518}

520/// Kupyna-256: 512-bit internal state, 10 rounds, 256-bit (32-byte) output.
521pub struct Kupyna256;
522
523impl Kupyna256 {
524    /// Hashes `message` (byte-aligned only) and returns the 256-bit digest.
525    #[must_use]
526    pub fn digest(message: &[u8]) -> [u8; 32] {
527        let full = digest_generic(message, 8, 10, 7, 32);
528        let mut out = [0u8; 32];
529        out.copy_from_slice(&full[..32]);
530        out
531    }

545/// Streaming Kupyna-256: same construction as [`Kupyna256`], for messages fed incrementally
546/// (e.g. from a reader) instead of available as one contiguous slice up front.
547pub struct Kupyna256Hasher(KupynaCore);
548
549impl Kupyna256Hasher {
550    #[must_use]
551    pub fn new() -> Self {
552        Self(KupynaCore::new(8, 10, 7))
553    }
554
555    /// Feeds more message bytes in. May be called any number of times, with any chunking -
556    /// `Kupyna256Hasher::new().update(a); ...update(b)` is equivalent to one `update(a ++ b)`
557    /// call, and both are equivalent to `Kupyna256::digest(a ++ b)`.
558    pub fn update(&mut self, data: &[u8]) {
559        self.0.update(data);
560    }

565        let full = self.0.finalize(32);
566        let mut out = [0u8; 32];
567        out.copy_from_slice(&full[..32]);
568        out
569    }
570}

648/// `COLUMNS` value the crate actually instantiates (8, 16), rather than re-deriving correctness
