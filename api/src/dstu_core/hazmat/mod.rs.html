31/// A byte count as a DSTU 7624 GCM/GMAC 64-bit bit-length field, or `None` past `2^61 - 1` bytes
32/// (T-283). No real slice gets near that, but the bound is checked here, not assumed.
33pub(crate) fn checked_bit_len(bytes: usize) -> Option<u64> {
34    u64::try_from(bytes).ok()?.checked_mul(8)
35}
36
37#[cfg(test)]
38mod tests {
39    use super::checked_bit_len;
40
41    #[test]
42    fn checked_bit_len_refuses_lengths_whose_bit_count_overflows() {
43        assert_eq!(checked_bit_len(0), Some(0));
44        assert_eq!(checked_bit_len(17), Some(136));
45        let max_ok = u64::MAX / 8;
46        if let Ok(n) = usize::try_from(max_ok) {
47            assert_eq!(checked_bit_len(n), Some(max_ok * 8));
48            if let Some(over) = n.checked_add(1) {
49                assert_eq!(checked_bit_len(over), None);
50            }
51        }
52        if usize::BITS >= 64 {
53            assert_eq!(checked_bit_len(usize::MAX), None);
54        }
55    }
