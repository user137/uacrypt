2//! (Kalyna itself, and Kupyna - which reuses Kalyna's kappa/eta/pi/tau round functions per
3//! `docs/pseudocode/kupyna.md`). Kept in one place, `pub(crate)`-only, so the identity D-10 already
4//! asserts (Kupyna's S-boxes/MDS matrix are byte-identical to Kalyna's `sboxes_enc`/`mds_matrix`)
5//! is structural rather than two hand-copied literals that could silently drift apart.
6//!
7//! `apply_matrix` originally recomputed every `GF(2^8)` multiplication via `gf_mul` on every call
8//! (up to 64 per column). **Switched 2026-07-22 (`docs/DECISIONS.md` D-27) to `MDS_TABLE`/
9//! `MDS_INV_TABLE`**: for each of the 8 input-row positions and each possible byte value, the
10//! combined 8-byte contribution that byte makes to the output column (i.e. `gf_mul(v,
11//! matrix[out_row][in_row])` for every `out_row`, packed into one `u64`) is precomputed - the same
12//! T-table technique already applied to Strumok's `T0..T7` (D-26), just generalized to an
13//! arbitrary matrix instead of one fixed matrix + a fixed set of 4 S-boxes. `apply_matrix` becomes
14//! 8 table lookups + 7 XORs per column instead of 64 `gf_mul` calls. The tables were *generated*,
15//! not hand-transcribed - a one-off Python script computed them from this file's own already-
16//! verified `gf_mul`/`MDS_MATRIX`/`MDS_INV_MATRIX`, then cross-checked the table-based result
17//! against the original loop-based computation over 2000 random columns (0 mismatches) before this
18//! file was written - so correctness here rests on the existing, already-verified `gf_mul` and the
19//! matrices themselves, not on trusting a new external source.

4378/// GF(2^8) multiplication, reduction polynomial x^8+x^4+x^3+x^2+1 (`0x11D`). Mirrors
4379/// `oracles/kalyna-reference/kalyna.c` `MultiplyGF` (and `oracles/kupyna-reference/kupyna.c`
4380/// `MultiplyGF`, identical). Under the default (fused-table) profile this is dead in production,
4381/// kept only for the exhaustive test below; under `small-tables` (`docs/DECISIONS.md` D-35/D-38) it's a
4382/// live production function, called from `apply_matrix_via_gf_mul`/`mds_column_via_gf_mul` below.
4383#[allow(dead_code)]
4384pub(crate) fn gf_mul(mut x: u8, mut y: u8) -> u8 {
4385    let mut result = 0u8;
4386    for _ in 0..8 {
4387        if y & 1 == 1 {
4388            result ^= x;
4389        }
4390        let high_bit = x & 0x80;
4391        x <<= 1;
4392        if high_bit == 0x80 {
4393            x ^= 0x1D;
4394        }
4395        y >>= 1;
4396    }
4397    result
4398}

4400/// Linear layer (tau) via the precomputed `table` (`MDS_TABLE`/`MDS_INV_TABLE`) - the default
4401/// resource profile. See module doc.
4402#[cfg(not(feature = "small-tables"))]
4403fn apply_matrix(state: &mut [[u8; ROWS]], table: &[[u64; 256]; ROWS]) {
4404    for column in state.iter_mut() {
4405        let mut acc = 0u64;
4406        for (in_row, &byte) in column.iter().enumerate() {
4407            acc ^= table[in_row][byte as usize];
4408        }
4409        *column = acc.to_le_bytes();
4410    }

4413/// Linear layer (tau) via `gf_mul` directly against `matrix` (`MDS_MATRIX`/`MDS_INV_MATRIX`) - the
4414/// `small-tables` resource profile (`docs/DECISIONS.md` D-35/D-38): no precomputed table, 64 `gf_mul`
4415/// calls per column instead of 8 table lookups. This is exactly `apply_matrix`'s pre-D-27 form.
4416#[cfg(feature = "small-tables")]
4417fn apply_matrix_via_gf_mul(state: &mut [[u8; ROWS]], matrix: &[[u8; ROWS]; ROWS]) {
4418    for column in state.iter_mut() {

4421            for (in_row, &byte) in column.iter().enumerate() {
4422                *out_byte ^= gf_mul(byte, matrix[out_row][in_row]);
4423            }
4424        }
4425        *column = out;
4426    }
4427}

4453/// One (pre-substitution) byte's combined forward S-box + MDS contribution, computed via
4454/// `gf_mul` on demand instead of looked up from `SBOX_MDS` - the `small-tables` resource profile's
4455/// per-byte primitive for `hazmat::kalyna::encipher_round`/`hazmat::kupyna::sub_shift_mix`'s
4456/// gather-and-XOR loop. Same formula `build_sbox_mds` composes at compile time (`gf_mul(byte,
4457/// matrix[out_row][in_row])` for every `out_row`, packed little-endian), just evaluated per call
4458/// instead of baked into a `[[u64; 256]; ROWS]` table.
4459#[cfg(feature = "small-tables")]
4460fn mds_column_via_gf_mul(matrix: &[[u8; ROWS]; ROWS], in_row: usize, byte: u8) -> u64 {
4461    let mut word = 0u64;
4462    // `out_row` drives both the `matrix` index and the byte-shift amount, not a plain
4463    // single-collection enumerate candidate.
4464    #[allow(clippy::needless_range_loop)]
4465    for out_row in 0..ROWS {
4466        word |= u64::from(gf_mul(byte, matrix[out_row][in_row])) << (8 * out_row);
4467    }
4468    word
4469}

4471/// Combined S-box + MDS forward lookup: `SBOX_MDS[row][byte] = MDS_TABLE[row][SBOXES[row %
4472/// 4][byte]]` - the contribution a raw (pre-substitution) byte sitting at row `row` makes to the
4473/// output column once substituted and mixed. Computed by the compiler at build time by directly
4474/// composing the two already-verified tables above (`const fn`, `while`-loop since `for`/iterators
4475/// aren't allowed in const context) - no hand transcription, no generation script, so there is no
4476/// new correctness risk beyond `SBOXES`/`MDS_TABLE` themselves.
4477///
4478/// This is `nb`/block-size independent, unlike `D-27`'s stated blocker for full S-box+shift+MDS
4479/// fusion assumed: `sub_bytes` substitutes per row, and `shift_rows`/Kupyna's `shift_bytes` permute
4480/// columns while preserving row - the two operations commute, so this table alone (no per-variant
4481/// copies) is enough. The `nb`- or `columns`-dependent part is only the *gather index* used by the
4482/// caller (`hazmat::kalyna::encipher_round`, `hazmat::kupyna::t_transform`/`t_plus_transform`),
4483/// which is cheap arithmetic on `nb`/`shift`, not a table. See `docs/DECISIONS.md` D-28.
4484///
4485/// Not compiled under `small-tables` (`docs/DECISIONS.md` D-35/D-38) - `forward_sbox_mds` below calls
4486/// `mds_column_via_gf_mul` on demand instead in that profile.
4487#[cfg(not(feature = "small-tables"))]
4488const fn build_sbox_mds() -> [[u64; 256]; ROWS] {
4489    let mut table = [[0u64; 256]; ROWS];
4490    let mut row = 0;
4491    while row < ROWS {
4492        let mut byte = 0usize;
4493        while byte < 256 {
4494            let substituted = SBOXES[row % 4][byte] as usize;
4495            table[row][byte] = MDS_TABLE[row][substituted];
4496            byte += 1;
4497        }
4498        row += 1;
4499    }
4500    table
4501}
