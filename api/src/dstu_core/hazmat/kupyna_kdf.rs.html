1//! Kupyna-based KDF - `crypto_kdf` equivalent (`docs/dstu-crypto-project.md` "Mapping onto the
2//! libsodium API").
3//!
4//! **Not a DSTU-specified construction, and not oracle-verified** - a different posture from every
5//! other primitive in this project. There is no separate national KDF standard, and no reference
6//! implementation of "a KDF using Kupyna" exists anywhere to port or cross-check against. Modeled
7//! after libsodium's `crypto_kdf_derive_from_key` *shape* (one master key + an id + a context,
8//! a single keyed-hash call per subkey, no separate Extract stage - deliberately not full RFC 5869
9//! HKDF, whose proof is stated in terms of HMAC specifically, which `hazmat::kupyna_kmac`'s
10//! construction is not) using [`super::kupyna_kmac`] (`docs/TASKS.md` T-38) as the underlying keyed
11//! primitive. Full reasoning and citation in `docs/pseudocode/kupyna-kdf.md`.
12
13use super::kupyna_kmac::{Kupyna256Kmac, Kupyna384Kmac, Kupyna512Kmac};
14
15macro_rules! kdf_variant {
16    ($name:ident, $kmac:ty, $key_len:literal) => {
17        pub struct $name;
18
19        impl $name {
20            /// Derives a subkey from `master_key`, `subkey_id`, and `context` -
21            /// `context || subkey_id` (little-endian) fed as the message to
22            #[doc = concat!("[`", stringify!($kmac), "::mac`].")]
23            /// Different `subkey_id`/`context` values (holding the others fixed) produce
24            /// different subkeys - see `docs/pseudocode/kupyna-kdf.md` for why this can only be
25            /// checked by property test, not against a fixed vector.
26            #[must_use]
27            pub fn derive_subkey(
28                master_key: &[u8; $key_len],
29                subkey_id: u64,
30                context: &[u8; 8],
31            ) -> [u8; $key_len] {
32                let mut message = [0u8; 16];
33                message[..8].copy_from_slice(context);
34                message[8..].copy_from_slice(&subkey_id.to_le_bytes());
35                <$kmac>::mac(master_key, &message)
36                    .expect("master_key's length always matches this KMAC variant's mac_len")
37            }
38        }
39    };
40}
41
42kdf_variant!(Kupyna256Kdf, Kupyna256Kmac, 32);
43kdf_variant!(Kupyna384Kdf, Kupyna384Kmac, 48);
44kdf_variant!(Kupyna512Kdf, Kupyna512Kmac, 64);
