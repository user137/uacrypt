1//! Kalyna-CFB: DSTU 7624:2014 mode of operation #3 ("Гамування зі зворотним зв'язком за
2//! шифротекстом" / cipher feedback) - a keystream `gamma` seeded from the IV, self-updating by
3//! re-encrypting a `feed` register that always absorbs the **ciphertext** `q` bytes at a time
4//! (`q` is a caller-chosen feedback width, one of 1/8/16/32/64 bytes). Cited to
5//! `oracles/uapki/library/uapkic/src/dstu7624.c`'s `encrypt_cfb` (lines 3186-3234), `decrypt_cfb`
6//! (lines 3762-3810), and `dstu7624_init_cfb` (lines 3971-3994). `docs/DECISIONS.md` D-53 has the full
7//! citation and roadmap context.
8//!
9//! # Not a textbook shift register - transcribed exactly, not simplified by analogy
10//!
11//! This construction's `feed` register does **not** literally shift old ciphertext bytes out as
12//! new ones come in the way NIST SP 800-38A's CFB is often diagrammed. Each round, `feed` is
13//! rebuilt as (the just-generated `gamma` block's leading `block_len - q` bytes) with its last `q`
14//! bytes overwritten by the newest `q` ciphertext bytes, then `gamma = E_K(feed)` - `CLAUDE.md`'s
15//! "don't simplify by analogy to textbook X" lesson applies directly here, so this was transcribed
16//! from the C source exactly, verified against 8 official vectors spanning both partial (`q` <
17//! block size) and full (`q` == block size) feedback widths, not assumed correct by inspection.
18//!
19//! # No integrity, and no padding
20//!
21//! **Do not use this for new designs without a specific, understood reason.** Like every raw mode
22//! in this crate, CFB provides no authentication. Prefer [`crate::crypto_secretbox`] unless you
23//! specifically need raw CFB and understand these tradeoffs.
24//!
25//! # Stateful, streaming API, and *not* self-inverse
26//!
27//! Unlike [`super::kalyna_ofb`], `encrypt_in_place` and `decrypt_in_place` are genuinely different
28//! operations here (the C source has two separate functions, not one shared one) - the `feed`
29//! register absorbs the produced output on encrypt but the raw input on decrypt, which happen to
30//! be the same bytes (ciphertext) in both directions, but are read from different sources. Both
31//! methods take `&mut self` and may be called repeatedly over successive chunks of one message -
32//! **but unlike [`super::kalyna_ofb`]/[`super::kalyna_cbc`], call boundaries are not arbitrary.**
33//! Every call except the last must supply a length that is a multiple of `q` - a call boundary
34//! landing mid-way through a `q`-sized group leaves the internal state referencing a position a
35//! later call's own bookkeeping cannot correctly resume from. This is a transcribed property of
36//! `dstu7624.c`'s own construction (its self-test never exercises multi-call chaining with a
37//! non-`q`-aligned boundary either, and the reference has no bounds checking there at all),
38//! confirmed directly (not assumed) by `tests/kalyna_cfb.rs`'s `proptest`, which restricts
39//! intermediate chunk lengths to multiples of `q` for exactly this reason.
40//!
41//! **Checked, not a panic** (`docs/TASKS.md` T-101, `docs/DECISIONS.md`): both methods return
42//! [`Result<(), CfbError>`] and reject a call with `Err(CfbError::NonAlignedIntermediateCall)`
43//! rather than indexing out of bounds if the internal state isn't currently at a `q`-aligned
44//! resume point - `used_gamma_len % q == 0` is checked on entry, which is both necessary and
45//! sufficient given `block_bytes % q == 0` for every `(block_bytes, q)` pair this crate
46//! constructs (`tests/kalyna_cfb.rs`'s `feedback_width_divides_block_length` makes this an
47//! executable fact per variant, not just an argument in this comment).
48
49#[derive(Debug, Clone, Copy, PartialEq, Eq)]
50pub enum CfbError {
51    /// `q` is not one of the DSTU-defined feedback widths (1, 8, 16, 32, or 64 bytes), or exceeds
52    /// the block size.
53    InvalidFeedbackWidth,
54    /// A call was made while the internal state wasn't at a `q`-aligned resume point - the
55    /// previous call ended mid-way through a `q`-sized group (a length that wasn't a multiple of
56    /// `q`) and was not, in fact, the last call. See the module doc comment.
57    NonAlignedIntermediateCall,
58}
59
60macro_rules! kalyna_cfb_variant {
61    ($name:ident, $expanded:ident, $key_bytes:literal, $block_bytes:literal) => {
62        #[doc = concat!(
63            "CFB mode over [`super::kalyna::", stringify!($expanded), "`] - see the module doc ",
64            "comment for the citation and the misuse warning."
65        )]
66        pub struct $name {
67            key: super::kalyna::$expanded,
68            gamma: [u8; $block_bytes],
69            feed: [u8; $block_bytes],
70            q: usize,
71            used_gamma_len: usize,
72        }
73
74        impl $name {
75            /// `q` (the feedback width) must be one of 1, 8, 16, 32, or 64 bytes, and at most the
76            /// block size - matching `dstu7624_init_cfb`'s own `CHECK_PARAM`s exactly.
77            ///
78            /// # Errors
79            ///
80            /// Returns [`CfbError::InvalidFeedbackWidth`] if `q` fails either check.
81            pub fn new(
82                key: &[u8; $key_bytes],
83                iv: &[u8; $block_bytes],
84                q: usize,
85            ) -> Result<Self, CfbError> {
86                if q == 0 || q > $block_bytes || !matches!(q, 1 | 8 | 16 | 32 | 64) {
87                    return Err(CfbError::InvalidFeedbackWidth);
88                }
89                Ok(Self {
90                    key: super::kalyna::$expanded::new(key),
91                    gamma: *iv,
92                    feed: *iv,
93                    q,
94                    used_gamma_len: $block_bytes,
95                })
96            }
97
98            /// Encrypts `buf` in place. See the module doc comment for the `feed`/`gamma`
99            /// bookkeeping this transcribes.
100            ///
101            /// # Errors
102            ///
103            /// Returns [`CfbError::NonAlignedIntermediateCall`] if the internal state isn't at a
104            /// `q`-aligned resume point - see the module doc comment's "not arbitrary" call
105            /// boundary rule.
106            pub fn encrypt_in_place(&mut self, buf: &mut [u8]) -> Result<(), CfbError> {
107                if self.used_gamma_len % self.q != 0 {
108                    return Err(CfbError::NonAlignedIntermediateCall);
109                }
110
111                let mut data_off = 0usize;
112                let mut offset = self.used_gamma_len;
113
114                if offset != 0 {
115                    while offset < self.q && data_off < buf.len() {
116                        buf[data_off] ^= self.gamma[offset];
117                        self.feed[offset] = buf[data_off];
118                        offset += 1;
119                        data_off += 1;
120                    }
121                    if offset == $block_bytes {
122                        self.gamma = self.key.encrypt_block(&self.feed);
123                        offset = $block_bytes - self.q;
124                    }
125                }
126
127                while data_off + self.q <= buf.len() {
128                    for (b, g) in buf[data_off..data_off + self.q]
129                        .iter_mut()
130                        .zip(&self.gamma[offset..offset + self.q])
131                    {
132                        *b ^= *g;
133                    }
134                    self.feed[..$block_bytes].copy_from_slice(&self.gamma);
135                    self.feed[offset..offset + self.q].copy_from_slice(&buf[data_off..data_off + self.q]);
136                    self.gamma = self.key.encrypt_block(&self.feed);
137                    data_off += self.q;
138                }
139                while data_off < buf.len() {
140                    buf[data_off] ^= self.gamma[$block_bytes - (buf.len() - data_off)];
141                    self.feed[offset] = buf[data_off];
142                    offset += 1;
143                    data_off += 1;
144                }
145
146                self.used_gamma_len = offset;
147                Ok(())
148            }
149
150            /// Decrypts `buf` in place. See the module doc comment for why this is not simply
151            /// [`Self::encrypt_in_place`] run again - the `feed` register absorbs the raw
152            /// ciphertext bytes read from `buf`, not the just-computed plaintext.
153            ///
154            /// # Errors
155            ///
156            /// Returns [`CfbError::NonAlignedIntermediateCall`] if the internal state isn't at a
157            /// `q`-aligned resume point - see the module doc comment's "not arbitrary" call
158            /// boundary rule.
159            pub fn decrypt_in_place(&mut self, buf: &mut [u8]) -> Result<(), CfbError> {
160                if self.used_gamma_len % self.q != 0 {
161                    return Err(CfbError::NonAlignedIntermediateCall);
162                }
163
164                let mut data_off = 0usize;
165                let mut offset = self.used_gamma_len;
166
167                if offset != 0 {
168                    while offset < self.q && data_off < buf.len() {
169                        let ciphertext_byte = buf[data_off];
170                        buf[data_off] ^= self.gamma[offset];
171                        self.feed[offset] = ciphertext_byte;
172                        offset += 1;
173                        data_off += 1;
174                    }
175                    if offset == $block_bytes {
176                        self.gamma = self.key.encrypt_block(&self.feed);
177                        offset = $block_bytes - self.q;
178                    }
179                }
180
181                while data_off + self.q <= buf.len() {
182                    let mut ciphertext_chunk = [0u8; $block_bytes];
183                    ciphertext_chunk[..self.q].copy_from_slice(&buf[data_off..data_off + self.q]);
184                    for (b, g) in buf[data_off..data_off + self.q]
185                        .iter_mut()
186                        .zip(&self.gamma[offset..offset + self.q])
187                    {
188                        *b ^= *g;
189                    }
190                    self.feed[..$block_bytes].copy_from_slice(&self.gamma);
191                    self.feed[offset..offset + self.q].copy_from_slice(&ciphertext_chunk[..self.q]);
192                    self.gamma = self.key.encrypt_block(&self.feed);
193                    data_off += self.q;
194                }
195                while data_off < buf.len() {
196                    let ciphertext_byte = buf[data_off];
197                    buf[data_off] ^= self.gamma[$block_bytes - (buf.len() - data_off)];
198                    self.feed[offset] = ciphertext_byte;
199                    offset += 1;
200                    data_off += 1;
201                }
202
203                self.used_gamma_len = offset;
204                Ok(())
205            }
206        }
207    };
208}
