1//! Kalyna-CMAC: DSTU 7624:2014 mode of operation #4 - CBC-MAC over all blocks but the last, then a
2//! final block `XOR`ed against a subkey and encrypted once more. **Not** GF-doubling-subkey
3//! CMAC/OMAC the way AES-CMAC derives its subkeys - transcribed as found, not by analogy to that
4//! more familiar construction. Cited to `oracles/uapki/library/uapkic/src/dstu7624.c`'s
5//! `cmac_update`/`cmac_final` (lines 4221-4310), `padding` (lines 2572-2592), and
6//! `dstu7624_init_cmac` (lines 4070-4087); `Dstu7624Ctx` is confirmed zero-initialized
7//! (`dstu7624_alloc`'s `CALLOC_CHECKED`), so the running state starts at the zero block, not an
8//! IV. `docs/DECISIONS.md` D-54 has the full citation, oracle-coverage breakdown, and roadmap context.
9//!
10//! # Not a secure MAC against chosen messages (T-298)
11//!
12//! The final-block key is `E_K(0)` itself, and after one zero block the CBC state is that same
13//! `E_K(0)`, so the two cancel: `MAC(0^b || Y) = E_K(Y)[..16]`. Anyone who can get chosen
14//! messages authenticated gets a block-encryption oracle under `K` (the full block for a 128-bit
15//! block, a truncated one otherwise), forges tags on messages never authenticated, and - under the
16//! same key - learns GCM's hash key `E_K(0)` and breaks CTR/CCM/KW. This is §9 as written (UAPKI
17//! and Bouncy Castle compute the same), kept for interop with that text only. Use
18//! `crypto_auth` (Kupyna-KMAC) for anything new; `tests/kalyna_cmac.rs` pins the property.
19//!
20//! # One-shot, not streaming
21//!
22//! Unlike [`super::kalyna_cbc`]/[`super::kalyna_cfb`]/etc., this module takes the whole message in
23//! one call rather than exposing incremental `update`. Nothing in this crate consumes an
24//! incremental MAC yet (the same position [`super::kupyna_kmac`] was in before any `crypto_auth`
25//! wrapper existed) - this module follows that module's shape directly rather than re-deriving the
26//! C source's own multi-call buffering state machine for no present benefit.
27//!
28//! # `q` fixed at 16 bytes, not a caller-configurable knob
29//!
30//! The C source allows a caller-chosen tag length `1..=block_len`, but every available oracle
31//! vector (uapki's own KATs and Bouncy Castle's corroborating ones) uses `q = 16` regardless of
32//! block size - it is the only value ever exercised by any oracle. Exposing a wider `q` would ship
33//! an untested code path, which this project's citation discipline forbids.
34//!
35//! # No key separation from any encryption key
36//!
37//! **Do not use this for new designs without a specific, understood reason.** This module provides
38//! no key derivation or domain separation - reusing the same key for CMAC and any encryption mode
39//! in this crate is a misuse risk, not something this API prevents. Prefer a future `crypto_auth`
40//! wrapper once one exists (matching [`super::kupyna_kmac`]'s framing) unless a raw MAC over Kalyna
41//! specifically is genuinely needed.
42
43use subtle::ConstantTimeEq;
44
45#[derive(Debug, Clone, Copy, PartialEq, Eq)]
46pub enum CmacError {
47    /// DSTU 7624:2014 §9 defines CMAC only for `n_o >= 1` blocks (and annex Б.2 pads only `N >= 1`
48    /// bits), so an empty message is outside the standard's domain (T-236, D-206).
49    EmptyMessage,
50    TagMismatch,
51}
52
53macro_rules! kalyna_cmac_variant {
54    ($name:ident, $expanded:ident, $key_bytes:literal, $block_bytes:literal) => {
55        #[doc = concat!(
56            "CMAC over [`super::kalyna::", stringify!($expanded), "`] - see the module doc ",
57            "comment for the citation and the misuse warning."
58        )]
59        pub struct $name;
60
61        impl $name {
62            /// Computes the 16-byte CMAC tag of `message` under `key` - expands the key schedule
63            /// fresh on every call. Prefer [`Self::mac_with_cipher`] for a caller computing a MAC
64            /// for more than one message under the same key (see its doc comment for why).
65            ///
66            /// # Errors
67            ///
68            /// Returns `Err(CmacError::EmptyMessage)` if `message` is empty (D-206).
69            pub fn mac(key: &[u8; $key_bytes], message: &[u8]) -> Result<[u8; 16], CmacError> {
70                let cipher = super::kalyna::$expanded::new(key);
71                Self::mac_with_cipher(&cipher, message)
72            }
73
74            /// Computes the 16-byte CMAC tag of `message` using an already-expanded key schedule -
75            /// the cached-schedule counterpart to [`Self::mac`]. `docs/DECISIONS.md` D-76 / `docs/TASKS.md`
76            /// T-127: `mac` re-derives the full Kalyna round-key schedule on every invocation, an
77            /// avoidable cost (comparable to several block-cipher calls) for any caller computing a MAC for more
78            /// than one message under the same key - this method lets such a caller build the
79            /// schedule once and reuse it, exactly like [`super::kalyna_gcm`]/[`super::kalyna_xts`]
80            /// already do for their own modes.
81            ///
82            /// # Errors
83            ///
84            /// Returns `Err(CmacError::EmptyMessage)` if `message` is empty (D-206).
85            pub fn mac_with_cipher(
86                cipher: &super::kalyna::$expanded,
87                message: &[u8],
88            ) -> Result<[u8; 16], CmacError> {
89                if message.is_empty() {
90                    return Err(CmacError::EmptyMessage);
91                }
92                let len = message.len();
93                let chain_len = if len % $block_bytes == 0 {
94                    len - $block_bytes
95                } else {
96                    (len / $block_bytes) * $block_bytes
97                };
98                let tail = &message[chain_len..];
99
100                let mut state = [0u8; $block_bytes];
101                for block in message[..chain_len].chunks_exact($block_bytes) {
102                    for (s, b) in state.iter_mut().zip(block) {
103                        *s ^= *b;
104                    }
105                    state = cipher.encrypt_block(&state);
106                }
107
108                let mut last_block = [0u8; $block_bytes];
109                last_block[..tail.len()].copy_from_slice(tail);
110                if tail.len() % $block_bytes != 0 {
111                    last_block[tail.len()] = 0x80;
112                }
113
114                let mut rkey = [0u8; $block_bytes];
115                rkey[0] = u8::from(tail.len() % $block_bytes != 0);
116                let rkey = cipher.encrypt_block(&rkey);
117
118                for (s, (l, r)) in state.iter_mut().zip(last_block.iter().zip(rkey.iter())) {
119                    *s ^= *l ^ *r;
120                }
121                let tag_block = cipher.encrypt_block(&state);
122
123                let mut tag = [0u8; 16];
124                tag.copy_from_slice(&tag_block[..16]);
125                Ok(tag)
126            }
127
128            /// Recomputes the CMAC tag and compares it against `expected` in constant time
129            /// (`subtle::ConstantTimeEq`, per `docs/SECURITY.md`'s hard constraint on secret
130            /// comparisons - a MAC tag is exactly this category).
131            ///
132            /// # Errors
133            ///
134            /// Returns `Err(CmacError::EmptyMessage)` for an empty `message`, or
135            /// `Err(CmacError::TagMismatch)` if the recomputed tag doesn't match
136            /// `expected`.
137            pub fn verify(
138                key: &[u8; $key_bytes],
139                message: &[u8],
140                expected: &[u8; 16],
141            ) -> Result<(), CmacError> {
142                let cipher = super::kalyna::$expanded::new(key);
143                Self::verify_with_cipher(&cipher, message, expected)
144            }
145
146            /// The cached-schedule counterpart to [`Self::verify`] - see
147            /// [`Self::mac_with_cipher`]'s doc comment for why this exists.
148            ///
149            /// # Errors
150            ///
151            /// Returns `Err(CmacError::EmptyMessage)` for an empty `message`, or
152            /// `Err(CmacError::TagMismatch)` if the recomputed tag doesn't match
153            /// `expected`.
154            pub fn verify_with_cipher(
155                cipher: &super::kalyna::$expanded,
156                message: &[u8],
157                expected: &[u8; 16],
158            ) -> Result<(), CmacError> {
159                let tag = Self::mac_with_cipher(cipher, message)?;
160                if tag.ct_eq(expected).into() {
161                    Ok(())
162                } else {
163                    Err(CmacError::TagMismatch)
164                }
165            }
166        }
167    };
168}
