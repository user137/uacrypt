1//! Kalyna-CMAC: DSTU 7624:2014 mode of operation #4 - CBC-MAC over all blocks but the last, then a
2//! final block `XOR`ed against a subkey and encrypted once more. **Not** GF-doubling-subkey
3//! CMAC/OMAC the way AES-CMAC derives its subkeys - transcribed as found, not by analogy to that
4//! more familiar construction. Cited to `oracles/uapki/library/uapkic/src/dstu7624.c`'s
5//! `cmac_update`/`cmac_final` (lines 4221-4310), `padding` (lines 2572-2592), and
6//! `dstu7624_init_cmac` (lines 4070-4087); `Dstu7624Ctx` is confirmed zero-initialized
7//! (`dstu7624_alloc`'s `CALLOC_CHECKED`), so the running state starts at the zero block, not an
8//! IV. `docs/DECISIONS.md` D-54 has the full citation, oracle-coverage breakdown, and roadmap context.
9//!
10//! # One-shot, not streaming
11//!
12//! Unlike [`super::kalyna_cbc`]/[`super::kalyna_cfb`]/etc., this module takes the whole message in
13//! one call rather than exposing incremental `update`. Nothing in this crate consumes an
14//! incremental MAC yet (the same position [`super::kupyna_kmac`] was in before any `crypto_auth`
15//! wrapper existed) - this module follows that module's shape directly rather than re-deriving the
16//! C source's own multi-call buffering state machine for no present benefit.
17//!
18//! # `q` fixed at 16 bytes, not a caller-configurable knob
19//!
20//! The C source allows a caller-chosen tag length `1..=block_len`, but every available oracle
21//! vector (uapki's own KATs and Bouncy Castle's corroborating ones) uses `q = 16` regardless of
22//! block size - it is the only value ever exercised by any oracle. Exposing a wider `q` would ship
23//! an untested code path, which this project's citation discipline forbids.
24//!
25//! # No key separation from any encryption key
26//!
27//! **Do not use this for new designs without a specific, understood reason.** This module provides
28//! no key derivation or domain separation - reusing the same key for CMAC and any encryption mode
29//! in this crate is a misuse risk, not something this API prevents. Prefer a future `crypto_auth`
30//! wrapper once one exists (matching [`super::kupyna_kmac`]'s framing) unless a raw MAC over Kalyna
31//! specifically is genuinely needed.
32
33use subtle::ConstantTimeEq;
34
35#[derive(Debug, Clone, Copy, PartialEq, Eq)]
36pub enum CmacError {
37    TagMismatch,
38}
39
40macro_rules! kalyna_cmac_variant {
41    ($name:ident, $expanded:ident, $key_bytes:literal, $block_bytes:literal) => {
42        #[doc = concat!(
43            "CMAC over [`super::kalyna::", stringify!($expanded), "`] - see the module doc ",
44            "comment for the citation and the misuse warning."
45        )]
46        pub struct $name;
47
48        impl $name {
49            /// Computes the 16-byte CMAC tag of `message` under `key` - expands the key schedule
50            /// fresh on every call. Prefer [`Self::mac_with_cipher`] for a caller computing a MAC
51            /// for more than one message under the same key (see its doc comment for why).
52            #[must_use]
53            pub fn mac(key: &[u8; $key_bytes], message: &[u8]) -> [u8; 16] {
54                let cipher = super::kalyna::$expanded::new(key);
55                Self::mac_with_cipher(&cipher, message)
56            }
57
58            /// Computes the 16-byte CMAC tag of `message` using an already-expanded key schedule -
59            /// the cached-schedule counterpart to [`Self::mac`]. `docs/DECISIONS.md` D-76 / `docs/TASKS.md`
60            /// T-127: `mac` re-derives the full Kalyna round-key schedule on every invocation, an
61            /// avoidable cost (comparable to several block-cipher calls) for any caller computing a MAC for more
62            /// than one message under the same key - this method lets such a caller build the
63            /// schedule once and reuse it, exactly like [`super::kalyna_gcm`]/[`super::kalyna_xts`]
64            /// already do for their own modes.
65            #[must_use]
66            pub fn mac_with_cipher(cipher: &super::kalyna::$expanded, message: &[u8]) -> [u8; 16] {
67                let len = message.len();
68                let chain_len = if len == 0 {
69                    0
70                } else if len % $block_bytes == 0 {
71                    len - $block_bytes
72                } else {
73                    (len / $block_bytes) * $block_bytes
74                };
75                let tail = &message[chain_len..];
76
77                let mut state = [0u8; $block_bytes];
78                for block in message[..chain_len].chunks_exact($block_bytes) {
79                    for (s, b) in state.iter_mut().zip(block) {
80                        *s ^= *b;
81                    }
82                    state = cipher.encrypt_block(&state);
83                }
84
85                let mut last_block = [0u8; $block_bytes];
86                last_block[..tail.len()].copy_from_slice(tail);
87                if tail.len() % $block_bytes != 0 {
88                    last_block[tail.len()] = 0x80;
89                }
90
91                let mut rkey = [0u8; $block_bytes];
92                rkey[0] = u8::from(tail.len() % $block_bytes != 0);
93                let rkey = cipher.encrypt_block(&rkey);
94
95                for (s, (l, r)) in state.iter_mut().zip(last_block.iter().zip(rkey.iter())) {
96                    *s ^= *l ^ *r;
97                }
98                let tag_block = cipher.encrypt_block(&state);
99
100                let mut tag = [0u8; 16];
101                tag.copy_from_slice(&tag_block[..16]);
102                tag
103            }
104
105            /// Recomputes the CMAC tag and compares it against `expected` in constant time
106            /// (`subtle::ConstantTimeEq`, per `docs/SECURITY.md`'s hard constraint on secret
107            /// comparisons - a MAC tag is exactly this category).
108            ///
109            /// # Errors
110            ///
111            /// Returns `Err(CmacError::TagMismatch)` if the recomputed tag doesn't match
112            /// `expected`.
113            pub fn verify(
114                key: &[u8; $key_bytes],
115                message: &[u8],
116                expected: &[u8; 16],
117            ) -> Result<(), CmacError> {
118                let cipher = super::kalyna::$expanded::new(key);
119                Self::verify_with_cipher(&cipher, message, expected)
120            }
121
122            /// The cached-schedule counterpart to [`Self::verify`] - see
123            /// [`Self::mac_with_cipher`]'s doc comment for why this exists.
124            ///
125            /// # Errors
126            ///
127            /// Returns `Err(CmacError::TagMismatch)` if the recomputed tag doesn't match
128            /// `expected`.
129            pub fn verify_with_cipher(
130                cipher: &super::kalyna::$expanded,
131                message: &[u8],
132                expected: &[u8; 16],
133            ) -> Result<(), CmacError> {
134                let tag = Self::mac_with_cipher(cipher, message);
135                if tag.ct_eq(expected).into() {
136                    Ok(())
137                } else {
138                    Err(CmacError::TagMismatch)
139                }
140            }
141        }
142    };
143}
