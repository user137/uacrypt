1//! Twisted Edwards point arithmetic over E512/1 (T-192 Phase 2 - see
2//! `docs/pseudocode/dstu9041.md`, `docs/DECISIONS.md` D-176/D-177). Curve equation (same
3//! x/y-swapped form as `curve256.rs`, per the primary text): `x^2 + a*y^2 = d*x^2*y^2 (mod p)`,
4//! `a=2`, `d=0x10D`. Direct sibling of `curve256.rs` at the 512-bit field width - same addition
5//! law (Додаток Б.4), same complete-formula reasoning (`d` non-square, guaranteed by 3.18/7.2 for
6//! every recommended curve), same `Point{x,y}`-with-no-`Infinity`-variant shape.
7//!
8//! **Finding 1/2 re-derived for E512/1 specifically, not copied from E256/1 (D-176)**: cofactor 4
9//! (independently re-checked via the Hasse-interval method), and `x=p-1` reconstructs a genuine
10//! order-2 point outside `<P>` (pure algebra from the curve's own `y=0` cross-section, so it
11//! recurs structurally - still verified here, in `tests/dstu9041_curve_512.rs`, not assumed).
12//! [`point_from_x`] below closes both the same way `curve256.rs`'s does: explicit exclusion of
13//! `x in {0,1,p-1}`/`x^2=a*d^-1`, plus a general subgroup-membership check
14//! (`n*candidate == NEUTRAL`) that catches order-4 points too, independent of locating one by
15//! coordinates.
16

54#[derive(Clone, Copy, Debug, PartialEq, Eq)]
55pub struct Point {
56    pub x: FieldElement,
57    pub y: FieldElement,
58}
59
60impl Point {
61    /// The neutral element `(1, 0)`.
62    pub const NEUTRAL: Self = Self {
63        x: FieldElement::ONE,
64        y: FieldElement::ZERO,
65    };
66
67    /// Checks `x^2 + a*y^2 == d*x^2*y^2 + 1 (mod p)`. Public-data check, matches `curve256.rs`'s
68    /// own reasoning for why an ordinary branch is fine here.
69    #[must_use]
70    pub fn is_on_curve(self) -> bool {
71        let x2 = self.x.square();
72        let y2 = self.y.square();
73        let lhs = x2.add(curve_a().multiply(y2));
74        let rhs = curve_d().multiply(x2).multiply(y2).add(FieldElement::ONE);
75        lhs == rhs
76    }
77
78    /// Point addition (also correct for doubling and the neutral element - the formula is
79    /// complete). A convenience wrapper around `ProjectivePoint::add`.
80    #[must_use]
81    #[allow(clippy::should_implement_trait)]
82    pub fn add(self, other: Self) -> Self {
83        ProjectivePoint::from_affine(self)
84            .add(ProjectivePoint::from_affine(other))
85            .to_affine()
86    }
87
88    /// Constant-time scalar multiplication, fixed 512 iterations MSB-first, `scalar` big-endian.
89    #[must_use]
90    pub fn scalar_multiply(self, scalar: &[u8; 64]) -> Self {
91        let base = ProjectivePoint::from_affine(self);
92        let mut acc = ProjectivePoint::from_affine(Self::NEUTRAL);
93        for &byte in scalar {
94            for bit_idx in (0..8).rev() {
95                acc = acc.add(acc);
96                let bit = u64::from((byte >> bit_idx) & 1);
97                let candidate = acc.add(base);
98                acc = ProjectivePoint::select(bit, candidate, acc);
99            }
100        }
101        acc.to_affine()
102    }
103}

105#[derive(Clone, Copy, Debug)]
106struct ProjectivePoint {
107    x: FieldElement,
108    y: FieldElement,
109    z: FieldElement,
110}
111
112impl ProjectivePoint {
113    fn from_affine(p: Point) -> Self {
114        Self {
115            x: p.x,
116            y: p.y,
117            z: FieldElement::ONE,
118        }
119    }
120
121    /// The one inversion point in the whole scalar-multiplication ladder - see
122    /// `curve256.rs::ProjectivePoint::to_affine`'s identical doc comment on the traced
123    /// constant-time dependency this composes into.
124    fn to_affine(self) -> Point {
125        let z_inv = self.z.invert();
126        Point {
127            x: self.x.multiply(z_inv),
128            y: self.y.multiply(z_inv),
129        }
130    }
131
132    /// Додаток Б.4's complete addition law - identical to `curve256.rs`'s own, just over the
133    /// 512-bit field.
134    #[allow(clippy::many_single_char_names)]
135    fn add(self, other: Self) -> Self {
136        let a = curve_a();
137        let d = curve_d();
138
139        let zz = self.z.multiply(other.z);
140        let b = zz.square();
141        let c = self.x.multiply(other.x);
142        let dd = self.y.multiply(other.y);
143        let e = d.multiply(c).multiply(dd);
144        let f = b.sub(e);
145        let g = b.add(e);
146
147        let x_sum = self.x.add(self.y);
148        let y_sum = other.x.add(other.y);
149        let cross = x_sum.multiply(y_sum);
150
151        let x_r = zz.multiply(g).multiply(c.sub(a.multiply(dd)));
152        let y_r = zz.multiply(f).multiply(cross.sub(c).sub(dd));
153        let z_r = f.multiply(g);
154
155        Self {
156            x: x_r,
157            y: y_r,
158            z: z_r,
159        }
160    }
161
162    fn select(bit: u64, a: Self, b: Self) -> Self {
163        Self {
164            x: FieldElement::select(bit, a.x, b.x),
165            y: FieldElement::select(bit, a.y, b.y),
166            z: FieldElement::select(bit, a.z, b.z),
167        }
168    }
169}

171/// Reconstructs a point on the curve from just its `x`-coordinate - direct sibling of
172/// `curve256.rs::point_from_x`, same rejection gauntlet: `x in {0, 1, p-1}`, `x^2 = a*d^-1`, `v`
173/// not a quadratic residue, or the reconstructed candidate outside `<P>`.
174#[must_use]
175pub fn point_from_x(x: FieldElement) -> Option<Point> {
176    let a = curve_a();
177    let d = curve_d();
178    let p_minus_1 = FieldElement::ZERO.sub(FieldElement::ONE);
179    let x_squared = x.square();
180
181    if x == FieldElement::ZERO
182        || x == FieldElement::ONE
183        || x == p_minus_1
184        || x_squared == a.multiply(d.invert())
185    {
186        return None;
187    }
188
189    let numerator = FieldElement::ONE.sub(x_squared);
190    let denominator = a.sub(d.multiply(x_squared));
191    let v = numerator.multiply(denominator.invert());
192
193    if !v.euler_criterion() {
194        return None;
195    }
196
197    let candidate = Point { x, y: v.sqrt() };
198
199    if candidate.scalar_multiply(&order()) != Point::NEUTRAL {
200        return None;
201    }
202
203    Some(candidate)
204}

219#[allow(clippy::needless_range_loop)]
220fn bytes_be_to_limbs(bytes: &[u8; 64]) -> [u64; 8] {
221    let mut limbs = [0u64; 8];
222    let mut limb_bytes = [0u8; 8];
223    for i in 0..8 {
224        limb_bytes.copy_from_slice(&bytes[i * 8..i * 8 + 8]);
225        limbs[7 - i] = u64::from_be_bytes(limb_bytes);
226    }
227    limbs
228}

230/// `true` iff `a < b`, both interpreted as big-endian unsigned 512-bit integers. Branchless
231/// (fixed 8-limb `sbb` chain), no data-dependent control flow.
232#[allow(clippy::needless_range_loop)]
233fn is_less_than(a: &[u8; 64], b: &[u8; 64]) -> bool {
234    let al = bytes_be_to_limbs(a);
235    let bl = bytes_be_to_limbs(b);
236    let mut borrow = 0u64;
237    for i in 0..8 {
238        let (_, bw) = sbb(al[i], bl[i], borrow);
239        borrow = bw;
240    }
241    borrow == 1
242}

244/// `1 < scalar < n-1`, constant-time - same precondition `curve256.rs::is_valid_scalar` enforces.
245#[must_use]
246pub fn is_valid_scalar(scalar: &[u8; 64]) -> bool {
247    let mut one = [0u8; 64];
248    one[63] = 1;
249    let mut n_minus_1 = ORDER_N;
250    // n is odd (prime) - decrementing its last byte alone never borrows across bytes.
251    n_minus_1[63] -= 1;
252
253    let scalar_gt_one = is_less_than(&one, scalar);
254    let scalar_lt_n_minus_1 = is_less_than(scalar, &n_minus_1);
255    scalar_gt_one & scalar_lt_n_minus_1
256}
