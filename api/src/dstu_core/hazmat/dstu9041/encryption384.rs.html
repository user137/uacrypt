1//! Encrypt/decrypt composition for DSTU 9041's `l(p)=384` case (T-297 - clauses 11/12, the clause
2//! set `encryption.rs`/`encryption512.rs` cite; `docs/DECISIONS.md` D-241/D-244).
3//!
4//! Two differences from 256/512, both from the text:
5//! - The Kalyna-KW key is `kappa = x_T^(k)`, the low-order `k = 256` bits of `x_T` (clause 5.2;
6//!   Table 2 gives Kalyna-256/256 for `l(p)=384`). At 256/512 `k` equals `l(p)`, so all of `x_T`.
7//! - `M'` is 48 bytes, not a multiple of the 32-byte block, so `t` is Kalyna-256/256-KW-p over `M'`
8//!   itself (DSTU 7624 §15.4, [`crate::hazmat::kalyna_kw_p`]). 256/512 wrap `M' || 0^l` with plain
9//!   KW, the convention their worked examples confirmed (D-165/D-180).
10//!
11//! Додаток Г.2 reproduces here up to `kappa`, `H` and `M'`; its printed `t` does not follow §15.4,
12//! so this module neither produces it nor accepts the document's `C` (D-241, the owner's choice).
13//!
14//! `DecryptError` is collapsed to one variant, and `point_from_x` carries the Finding 1/2 guards
15//! (re-derived for E384/1 in D-241), as in `encryption.rs`/`encryption512.rs`.
16
17use super::curve384::{base_point, is_valid_scalar, point_from_x, Point};
18use super::fp384::from_candidate_bytes;
19use super::message384::{
20    build_m_prime, encode_l_m_tilde, format_m_tilde, parse_m_prime, M_PRIME_BYTES,
21};
22use crate::hazmat::kalyna_kw_p::Kalyna256_256KwP;
23use zeroize::Zeroizing;

25/// Hash function identifier for Kupyna-256 (clause 5.7's `i_H`), the worked example's `01`.
26const HASH_ID_KUPYNA256: u8 = 0x01;
27
28/// `t`'s length: KW-p over the 48-byte `M'` (48 + 16 + the always-present Б.2 block = 96, plus KW's
29/// checksum block).
30const T_BYTES: usize = 128;
31/// `r || t`.
32pub const CIPHERTEXT_BYTES: usize = 48 + T_BYTES;
33
34#[derive(Debug, Clone, Copy, PartialEq, Eq)]
35pub enum EncryptError {
36    /// `message_bits` is `0`, exceeds `L_MAX_P`, or `message`'s length doesn't match it.
37    InvalidMessage,
38    /// `epsilon` is outside the valid scalar range `{2, ..., n-2}`.
39    InvalidEphemeralKey,
40    /// `q` is off the curve or of order 1, 2 or 4 (T-307).
41    InvalidPublicKey,
42}
43
44/// T-307: `q` is on the curve and its order is not 1, 2 or 4, so `T = eps*q` is not computable
45/// without `eps`. `4q` takes two additions; a point at infinity on the way comes out of the
46/// projective formula as `(0, 0)`, which is off the curve, so it is refused too. Cheap on purpose:
47/// a large-order `q` with a small-order component is the recipient's own problem (it cannot
48/// decrypt), not a leak - the full `n*q` check stays in `point_from_x` for keys built from bytes.
49fn is_usable_recipient(q: Point) -> bool {
50    let q2 = q.add(q);
51    let q4 = q2.add(q2);
52    q.is_on_curve() && q4 != Point::NEUTRAL && q4.is_on_curve()
53}
54
55#[derive(Debug, Clone, Copy, PartialEq, Eq)]
56pub enum DecryptError {
57    /// Any late-stage failure (bad `r`, KW-p integrity, hash mismatch, bad padding) or an invalid
58    /// `e` - deliberately collapsed, see this module's doc comment.
59    InvalidCiphertext,
60}
61
62/// `kappa = x_T^(k)`: the low-order 32 bytes of the big-endian `x_T`.
63fn kappa_from_x(x: [u8; 48]) -> Zeroizing<[u8; 32]> {
64    let x = Zeroizing::new(x);
65    let mut kappa = Zeroizing::new([0u8; 32]);
66    kappa.copy_from_slice(&x[16..]);
67    kappa
68}
69
70/// Clause 11, steps 2-15. Returns `r || t` (48 + 128 bytes).
71///
72/// # Errors
73///
74/// See [`EncryptError`]'s variants.
75pub fn encrypt(
76    message: &[u8],
77    message_bits: usize,
78    q: Point,
79    epsilon: &[u8; 48],
80) -> Result<[u8; CIPHERTEXT_BYTES], EncryptError> {
81    if !is_valid_scalar(epsilon) {
82        return Err(EncryptError::InvalidEphemeralKey);
83    }
84    if !is_usable_recipient(q) {
85        return Err(EncryptError::InvalidPublicKey);
86    }
87    // T-302: `m_tilde` is the plaintext (the KEM seed under `crypto_box*`).
88    let m_tilde = Zeroizing::new(
89        format_m_tilde(message, message_bits).map_err(|_| EncryptError::InvalidMessage)?,
90    );
91    let l_m_tilde = encode_l_m_tilde(message_bits);
92    let m_prime = Zeroizing::new(build_m_prime(HASH_ID_KUPYNA256, &m_tilde, &l_m_tilde));
93
94    let r_point = base_point().scalar_multiply(epsilon);
95    let r_bytes = r_point.x.to_be_bytes();
96
97    let t_point = Zeroizing::new(q.scalar_multiply(epsilon));
98    let kappa = kappa_from_x(t_point.x.to_be_bytes());
99
100    let mut ciphertext = [0u8; CIPHERTEXT_BYTES];
101    ciphertext[..48].copy_from_slice(&r_bytes);
102    // `M'` is always 48 bytes and `t` 128, the one shape `wrap` accepts for that length; its only
103    // failure (`KwError::InvalidLength`) cannot occur, and is mapped rather than unwrapped so this
104    // stays panic-free by construction (the `encryption512.rs` reasoning).
105    Kalyna256_256KwP::wrap(&kappa, &m_prime[..], &mut ciphertext[48..])
106        .map_err(|_| EncryptError::InvalidMessage)?;
107    Ok(ciphertext)
108}

110/// Clause 12, steps 1-19. Returns the recovered `M~` (37 bytes, left-padded - the caller takes the
111/// low-order `l(M)` bits) and `l(M)`. Needs no public key (`T' = e*R'`), as `encryption.rs::decrypt`.
112///
113/// # Errors
114///
115/// [`DecryptError::InvalidCiphertext`] for any tampered ciphertext, invalid `e`, or malformed `r` -
116/// deliberately not distinguished.
117pub fn decrypt(
118    ciphertext: &[u8; CIPHERTEXT_BYTES],
119    e: &[u8; 48],
120) -> Result<([u8; 37], usize), DecryptError> {
121    if !is_valid_scalar(e) {
122        return Err(DecryptError::InvalidCiphertext);
123    }
124
125    let mut r_bytes = [0u8; 48];
126    r_bytes.copy_from_slice(&ciphertext[..48]);
127    let r_field = from_candidate_bytes(&r_bytes).ok_or(DecryptError::InvalidCiphertext)?;
128
129    // Steps 2-6 plus the subgroup check beyond clause 12's literal text: `curve384::point_from_x`.
130    let r_prime = point_from_x(r_field).ok_or(DecryptError::InvalidCiphertext)?;
131
132    // Steps 7-8: T' = e*R'; kappa = x_T'^(k).
133    let t_prime = Zeroizing::new(r_prime.scalar_multiply(e));
134    let kappa = kappa_from_x(t_prime.x.to_be_bytes());
135
136    // Step 9: KW-p unwrap; its checksum and padding checks are constant-time (D-243).
137    let mut recovered = Zeroizing::new([0u8; T_BYTES - 32]);
138    let len = Kalyna256_256KwP::unwrap(&kappa, &ciphertext[48..], &mut recovered[..])
139        .map_err(|_| DecryptError::InvalidCiphertext)?;
140    // A valid KW-p of any other length is not an M' for this curve.
141    if len != M_PRIME_BYTES {
142        return Err(DecryptError::InvalidCiphertext);
143    }
144    let mut m_prime = Zeroizing::new([0u8; M_PRIME_BYTES]);
145    m_prime.copy_from_slice(&recovered[..M_PRIME_BYTES]);
146
147    // Steps 10-17 (hash and zero-padding checks, both constant-time - see message384.rs).
148    let parsed = parse_m_prime(&m_prime).map_err(|_| DecryptError::InvalidCiphertext)?;
149    Ok((parsed.m_tilde, parsed.bit_length))
150}
