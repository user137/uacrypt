1//! `F_p` arithmetic for DSTU 9041's `l(p)=512` case, E512/1's prime `p = 2^512 - 875` (T-192
2//! Phase 1 - see `docs/pseudocode/dstu9041.md`, `docs/DECISIONS.md` D-176 for the citation and
3//! independent verification this modulus was checked against). `p`'s closeness to a power of two
4//! (a 10-bit complement `C=875`) is what `reduce_wide` exploits: `2^512 ≡ C (mod p)`, so a
5//! 1024-bit product folds down via a couple of small multiply-and-add passes instead of general
6//! long division - the same shape `fp256.rs` uses for `l(p)=256`'s `p = 2^256 - 435`, confirmed
7//! (not assumed) to carry over once `p`'s own subtrahend was independently transcribed (D-176).
8//!
9//! Fixed-width `[u64; 8]` (little-endian limbs), mirroring `fp256.rs`'s own `[u64; 4]` shape - a
10//! sibling module, not a generic-over-width one, per this project's existing `gf2m163`/`fp256`
11//! precedent of one fixed-width type per field rather than premature generics.
12

64#[inline]
65fn adc(a: u64, b: u64, carry_in: u64) -> (u64, u64) {
66    let (s1, c1) = a.overflowing_add(b);
67    let (s2, c2) = s1.overflowing_add(carry_in);
68    (s2, u64::from(c1) | u64::from(c2))
69}
70
71#[inline]
72pub(crate) fn sbb(a: u64, b: u64, borrow_in: u64) -> (u64, u64) {
73    let (d1, b1) = a.overflowing_sub(b);
74    let (d2, b2) = d1.overflowing_sub(borrow_in);
75    (d2, u64::from(b1) | u64::from(b2))
76}

78#[inline]
79#[allow(clippy::needless_range_loop)]
80#[allow(clippy::many_single_char_names)]
81fn add_limbs(a: [u64; 8], b: [u64; 8]) -> ([u64; 8], u64) {
82    let mut r = [0u64; 8];
83    let mut carry = 0u64;
84    for i in 0..8 {
85        let (s, c) = adc(a[i], b[i], carry);
86        r[i] = s;
87        carry = c;
88    }
89    (r, carry)
90}

92/// Subtracts `p` from `x` if `x >= p`, otherwise returns `x` unchanged - branchless (mask-select),
93/// no data-dependent control flow.
94#[inline]
95#[allow(clippy::needless_range_loop)]
96fn conditional_sub_p(x: [u64; 8]) -> [u64; 8] {
97    let mut t = [0u64; 8];
98    let mut borrow = 0u64;
99    for i in 0..8 {
100        let (d, bw) = sbb(x[i], P_LIMBS[i], borrow);
101        t[i] = d;
102        borrow = bw;
103    }
104    let mask = 0u64.wrapping_sub(borrow ^ 1);
105    let mut out = [0u64; 8];
106    for i in 0..8 {
107        out[i] = (t[i] & mask) | (x[i] & !mask);
108    }
109    out
110}

112/// `a * c` for a small constant `c < 2^16`, returning the low 512 bits plus a tiny overflow limb.
113#[inline]
114#[allow(clippy::needless_range_loop)]
115#[allow(clippy::cast_possible_truncation)]
116fn mul_small(a: [u64; 8], c: u64) -> ([u64; 8], u64) {
117    let mut r = [0u64; 8];
118    let mut carry: u128 = 0;
119    for i in 0..8 {
120        let sum = u128::from(a[i]) * u128::from(c) + carry;
121        r[i] = sum as u64;
122        carry = sum >> 64;
123    }
124    (r, carry as u64)
125}

127/// Schoolbook 8x8-limb multiply producing a 16-limb (1024-bit) wide product. Carry propagation
128/// after each row is a fixed-length pass over the remaining limbs (bounded by the row index `i`,
129/// a public loop position, not secret data) - no early exit.
130#[inline]
131#[allow(clippy::needless_range_loop)]
132#[allow(clippy::cast_possible_truncation)]
133fn wide_mul(a: [u64; 8], b: [u64; 8]) -> [u64; 16] {
134    let mut r = [0u64; 16];
135    for i in 0..8 {
136        let mut carry: u128 = 0;
137        for j in 0..8 {
138            let idx = i + j;
139            let sum = u128::from(r[idx]) + u128::from(a[i]) * u128::from(b[j]) + carry;
140            r[idx] = sum as u64;
141            carry = sum >> 64;
142        }
143        let mut idx = i + 8;
144        let mut c = carry as u64;
145        while idx < 16 {
146            let sum = u128::from(r[idx]) + u128::from(c);
147            r[idx] = sum as u64;
148            c = (sum >> 64) as u64;
149            idx += 1;
150        }
151    }
152    r
153}

155/// Folds a 1024-bit wide product down to a fully-reduced `< p` value, using `2^512 ≡ C (mod p)`.
156fn reduce_wide(wide: [u64; 16]) -> FieldElement {
157    let low: [u64; 8] = [
158        wide[0], wide[1], wide[2], wide[3], wide[4], wide[5], wide[6], wide[7],
159    ];
160    let high: [u64; 8] = [
161        wide[8], wide[9], wide[10], wide[11], wide[12], wide[13], wide[14], wide[15],
162    ];
163
164    let (hc, hc_top) = mul_small(high, C);
165    let (acc, carry) = add_limbs(low, hc);
166    let overflow = carry + hc_top;
167
168    let addition = overflow.wrapping_mul(C);
169    let (acc2, carry2) = add_limbs(acc, [addition, 0, 0, 0, 0, 0, 0, 0]);
170
171    let addition2 = carry2.wrapping_mul(C);
172    let (acc3, _carry3) = add_limbs(acc2, [addition2, 0, 0, 0, 0, 0, 0, 0]);
173
174    let r1 = conditional_sub_p(acc3);
175    let r2 = conditional_sub_p(r1);
176    let r3 = conditional_sub_p(r2);
177    FieldElement(r3)
178}

180impl FieldElement {
181    pub const ZERO: Self = Self([0, 0, 0, 0, 0, 0, 0, 0]);
182    pub const ONE: Self = Self([1, 0, 0, 0, 0, 0, 0, 0]);

184    /// Interprets `bytes` (big-endian) as a field element. Precondition: `bytes` encodes a value
185    /// `< p` - callers that can't guarantee this should use [`from_candidate_bytes`] instead.
186    #[must_use]
187    #[allow(clippy::needless_range_loop)]
188    pub fn from_be_bytes(bytes: &[u8; 64]) -> Self {
189        let mut limbs = [0u64; 8];
190        let mut limb_bytes = [0u8; 8];
191        for i in 0..8 {
192            limb_bytes.copy_from_slice(&bytes[i * 8..i * 8 + 8]);
193            limbs[7 - i] = u64::from_be_bytes(limb_bytes);
194        }
195        Self(limbs)
196    }
197
198    #[must_use]
199    #[allow(clippy::needless_range_loop)]
200    pub fn to_be_bytes(self) -> [u8; 64] {
201        let mut out = [0u8; 64];
202        for i in 0..8 {
203            out[i * 8..i * 8 + 8].copy_from_slice(&self.0[7 - i].to_be_bytes());
204        }
205        out
206    }
207
208    #[must_use]
209    #[allow(clippy::should_implement_trait)]
210    #[allow(clippy::needless_range_loop)]
211    pub fn add(self, other: Self) -> Self {
212        let (r, carry) = add_limbs(self.0, other.0);
213        let mut t = [0u64; 8];
214        let mut borrow = 0u64;
215        for i in 0..8 {
216            let (d, bw) = sbb(r[i], P_LIMBS[i], borrow);
217            t[i] = d;
218            borrow = bw;
219        }
220        let take_t = carry | (borrow ^ 1);
221        let mask = 0u64.wrapping_sub(take_t);
222        let mut out = [0u64; 8];
223        for i in 0..8 {
224            out[i] = (t[i] & mask) | (r[i] & !mask);
225        }
226        Self(out)
227    }
228
229    #[must_use]
230    #[allow(clippy::should_implement_trait)]
231    #[allow(clippy::needless_range_loop)]
232    pub fn sub(self, other: Self) -> Self {
233        let mut r = [0u64; 8];
234        let mut borrow = 0u64;
235        for i in 0..8 {
236            let (d, bw) = sbb(self.0[i], other.0[i], borrow);
237            r[i] = d;
238            borrow = bw;
239        }
240        // borrow==1 means self < other: r currently equals (self - other) mod 2^512, i.e.
241        // (self-other)+2^512; subtracting C once corrects it to (self-other)+p, landing in [0,p).
242        let mask = 0u64.wrapping_sub(borrow);
243        let mut out = [0u64; 8];
244        let mut inner_borrow = 0u64;
245        for i in 0..8 {
246            let sub_val = if i == 0 { C & mask } else { 0 };
247            let (d, bw) = sbb(r[i], sub_val, inner_borrow);
248            out[i] = d;
249            inner_borrow = bw;
250        }
251        Self(out)
252    }
253
254    #[must_use]
255    pub fn multiply(self, other: Self) -> Self {
256        reduce_wide(wide_mul(self.0, other.0))
257    }
258
259    #[must_use]
260    pub fn square(self) -> Self {
261        self.multiply(self)
262    }

264    /// `bit` must be exactly `0` or `1` - see `fp256.rs::select`'s identical doc comment for why
265    /// this is checked rather than trusted at each call site.
266    #[inline]
267    #[allow(clippy::needless_range_loop)]
268    pub(crate) fn select(bit: u64, a: Self, b: Self) -> Self {
269        debug_assert!(bit <= 1, "select's bit argument must be 0 or 1");
270        let mask = 0u64.wrapping_sub(bit);
271        let mut out = [0u64; 8];
272        for i in 0..8 {
273            out[i] = (a.0[i] & mask) | (b.0[i] & !mask);
274        }
275        Self(out)
276    }

278    /// Constant-time square-and-multiply, fixed 512 iterations MSB-first, `exponent` big-endian.
279    /// SECURITY: routes secret-scalar-derived data once `invert` (via Fermat) and
280    /// `ProjectivePoint::to_affine` compose - same traced dependency `fp256.rs::pow_mod` documents.
281    #[must_use]
282    pub fn pow_mod(self, exponent: &[u8; 64]) -> Self {
283        let mut result = Self::ONE;
284        for &byte in exponent {
285            for bit_idx in (0..8).rev() {
286                result = result.square();
287                let bit = u64::from((byte >> bit_idx) & 1);
288                let candidate = result.multiply(self);
289                result = Self::select(bit, candidate, result);
290            }
291        }
292        result
293    }

295    /// Multiplicative inverse via Fermat's little theorem (`self^(p-2)`). `invert(ZERO) == ZERO`
296    /// (a defined-but-meaningless result, never a panic) - see the boundary test in
297    /// `tests/dstu9041_field_512.rs`.
298    #[must_use]
299    pub fn invert(self) -> Self {
300        self.pow_mod(&P_MINUS_2)
301    }
302
303    /// Euler's criterion: `true` iff `self` is a nonzero quadratic residue mod `p`. Returns
304    /// `false` for `ZERO`.
305    #[must_use]
306    pub fn euler_criterion(self) -> bool {
307        self.pow_mod(&P_MINUS_1_OVER_2) == Self::ONE
308    }
309
310    /// Square root for `p ≡ 5 (mod 8)`. Unconditional - callers must check
311    /// [`Self::euler_criterion`] first if they need to know whether `self` is actually a residue;
312    /// `sqrt(ZERO) == ZERO` by construction (both branches of the formula yield zero).
313    #[must_use]
314    pub fn sqrt(self) -> Self {
315        let f = self.pow_mod(&P_MINUS_1_OVER_4);
316        let z = self.pow_mod(&P_PLUS_3_OVER_8);
317        // p-1: P_LIMBS[0] (ends ..FC95) minus 1 needs no borrow (its low byte is 0x95 > 0).
318        let p_minus_1 = Self([
319            P_LIMBS[0] - 1,
320            P_LIMBS[1],
321            P_LIMBS[2],
322            P_LIMBS[3],
323            P_LIMBS[4],
324            P_LIMBS[5],
325            P_LIMBS[6],
326            P_LIMBS[7],
327        ]);
328        let is_minus_one = u64::from(f == p_minus_1);
329        let w = Self(W);
330        let z_corrected = w.multiply(z);
331        Self::select(is_minus_one, z_corrected, z)
332    }
333}

350/// Rejection-samples `bytes` (big-endian) into a [`FieldElement`], `None` if `bytes >= p`.
351/// Accepts the full `[0, p-1]` range including both `0` and `p-1`.
352#[must_use]
353#[allow(clippy::needless_range_loop)]
354pub fn from_candidate_bytes(bytes: &[u8; 64]) -> Option<FieldElement> {
355    let candidate = FieldElement::from_be_bytes(bytes);
356    let mut borrow = 0u64;
357    for i in 0..8 {
358        let (_, bw) = sbb(candidate.0[i], P_LIMBS[i], borrow);
359        borrow = bw;
360    }
361    if borrow == 1 {
362        Some(candidate)
363    } else {
364        None
365    }
366}

393/// Kani proof harness, mirroring `fp256.rs`'s own `kani_proofs` module (T-177, D-102/D-112) at the
394/// wider 8-limb size. Full `multiply`/`wide_mul` symbolic equivalence is deliberately not
395/// attempted here for the same reason `fp256.rs` doesn't attempt it - D-112's already-established
396/// intractability for this multiplier-equivalence class, and an 8-limb schoolbook multiply is a
397/// harder instance of the same class, not a new question.
398#[cfg(kani)]
399mod kani_proofs {
400    use super::*;
401
402    fn less_than_p(x: [u64; 8]) -> bool {
403        let mut borrow = 0u64;
404        for i in 0..8 {
405            let (_, bw) = sbb(x[i], P_LIMBS[i], borrow);
406            borrow = bw;
407        }
408        borrow == 1
409    }
410
411    #[kani::proof]
412    fn conditional_sub_p_is_always_fully_reduced() {
413        let x: [u64; 8] = kani::any();
414        let r = conditional_sub_p(x);
415        assert!(less_than_p(r));
416    }
417
418    #[kani::proof]
419    fn select_matches_spec() {
420        let bit: u64 = kani::any();
421        kani::assume(bit <= 1);
422        let a: [u64; 8] = kani::any();
423        let b: [u64; 8] = kani::any();
424        let r = FieldElement::select(bit, FieldElement(a), FieldElement(b));
425        if bit == 1 {
426            assert_eq!(r, FieldElement(a));
427        } else {
428            assert_eq!(r, FieldElement(b));
429        }
430    }
431
432    #[kani::proof]
433    fn add_of_reduced_operands_is_fully_reduced() {
434        let a: [u64; 8] = kani::any();
435        let b: [u64; 8] = kani::any();
436        kani::assume(less_than_p(a));
437        kani::assume(less_than_p(b));
438        let r = FieldElement(a).add(FieldElement(b));
439        assert!(less_than_p(r.0));
440    }
441
442    #[kani::proof]
443    fn sub_of_reduced_operands_is_fully_reduced() {
444        let a: [u64; 8] = kani::any();
445        let b: [u64; 8] = kani::any();
446        kani::assume(less_than_p(a));
447        kani::assume(less_than_p(b));
448        let r = FieldElement(a).sub(FieldElement(b));
449        assert!(less_than_p(r.0));
450    }
451
452    #[kani::proof]
453    fn reduce_wide_is_always_fully_reduced() {
454        let wide: [u64; 16] = kani::any();
455        let r = reduce_wide(wide);
456        assert!(less_than_p(r.0));
457    }
