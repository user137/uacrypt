1//! Encrypt/decrypt composition for DSTU 9041's `l(p)=512` case (T-192 Phase 4 - clauses 11/12,
2//! same clause set `encryption.rs` cites for `l(p)=256`). Direct sibling of `encryption.rs` at
3//! this field's own byte widths.
4//!
5//! `DecryptError` is deliberately collapsed to one variant (`InvalidCiphertext`) - same
6//! padding-oracle reasoning as `encryption.rs`'s own doc comment (`docs/SECURITY.md`'s threat
7//! model), not repeated here.
8//!
9//! **Finding 1/2 guards apply here exactly as in `curve256.rs`/`encryption.rs`** - `point_from_x`
10//! (this module's own `curve512::point_from_x`) already rejects `r in {0,1,p-1}` and enforces
11//! subgroup membership (`n*R' == NEUTRAL`), independently re-derived for E512/1 in D-176/D-178,
12//! not assumed to carry over unchecked.
13//!
14//! **`kw_plaintext_from_m_prime`'s "append one all-zero block" convention is confirmed against
15//! Додаток Г.3** (T-192 Phase 4, D-180) - unlike `l(p)=384`'s still-unimplemented `KW-p` padding
16//! case, `l(p)=512` uses plain `Kalyna512_512Kw` with no padding variant needed.
17
18use super::curve512::{base_point, is_valid_scalar, point_from_x, Point};
19use super::fp512::from_candidate_bytes;
20use super::message512::{
21    build_m_prime, encode_l_m_tilde, format_m_tilde, kw_plaintext_from_m_prime, parse_m_prime,
22};
23use crate::hazmat::kalyna_kw::Kalyna512_512Kw;

25/// Hash function identifier for Kupyna-256 (clause 5.7's `i_H` registry - only one value is wired
26/// up by this module, matching the worked example's own `hash_function_id_hex: "01"`).
27const HASH_ID_KUPYNA256: u8 = 0x01;
28
29#[derive(Debug, Clone, Copy, PartialEq, Eq)]
30pub enum EncryptError {
31    /// `message_bits` is `0`, exceeds `L_MAX_P`, or `message`'s length doesn't match it.
32    InvalidMessage,
33    /// `epsilon` is outside the valid scalar range `{2, ..., n-2}`.
34    InvalidEphemeralKey,
35}
36
37#[derive(Debug, Clone, Copy, PartialEq, Eq)]
38pub enum DecryptError {
39    /// Any late-stage failure (bad `r`, KW checksum mismatch, hash mismatch, bad padding) or an
40    /// invalid `e` - deliberately collapsed, see this module's own doc comment.
41    InvalidCiphertext,
42}
43
44/// Clause 11, steps 2-15. `ciphertext_C` is `r || t` (64 + 192 bytes, 256 total for `l(p)=512`).
45///
46/// # Errors
47///
48/// See [`EncryptError`]'s variants.
49pub fn encrypt(
50    message: &[u8],
51    message_bits: usize,
52    q: Point,
53    epsilon: &[u8; 64],
54) -> Result<[u8; 256], EncryptError> {
55    if !is_valid_scalar(epsilon) {
56        return Err(EncryptError::InvalidEphemeralKey);
57    }
58    let m_tilde =
59        format_m_tilde(message, message_bits).map_err(|_| EncryptError::InvalidMessage)?;
60    let l_m_tilde = encode_l_m_tilde(message_bits);
61    let m_prime = build_m_prime(HASH_ID_KUPYNA256, &m_tilde, &l_m_tilde);
62
63    let r_point = base_point().scalar_multiply(epsilon);
64    let r_bytes = r_point.x.to_be_bytes();
65
66    let t_point = q.scalar_multiply(epsilon);
67    let kappa = t_point.x.to_be_bytes();
68
69    let kw_plaintext = kw_plaintext_from_m_prime(&m_prime);
70    let mut t = [0u8; 192];
71    // Unreachable in practice: `kw_plaintext` is always exactly 128 bytes by construction, and `t`
72    // is always exactly 192 bytes - the only failure `wrap` can report (`KwError::InvalidLength`)
73    // requires a length mismatch this function's own fixed-size arrays can't produce. Handled
74    // instead of `.expect()`-ing (this crate denies `unwrap`/`expect` in library code) so this
75    // stays provably panic-free rather than "correct by inspection only" - same reasoning
76    // `encryption.rs::encrypt` already documents for its own `l(p)=256` case.
77    Kalyna512_512Kw::wrap(&kappa, &kw_plaintext, &mut t)
78        .map_err(|_| EncryptError::InvalidMessage)?;
79
80    let mut ciphertext = [0u8; 256];
81    ciphertext[..64].copy_from_slice(&r_bytes);
82    ciphertext[64..].copy_from_slice(&t);
83    Ok(ciphertext)
84}

86/// Clause 12, steps 1-19. Returns the recovered `M~` (53 bytes, left-padded - caller slices the
87/// low-order `l(M)` bits out using the returned bit length) and `l(M)`. Takes no public key - same
88/// reasoning as `encryption.rs::decrypt`'s own doc comment (`T' = e*R'` needs only the private key
89/// and the ciphertext's own `r`).
90///
91/// # Errors
92///
93/// Returns [`DecryptError::InvalidCiphertext`] for any tampered ciphertext, invalid `e`, or
94/// malformed `r` - deliberately not distinguished, see this module's own doc comment.
95pub fn decrypt(ciphertext: &[u8; 256], e: &[u8; 64]) -> Result<([u8; 53], usize), DecryptError> {
96    if !is_valid_scalar(e) {
97        return Err(DecryptError::InvalidCiphertext);
98    }
99
100    let mut r_bytes = [0u8; 64];
101    r_bytes.copy_from_slice(&ciphertext[..64]);
102    let r_field = from_candidate_bytes(&r_bytes).ok_or(DecryptError::InvalidCiphertext)?;
103
104    // Steps 2-6 (reject r in {0,1,p-1}, reject r^2=a*d^-1, compute v and reject non-residues,
105    // y=sqrt(v)) plus the subgroup check beyond clause 12's literal text are all
106    // `curve512::point_from_x` - see that function's own doc comment.
107    let r_prime = point_from_x(r_field).ok_or(DecryptError::InvalidCiphertext)?;
108
109    // Steps 7-8: T' = e*R'; kappa = x_T'.
110    let t_prime = r_prime.scalar_multiply(e);
111    let kappa = t_prime.x.to_be_bytes();
112
113    // Step 9: unwrap. kalyna_kw's own checksum (D-55) already constant-time-compares its trailing
114    // block.
115    let mut recovered = [0u8; 128];
116    Kalyna512_512Kw::unwrap(&kappa, &ciphertext[64..], &mut recovered)
117        .map_err(|_| DecryptError::InvalidCiphertext)?;
118
119    // The empirical "M' || 0x00*64" quirk, mirrored on decrypt: verify the appended block came
120    // back all-zero, in fixed-iteration constant time.
121    let mut appended_block_bad = 0u8;
122    for &byte in &recovered[64..] {
123        appended_block_bad |= u8::from(byte != 0);
124    }
125    if appended_block_bad != 0 {
126        return Err(DecryptError::InvalidCiphertext);
127    }
128
129    let mut m_prime = [0u8; 64];
130    m_prime.copy_from_slice(&recovered[..64]);
131
132    // Steps 10-17: parsed internally (hash + zero-padding checks, both constant-time - see
133    // message512.rs's own doc comment on why).
134    let parsed = parse_m_prime(&m_prime).map_err(|_| DecryptError::InvalidCiphertext)?;
135
136    Ok((parsed.m_tilde, parsed.bit_length))
137}
