1//! Twisted Edwards point arithmetic over E256/1 (clauses 3.18/7.2/Додаток Б.4 - see
2//! `docs/pseudocode/dstu9041.md`, `docs/DECISIONS.md` D-163/D-165/D-166). Curve equation
3//! (x/y-swapped relative to the textbook Bernstein-Lange form, per the primary text):
4//! `x^2 + a*y^2 = d*x^2*y^2 + 1 (mod p)`, `a=2`, `d=0x18`.
5//!
6//! `Point{x,y}` has no `Infinity` variant - not because Edwards curves lack one (they don't,
7//! clause 3.18 counts `D_{1,2}=(+-sqrt(a/d),inf)` in the curve's order), but because every point
8//! this code ever produces or accepts is unreachable from those two singular points: the random-
9//! point generator (clause 6.9, not implemented here) retries around them, and ciphertext
10//! reconstruction excludes them via the `r^2=a*d^-1` check (plus the `r=p-1` check this module's
11//! sibling `encryption.rs` adds - a genuine additional bad value found by advisor review, not
12//! covered by the standard's literal clause 12 text; see `r_equals_p_minus_1_reconstructs_the_order_2_point`
13//! in `tests/dstu9041_curve.rs` for the arithmetic proof).
14//!
15//! Scalar multiplication uses the complete Edwards addition law (Додаток Б.4) for both doubling
16//! and adding - `d` non-square (guaranteed by 3.18/7.2 for every recommended curve) means this
17//! formula has no exceptional/branching cases, so `scalar_multiply` is a fixed 256-iteration
18//! double-and-select loop with no early exit and no add/double distinction.
19

50#[derive(Clone, Copy, Debug, PartialEq, Eq)]
51pub struct Point {
52    pub x: FieldElement,
53    pub y: FieldElement,
54}
55
56impl Point {
57    /// The neutral element `(1, 0)`.
58    pub const NEUTRAL: Self = Self {
59        x: FieldElement::ONE,
60        y: FieldElement::ZERO,
61    };
62
63    /// Checks `x^2 + a*y^2 == d*x^2*y^2 + 1 (mod p)`. Public-data check (coordinates are never
64    /// secret by the time this is called) - an ordinary branch is fine here, unlike
65    /// `scalar_multiply`.
66    #[must_use]
67    pub fn is_on_curve(self) -> bool {
68        let x2 = self.x.square();
69        let y2 = self.y.square();
70        let lhs = x2.add(curve_a().multiply(y2));
71        let rhs = curve_d().multiply(x2).multiply(y2).add(FieldElement::ONE);
72        lhs == rhs
73    }
74
75    /// Point addition (also correct for doubling and the neutral element - the formula is
76    /// complete). A convenience wrapper around `ProjectivePoint::add`; `scalar_multiply` uses the
77    /// projective form directly to avoid a per-iteration inversion.
78    #[must_use]
79    #[allow(clippy::should_implement_trait)] // deliberate: value-returning, not the mutating
80                                             // `std::ops::Add` shape - matches fp256's own style
81    pub fn add(self, other: Self) -> Self {
82        ProjectivePoint::from_affine(self)
83            .add(ProjectivePoint::from_affine(other))
84            .to_affine()
85    }
86
87    /// Constant-time scalar multiplication (clause 6.12), fixed 256 iterations MSB-first,
88    /// `scalar` big-endian. The accumulator starts at `NEUTRAL` and every iteration always
89    /// doubles and always computes the candidate add before selecting - no early exit, no
90    /// branch on the scalar's bits.
91    #[must_use]
92    pub fn scalar_multiply(self, scalar: &[u8; 32]) -> Self {
93        let base = ProjectivePoint::from_affine(self);
94        let mut acc = ProjectivePoint::from_affine(Self::NEUTRAL);
95        for &byte in scalar {
96            for bit_idx in (0..8).rev() {
97                acc = acc.add(acc);
98                let bit = u64::from((byte >> bit_idx) & 1);
99                let candidate = acc.add(base);
100                acc = ProjectivePoint::select(bit, candidate, acc);
101            }
102        }
103        acc.to_affine()
104    }
105}

107#[derive(Clone, Copy, Debug)]
108struct ProjectivePoint {
109    x: FieldElement,
110    y: FieldElement,
111    z: FieldElement,
112}
113
114impl ProjectivePoint {
115    fn from_affine(p: Point) -> Self {
116        Self {
117            x: p.x,
118            y: p.y,
119            z: FieldElement::ONE,
120        }
121    }
122
123    /// The one inversion point in the whole scalar-multiplication ladder - `self.z` is a function
124    /// of the secret scalar, so this composes into `FieldElement::invert`'s traced constant-time
125    /// dependency (see `fp256.rs`'s `pow_mod` doc comment).
126    fn to_affine(self) -> Point {
127        let z_inv = self.z.invert();
128        Point {
129            x: self.x.multiply(z_inv),
130            y: self.y.multiply(z_inv),
131        }
132    }
133
134    /// Додаток Б.4's complete addition law: `A=Z1*Z2, B=A^2, C=X1*X2, D=Y1*Y2, E=d*C*D,
135    /// F=B-E, G=B+E`, `X_R=A*G*(C-a*D)`, `Y_R=A*F*((X1+Y1)*(X2+Y2)-C-D)`, `Z_R=F*G`. Used for both
136    /// doubling (`self==other`) and the neutral element uniformly - no exceptional cases, because
137    /// `d` is a non-square (guaranteed by clause 3.18/7.2 for every recommended curve).
138    #[allow(clippy::many_single_char_names)] // a/d/b/c/e/f/g mirror Додаток Б.4's own notation
139    fn add(self, other: Self) -> Self {
140        let a = curve_a();
141        let d = curve_d();
142
143        let zz = self.z.multiply(other.z);
144        let b = zz.square();
145        let c = self.x.multiply(other.x);
146        let dd = self.y.multiply(other.y);
147        let e = d.multiply(c).multiply(dd);
148        let f = b.sub(e);
149        let g = b.add(e);
150
151        let x_sum = self.x.add(self.y);
152        let y_sum = other.x.add(other.y);
153        let cross = x_sum.multiply(y_sum);
154
155        let x_r = zz.multiply(g).multiply(c.sub(a.multiply(dd)));
156        let y_r = zz.multiply(f).multiply(cross.sub(c).sub(dd));
157        let z_r = f.multiply(g);
158
159        Self {
160            x: x_r,
161            y: y_r,
162            z: z_r,
163        }
164    }
165
166    fn select(bit: u64, a: Self, b: Self) -> Self {
167        Self {
168            x: FieldElement::select(bit, a.x, b.x),
169            y: FieldElement::select(bit, a.y, b.y),
170            z: FieldElement::select(bit, a.z, b.z),
171        }
172    }
173}

175/// Reconstructs a point on the curve from just its `x`-coordinate, choosing whichever `sqrt`
176/// branch [`FieldElement::sqrt`] returns for `y` and rejecting anything outside the base point's
177/// own prime-order subgroup `<P>`. Shared by `encryption::decrypt` (reconstructing `R'` from a
178/// ciphertext's `r`) and `crate::crypto_box::PublicKey::from_bytes` (reconstructing a compressed
179/// public key) - one security-critical gauntlet, not two independently-maintained copies of it.
180///
181/// Rejects (clause 12 step 2, plus this crate's own `r=p-1`/subgroup findings - see
182/// `encryption.rs`'s module doc): `x in {0, 1, p-1}`, `x^2 = a*d^-1 mod p` (`D_{1,2}`'s exclusion),
183/// `v = (1-x^2)*(a-d*x^2)^-1` not a quadratic residue, or the reconstructed candidate outside
184/// `<P>`.
185///
186/// Which `sqrt` branch is chosen does not matter for `crypto_box`'s own use of this: this curve's
187/// negation is `-(x,y) = (x,-y)` (the swapped-Edwards form, `docs/pseudocode/dstu9041.md`), so `x`
188/// never distinguishes a point from its negation, and `x_T = x_{-T}` for any point `T` - meaning
189/// `x_{k*Q} = x_{k*(-Q)}` for any scalar `k`, since `k*(-Q) = -(k*Q)`. Reconstructing a public key
190/// from only its `x`-coordinate therefore yields the same `kappa` on the encrypt side regardless
191/// of which of `{Q, -Q}` this function happens to return.
192#[must_use]
193pub fn point_from_x(x: FieldElement) -> Option<Point> {
194    let a = curve_a();
195    let d = curve_d();
196    let p_minus_1 = FieldElement::ZERO.sub(FieldElement::ONE);
197    let x_squared = x.square();
198
199    if x == FieldElement::ZERO
200        || x == FieldElement::ONE
201        || x == p_minus_1
202        || x_squared == a.multiply(d.invert())
203    {
204        return None;
205    }
206
207    let numerator = FieldElement::ONE.sub(x_squared);
208    let denominator = a.sub(d.multiply(x_squared));
209    let v = numerator.multiply(denominator.invert());
210
211    if !v.euler_criterion() {
212        return None;
213    }
214
215    let candidate = Point { x, y: v.sqrt() };
216
217    if candidate.scalar_multiply(&order()) != Point::NEUTRAL {
218        return None;
219    }
220
221    Some(candidate)
222}

237#[allow(clippy::needless_range_loop)]
238fn bytes_be_to_limbs(bytes: &[u8; 32]) -> [u64; 4] {
239    let mut limbs = [0u64; 4];
240    let mut limb_bytes = [0u8; 8];
241    for i in 0..4 {
242        limb_bytes.copy_from_slice(&bytes[i * 8..i * 8 + 8]);
243        limbs[3 - i] = u64::from_be_bytes(limb_bytes);
244    }
245    limbs
246}

248/// `true` iff `a < b`, both interpreted as big-endian unsigned 256-bit integers. Branchless
249/// (fixed 4-limb `sbb` chain), no data-dependent control flow - callers may pass secret data
250/// (e.g. a private-key-shaped scalar in `is_valid_scalar`).
251#[allow(clippy::needless_range_loop)]
252fn is_less_than(a: &[u8; 32], b: &[u8; 32]) -> bool {
253    let al = bytes_be_to_limbs(a);
254    let bl = bytes_be_to_limbs(b);
255    let mut borrow = 0u64;
256    for i in 0..4 {
257        let (_, bw) = sbb(al[i], bl[i], borrow);
258        borrow = bw;
259    }
260    borrow == 1
261}

263/// `1 < scalar < n-1` (clause 6.12's own precondition on a usable exponent - the strict range
264/// `{2, ..., n-2}`), constant-time.
265#[must_use]
266pub fn is_valid_scalar(scalar: &[u8; 32]) -> bool {
267    let mut one = [0u8; 32];
268    one[31] = 1;
269    let mut n_minus_1 = ORDER_N;
270    // n is odd (prime) - decrementing its last byte alone never borrows across bytes.
271    n_minus_1[31] -= 1;
272
273    let scalar_gt_one = is_less_than(&one, scalar);
274    let scalar_lt_n_minus_1 = is_less_than(scalar, &n_minus_1);
275    scalar_gt_one & scalar_lt_n_minus_1
276}
