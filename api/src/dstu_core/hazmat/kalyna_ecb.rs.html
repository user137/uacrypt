1//! Kalyna-ECB: DSTU 7624:2014 mode of operation #1 ("Проста заміна" / simple substitution) -
2//! each block of `buf` encrypted/decrypted independently under the same key, no chaining state at
3//! all. Cited to `oracles/uapki/library/uapkic/src/dstu7624.c`'s `encrypt_ecb`/`decrypt_ecb`
4//! (lines 2899-2961) and `dstu7624_init_ecb` (lines 3920-3934) - a trivial per-block loop calling
5//! the same block-cipher transform `hazmat::kalyna` already implements and dual-oracle-verifies
6//! (D-13). `docs/DECISIONS.md` D-53 has the full citation and verification-approach note.
7//!
8//! # No integrity, and the weakest confidentiality guarantee of any mode in this crate
9//!
10//! **Do not use this for new designs without a specific, understood reason.** ECB leaks any
11//! repeated-plaintext-block structure directly into the ciphertext (identical plaintext blocks
12//! under the same key always produce identical ciphertext blocks) - this is the textbook example
13//! cited across virtually every "how not to do encryption" guide (the ECB-encrypted-bitmap
14//! demonstration). It provides no authentication either. Prefer
15//! [`crate::crypto_secretbox`] unless you specifically need raw, unauthenticated, per-block-
16//! independent encryption and understand why ECB's pattern leakage is acceptable for your use case
17//! (e.g. re-encrypting already-random, block-sized, independent key material one block at a time).
18
19/// `buf`'s length is not a multiple of the block size.
20#[derive(Debug, Clone, Copy, PartialEq, Eq)]
21pub struct InvalidLength;
22
23macro_rules! kalyna_ecb_variant {
24    ($name:ident, $expanded:ident, $key_bytes:literal, $block_bytes:literal) => {
25        #[doc = concat!(
26            "ECB mode over [`super::kalyna::", stringify!($expanded), "`] - see the module doc ",
27            "comment for the citation and the misuse warning."
28        )]
29        pub struct $name {
30            key: super::kalyna::$expanded,
31        }
32
33        impl $name {
34            #[must_use]
35            pub fn new(key: &[u8; $key_bytes]) -> Self {
36                Self {
37                    key: super::kalyna::$expanded::new(key),
38                }
39            }
40
41            /// Encrypts `buf` in place, one block at a time, independently.
42            ///
43            /// # Errors
44            ///
45            /// Returns [`InvalidLength`] if `buf.len()` is not a multiple of the block size -
46            /// this mode has no padding of its own (`encrypt_ecb`'s own `in->len % block_len`
47            /// check, transcribed as found).
48            pub fn encrypt_in_place(&self, buf: &mut [u8]) -> Result<(), InvalidLength> {
49                if !buf.len().is_multiple_of($block_bytes) {
50                    return Err(InvalidLength);
51                }
52                for block in buf.chunks_exact_mut($block_bytes) {
53                    let mut input = [0u8; $block_bytes];
54                    input.copy_from_slice(block);
55                    block.copy_from_slice(&self.key.encrypt_block(&input));
56                }
57                Ok(())
58            }
59
60            /// Decrypts `buf` in place, one block at a time, independently.
61            ///
62            /// # Errors
63            ///
64            /// Returns [`InvalidLength`] under the same condition as
65            /// [`Self::encrypt_in_place`].
66            pub fn decrypt_in_place(&self, buf: &mut [u8]) -> Result<(), InvalidLength> {
67                if !buf.len().is_multiple_of($block_bytes) {
68                    return Err(InvalidLength);
69                }
70                for block in buf.chunks_exact_mut($block_bytes) {
71                    let mut input = [0u8; $block_bytes];
72                    input.copy_from_slice(block);
73                    block.copy_from_slice(&self.key.decrypt_block(&input));
74                }
75                Ok(())
76            }
77        }
78    };
79}
