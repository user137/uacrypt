1//! DSTU 4145-2002 sign/verify for `m=257` - same algorithm and structure as `signature::{sign,
2//! verify}` for `m=163` (see that module's own doc comment for the full citation and the
3//! `Q = -d*G` convention, which applies identically here), built on `gf2m257`/`curve257`/
4//! `scalar257` instead. `docs/TASKS.md` T-199, `docs/DECISIONS.md` D-185/D-186.
5//!
6//! **`verify`'s public-key validation uses the general (`SP 800-56A`-style) check, not
7//! `signature::verify`'s `m=163`-specific `x == 0` rejection** - `curve257`'s cofactor is `4`, not
8//! `2` (D-185's Bouncy Castle cross-check, `h_s[6] = FOUR`), so `m=163`'s fix (reject the curve's
9//! one order-2 point directly by its known `x = 0` coordinate) does not carry over: a cofactor-4
10//! curve's small subgroup also contains order-4 points, whose `x`-coordinates are not `0` and
11//! would slip past that specific check. Instead: reject `q` unless `q.scalar_multiply(&order())
12//! == Point::Infinity` - `q` lies in the prime-order subgroup if and only if this holds, true for
13//! *any* cofactor without needing to enumerate the small subgroup's actual points (`advisor()`
14//! review, `docs/TASKS.md` T-199) - one extra full scalar multiplication per `verify` call, public
15//! data throughout, same cost class as the two multiplications `verify` already pays. Tested
16//! against the curve's genuine order-2 point (`x = 0`, same construction `signature.rs`'s own
17//! `t189_public_key_validation` test uses) in `tests/dstu4145_signature257.rs` - sufficient to
18//! prove the general check actually rejects a real small-subgroup point, not just the `m=163`-style
19//! one this module deliberately does *not* special-case.
20//!
21//! `verify` is public-data-only throughout, same posture as `signature::verify` - ordinary
22//! branches and `==` are fine here. No `verify_combine`-style projective fast path exists for
23//! `curve257` yet (`curve257`'s own module doc) - `verify` pays three full `scalar_multiply` calls
24//! now (the subgroup check, plus the original two).
25

30/// Same construction as `signature::hash_to_field`, widened to `m=257`/33 bytes: keep the hash's
31/// last `min(len, 33)` bytes, masking the top byte to its low 1 bit if a full 33 were taken (257
32/// bits total - `33*8 - 257 = 7` padding bits, so only bit 0 of the top byte is meaningful).
33#[must_use]
34pub fn hash_to_field(hash: &[u8]) -> FieldElement {
35    let take = hash.len().min(33);
36    let mut bytes = [0u8; 33];
37    bytes[33 - take..].copy_from_slice(&hash[hash.len() - take..]);
38    if take == 33 {
39        bytes[0] &= 0x01; // low 257 bits: the top byte holds only bit 256
40    }
41    FieldElement::from_be_bytes(&bytes)
42}

44/// `truncate(y, n.bit_length() - 1)`: keeps the low `n.bit_length() - 1` bits of a field element's
45/// integer value. **`n.bit_length()` is 256 here, not 257** - `curve257::order()`'s own top byte
46/// is `0x80` (no leading-zero slack, `docs/DECISIONS.md` D-185), so `n.bit_length() == m` does
47/// *not* hold the way it happens to for `m=163` (whose `n` has bit-length 163, coincidentally
48/// matching `m`) - `signature::truncate_162`'s `m - 1 == n.bit_length() - 1` equivalence was a
49/// coincidence of that curve's specific order, not a general rule, and does not carry over here.
50/// This keeps the low **255** bits: byte 0 (global bits 256-263) cleared entirely, byte 1's top
51/// bit (global bit 255) also cleared (`0x7F` mask) - found and fixed via the BC-generated
52/// `signature_cases` oracle failing `verify` despite `sign` matching exactly (`docs/TASKS.md`
53/// T-199): `sign`'s own `r` output was correct either way (`Scalar::from_be_bytes` doesn't reduce,
54/// so an over-wide `r` still round-tripped through `sign`'s own output), but was never actually
55/// `< n` as `verify`'s domain check requires - a real correctness bug, not just a naming slip.
56///
57/// **This makes `r < n` (and `s < n`, since `s` is a `scalar257::Scalar` and therefore always
58/// reduced) provable, not just empirically likely**: this function's output is always `< 2^255`
59/// (255 bits kept), and `n >= 2^255` unconditionally (`n.bit_length() == 256` means `n` occupies
60/// `[2^255, 2^256)`) - so `r < 2^255 <= n` holds for *every* input `y`, not just the ones a random
61/// BC-oracle sample happened to cover. See `truncate_255_output_is_always_below_n` in
62/// `tests/dstu4145_signature257.rs` for the boundary case that pins this down directly (`y` with
63/// every bit set - the function's own maximum possible output).
64fn truncate_255(y: FieldElement) -> [u8; 33] {
65    let mut bytes = y.to_be_bytes();
66    bytes[0] = 0;
67    bytes[1] &= 0x7F;
68    bytes
69}

71fn is_zero(bytes: &[u8; 33]) -> bool {
72    bytes.iter().all(|&b| b == 0)
73}

75/// Big-endian byte-array comparison - public magnitude check (`r < n`, `s < n`).
76fn less_than(a: &[u8; 33], b: &[u8; 33]) -> bool {
77    a < b
78}

80/// `verifySignature`, `m=257`. `hash` is an already-computed message digest.
81#[must_use]
82#[allow(clippy::many_single_char_names)] // r, s, q, g, h, n match the pseudocode doc's own names
83pub fn verify(hash: &[u8], r: &[u8; 33], s: &[u8; 33], q: Point, g: Point) -> bool {
84    let n = curve257::order();
85    if is_zero(r) || is_zero(s) || !less_than(r, &n) || !less_than(s, &n) {
86        return false;
87    }
88
89    // Full public-key validation (on-curve, non-identity, in the prime-order subgroup) - see the
90    // module doc for why the general `n*q == Infinity` check is used here instead of `m=163`'s
91    // `x == 0` shortcut (cofactor 4, not 2 - that shortcut doesn't cover this curve's order-4
92    // points).
93    if !q.is_on_curve() || q.scalar_multiply(&n) != Point::Infinity {
94        return false;
95    }
96
97    let mut h = hash_to_field(hash);
98    if h == FieldElement::ZERO {
99        h = FieldElement::ONE;
100    }
101
102    let s_point = g.scalar_multiply(s);
103    let r_point = q.scalar_multiply(r);
104    let big_r = s_point + r_point;
105    let (rx, _) = match big_r {
106        Point::Affine(x, y) => (x, y),
107        Point::Infinity => return false,
108    };
109
110    let y = h.multiply(rx);
111    &truncate_255(y) == r
112}

114/// `generateSignature`, `m=257`. `e` is the caller-supplied ephemeral scalar - see
115/// `signature::sign`'s own doc comment for the freshness/secrecy requirement, identical here.
116#[must_use]
117#[allow(clippy::many_single_char_names)] // r, s, d, e, h match the pseudocode doc's own names
118pub fn sign(hash: &[u8], d: Scalar, e: Scalar, g: Point) -> Option<([u8; 33], [u8; 33])> {
119    let mut h = hash_to_field(hash);
120    if h == FieldElement::ZERO {
121        h = FieldElement::ONE;
122    }
123
124    let (fe_x, _) = match g.scalar_multiply(&e.to_be_bytes()) {
125        Point::Affine(x, y) => (x, y),
126        Point::Infinity => return None,
127    };
128    if fe_x == FieldElement::ZERO {
129        return None;
130    }
131
132    let y = h.multiply(fe_x);
133    let r_bytes = truncate_255(y);
134    if is_zero(&r_bytes) {
135        return None;
136    }
137
138    let r = Scalar::from_be_bytes(&r_bytes);
139    let s = r.multiply(d) + e;
140    if s.is_zero() {
141        return None;
142    }
143
144    Some((r_bytes, s.to_be_bytes()))
145}

147#[cfg(test)]
148mod tests {
149    use super::*;
150
151    /// Proves `truncate_255`'s output is always `< n`, not just empirically likely (see its own
152    /// doc comment for the bit-count argument) - the boundary case, `y` with every representable
153    /// bit set (the function's own maximum possible output), is the one input that actually
154    /// exercises that bound; a random `y` from the BC oracle only demonstrates it by chance.
155    #[test]
156    fn truncate_255_output_is_always_below_n() {
157        let all_ones = FieldElement::from_be_bytes(&[0xFFu8; 33]);
158        let r_bytes = truncate_255(all_ones);
159        assert!(
160            less_than(&r_bytes, &curve257::order()),
161            "truncate_255's own maximum output must stay below n"
162        );
163    }
