1//! DSTU 4145-2002's m=257 curve: `y^2 + xy = x^3 + a*x^2 + b` over GF(2^257), `a = 0` (unlike
2//! `curve163`'s `a = 1`) - domain parameters extracted from two independent real DSTU 4145
3//! certificates, byte-order-corrected and cross-checked against Bouncy Castle's
4//! `DSTU4145NamedCurves.java` `curves[6]` (`ECCurve.F2m(257, 12, ZERO, ...)`, cofactor `h = 4`,
5//! not `curve163`'s `h = 2`) - see `docs/DECISIONS.md` D-185/D-186 for the full provenance.
6//!
7//! `a = 0` removes the `+a`/`+a*x^2` terms `curve163`'s `double`/`add`/`is_on_curve` carry for
8//! `a = 1` - the Montgomery-ladder core of `scalar_multiply` and the y-coordinate recovery formula
9//! are both `a`-independent (a known property of the López-Dahab projective ladder and its
10//! standard recovery formula - `Guide to Elliptic Curve Cryptography`, Algorithm 3.40/3.41), so
11//! only those three functions actually differ in shape from `curve163`'s.
12//!
13//! Same public/secret-data split as `curve163` (see its own module doc): `double`/`add`/
14//! `is_on_curve` below branch on public data only, never on a secret scalar's intermediate state;
15//! `scalar_multiply` is the sole constant-time path, used for both secret (signing) and public
16//! (verification) scalars here - `curve163`'s `verify_combine` projective fast path (D-108) is not
17//! reproduced yet, T-199's own scope is `add`/`double`/`scalar_multiply`/`negate` only.

59/// An affine point on the curve, or the point at infinity (the group identity).
60#[derive(Clone, Copy, Debug, PartialEq, Eq)]
61pub enum Point {
62    Infinity,
63    Affine(FieldElement, FieldElement),
64}
65
66impl Point {
67    #[must_use]
68    pub fn generator() -> Self {
69        Point::Affine(gx(), gy())
70    }
71
72    /// `-P`: `(x, y) -> (x, x + y)` (char-2 identity, `a`-independent - same as `curve163::negate`).
73    #[must_use]
74    pub fn negate(self) -> Self {
75        match self {
76            Point::Infinity => Point::Infinity,
77            Point::Affine(x, y) => Point::Affine(x, x + y),
78        }
79    }
80
81    /// Checks `y^2 + xy == x^3 + b` (`a = 0` for this curve - no `x^2` term, unlike `curve163`'s
82    /// `a = 1`). `Infinity` is not a solution and returns `false` - same convention as
83    /// `curve163::is_on_curve`. Public-data check, never called on a secret-dependent point.
84    #[must_use]
85    pub fn is_on_curve(self) -> bool {
86        match self {
87            Point::Infinity => false,
88            Point::Affine(x, y) => {
89                let lhs = y.square() + x.multiply(y);
90                let rhs = x.multiply(x.square()) + b();
91                lhs == rhs
92            }
93        }
94    }
95
96    /// Affine point doubling for `y^2 + xy = x^3 + b` (`a = 0`): `x3 = lambda^2 + lambda` (no `+a`
97    /// term, unlike `curve163::double`'s `+ FieldElement::ONE`). Same formula family (`Guide to
98    /// Elliptic Curve Cryptography` §3.1.2) and the same public-data-only caveat as `curve163`.
99    #[must_use]
100    pub fn double(self) -> Self {
101        match self {
102            Point::Infinity => Point::Infinity,
103            Point::Affine(x1, y1) => {
104                if x1 == FieldElement::ZERO {
105                    return Point::Infinity;
106                }
107                let lambda = x1 + y1.multiply(x1.invert());
108                let x3 = lambda.square() + lambda;
109                let y3 = x1.square() + (lambda + FieldElement::ONE).multiply(x3);
110                Point::Affine(x3, y3)
111            }
112        }
113    }
114
115    /// Constant-time scalar multiplication: `k * self`, `k` a big-endian 257-bit scalar (top 7
116    /// bits of `k[0]` must be zero - values are always < the curve order, which is < `2^257`).
117    /// Same Montgomery-ladder shape as `curve163::scalar_multiply` (`Guide to Elliptic Curve
118    /// Cryptography`, Algorithm 3.40) - the ladder's per-iteration formulas and the final
119    /// y-recovery step are both `a`-independent (see module doc), so only the loop bound (257, not
120    /// 163) and field/byte widths differ from `curve163`. See `curve163::scalar_multiply`'s own
121    /// doc comment for the full derivation and the `z1`/`z2`-zero boundary-case handling this
122    /// mirrors exactly (`docs/DECISIONS.md` D-110/T-152) - not re-derived here, this curve's own
123    /// cofactor (`h = 4`, vs. `curve163`'s `h = 2`) has not yet been re-checked against those
124    /// boundary cases specifically, flagged for T-199's own active-attack test pass (step 6).
125    #[must_use]
126    pub fn scalar_multiply(self, k: &[u8; 33]) -> Self {
127        match self {
128            Point::Infinity => Point::Infinity,
129            Point::Affine(x, y) => {
130                let mut x1 = FieldElement::ONE;
131                let mut z1 = FieldElement::ZERO; // (x1 : z1) = Infinity
132                let mut x2 = x;
133                let mut z2 = FieldElement::ONE; // (x2 : z2) = P
134
135                for i in (0..257u32).rev() {
136                    let bit = bit_at(k, i);
137                    let swap = bit ^ 1;
138                    cswap(swap, &mut x1, &mut x2);
139                    cswap(swap, &mut z1, &mut z2);
140
141                    let t1 = z1;
142                    z1 = (x1.multiply(z2) + x2.multiply(z1)).square();
143                    x1 = x.multiply(z1) + x1.multiply(x2).multiply(t1).multiply(z2);
144                    let t2 = x2;
145                    x2 = x2.square().square() + b().multiply(z2.square().square());
146                    z2 = t2.square().multiply(z2.square());
147
148                    cswap(swap, &mut x1, &mut x2);
149                    cswap(swap, &mut z1, &mut z2);
150                }
151
152                // See `curve163::scalar_multiply`'s own comment for the full rationale - identical
153                // boundary-case handling, only the field width differs.
154                if is_zero_mask(z1) != 0 {
155                    return Point::Infinity;
156                }
157
158                let x1_affine = x1.multiply(z1.invert());
159                let x2_affine = x2.multiply(z2.invert());
160                let t1 = x1_affine + x;
161                let t2 = x2_affine + x;
162                let inner = t1.multiply(t2) + x.square() + y;
163                let y1_affine_formula = x.invert().multiply(t1).multiply(inner) + y;
164
165                let y1_affine = select(is_zero_mask(z2), x + y, y1_affine_formula);
166
167                Point::Affine(x1_affine, y1_affine)
168            }
169        }
170    }
171}

173impl core::ops::Add for Point {
174    type Output = Self;
175
176    /// Affine point addition, `a = 0`: `x3 = lambda^2 + lambda + x1 + x2` (no `+a` term, unlike
177    /// `curve163::add`'s `+ FieldElement::ONE`). Same public-data-only caveat as `double`.
178    fn add(self, other: Self) -> Self {
179        match (self, other) {
180            (Point::Infinity, q) => q,
181            (p, Point::Infinity) => p,
182            (Point::Affine(x1, y1), Point::Affine(x2, y2)) => {
183                if x1 == x2 {
184                    if y1 == y2 {
185                        return self.double();
186                    }
187                    return Point::Infinity;
188                }
189                let lambda = (y1 + y2).multiply((x1 + x2).invert());
190                let x3 = lambda.square() + lambda + x1 + x2;
191                let y3 = lambda.multiply(x1 + x3) + x3 + y1;
192                Point::Affine(x3, y3)
193            }
194        }
195    }
196}

198fn bit_at(bytes: &[u8; 33], i: u32) -> u64 {
199    let byte_index = 32 - (i / 8) as usize;
200    let bit_in_byte = i % 8;
201    u64::from((bytes[byte_index] >> bit_in_byte) & 1)
202}
203
204/// Constant-time conditional swap - same shape as `curve163::cswap`, widened to 5 limbs.
205fn cswap(swap: u64, a: &mut FieldElement, b: &mut FieldElement) {
206    let mask = 0u64.wrapping_sub(swap);
207    for i in 0..5 {
208        let t = mask & (a.0[i] ^ b.0[i]);
209        a.0[i] ^= t;
210        b.0[i] ^= t;
211    }
212}

214/// Constant-time "is this field element zero" test - same shape as `curve163::is_zero_mask`,
215/// widened to 5 limbs.
216fn is_zero_mask(a: FieldElement) -> u64 {
217    let combined = a.0[0] | a.0[1] | a.0[2] | a.0[3] | a.0[4];
218    let is_nonzero = (combined | combined.wrapping_neg()) >> 63;
219    0u64.wrapping_sub(1 ^ is_nonzero)
220}

222/// Constant-time select - same shape as `curve163::select`, widened to 5 limbs.
223fn select(mask: u64, if_mask: FieldElement, otherwise: FieldElement) -> FieldElement {
224    let mut out = [0u64; 5];
225    for ((out_limb, a), b) in out.iter_mut().zip(if_mask.0.iter()).zip(otherwise.0.iter()) {
226        *out_limb = b ^ (mask & (a ^ b));
227    }
228    FieldElement(out)
