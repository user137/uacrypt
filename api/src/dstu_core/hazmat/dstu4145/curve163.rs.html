1//! DSTU 4145-2002's m=163 curve: `y^2 + xy = x^3 + a*x^2 + b` over GF(2^163), `a = 1` - the curve
2//! dual-sourced in `tests/vectors/dstu4145/gf2m163.json` (`docs/DECISIONS.md` D-14), matching
3//! `oracles/bouncycastle-java/.../DSTU4145NamedCurves.java`'s `ECCurve.F2m(163, 3, 6, 7, ...)`.
4//!
5//! `double`/`add` below are plain affine formulas with ordinary branches (`==`) on the point
6//! coordinates. That's only safe on **public** data - both are meant for the verification path
7//! (`s*G + r*Q`, entirely public inputs), never for a secret scalar's intermediate state.
8//!
9//! `scalar_multiply` is the sole implementation for **secret**-scalar use (signing's ephemeral
10//! `e`, and `Q = -d*G`'s private-key-derived `d`) - identical in every build configuration, never
11//! replaced. Under the `small-tables` feature, it is *also* still used for `verify`'s public-scalar
12//! computation, exactly as before. By default (feature off), `verify` instead uses
13//! `verify_combine`'s projective (López-Dahab)/Shamir's-trick fast path below, which is **not**
14//! constant-time and must never be called with a secret scalar (`docs/DECISIONS.md` D-108) - the
15//! module's public/secret split is now carried by *which function is called*, not by one shared
16//! code path the way D-25 originally described it.

52/// An affine point on the curve, or the point at infinity (the group identity).
53#[derive(Clone, Copy, Debug, PartialEq, Eq)]
54pub enum Point {
55    Infinity,
56    Affine(FieldElement, FieldElement),
57}
58
59impl Point {
60    #[must_use]
61    pub fn generator() -> Self {
62        Point::Affine(gx(), gy())
63    }

65    /// `-P`: for this curve family, negation is `(x, y) -> (x, x + y)` (char-2 identity - see
66    /// `double`/`add`'s comment on the same fact). Used to derive a DSTU 4145 public key
67    /// `Q = -d*G` from a private key `d` (`docs/DECISIONS.md` D-25's follow-up note on this - Bouncy
68    /// Castle's `DSTU4145KeyPairGenerator` negates explicitly; the pseudocode doc's "`Q = d*G`"
69    /// line undersold this).
70    #[must_use]
71    pub fn negate(self) -> Self {
72        match self {
73            Point::Infinity => Point::Infinity,
74            Point::Affine(x, y) => Point::Affine(x, x + y),
75        }
76    }

78    /// Checks `y^2 + xy == x^3 + x^2 + b` (`a = 1` fixed for this curve, matching `double`'s own
79    /// assumption above). `Infinity` is not a solution of this affine equation and returns
80    /// `false` - callers that must also accept the group identity check for it separately (see
81    /// `signature::verify`, `docs/TASKS.md` T-189). Public-data check (never called on a
82    /// secret-dependent point, same posture as `double`/`add`).
83    #[must_use]
84    pub fn is_on_curve(self) -> bool {
85        match self {
86            Point::Infinity => false,
87            Point::Affine(x, y) => {
88                let lhs = y.square() + x.multiply(y);
89                let rhs = x.multiply(x.square()) + x.square() + b();
90                lhs == rhs
91            }
92        }
93    }

95    /// Affine point doubling for `y^2 + xy = x^3 + x^2 + b` (`a = 1` fixed for this curve).
96    /// Standard formulas (`Guide to Elliptic Curve Cryptography`, Hankerson/Menezes/Vanstone,
97    /// `§3.1.2`) - branches on `x1 == 0` are fine here, see the module doc: this is never called
98    /// on a secret-dependent point.
99    #[must_use]
100    pub fn double(self) -> Self {
101        match self {
102            Point::Infinity => Point::Infinity,
103            Point::Affine(x1, y1) => {
104                if x1 == FieldElement::ZERO {
105                    return Point::Infinity;
106                }
107                let lambda = x1 + y1.multiply(x1.invert());
108                let x3 = lambda.square() + lambda + FieldElement::ONE;
109                let y3 = x1.square() + (lambda + FieldElement::ONE).multiply(x3);
110                Point::Affine(x3, y3)
111            }
112        }
113    }
114
115    /// Constant-time scalar multiplication: `k * self`, `k` a big-endian 163-bit scalar (top 5
116    /// bits of `k[0]` must be zero - values are always < the curve order, which is < `2^163`).
117    ///
118    /// Montgomery's method for binary-curve point multiplication (`Guide to Elliptic Curve
119    /// Cryptography`, Algorithm 3.40) computes only X/Z-projective coordinates through the main
120    /// loop, recovering the affine `y` at the end from the original point. Two adaptations from
121    /// the textbook algorithm, both needed for `docs/DECISIONS.md` D-25's branchless posture:
122    ///
123    /// - The textbook version starts from `(P, 2P)` and loops only over the bits below `k`'s
124    ///   *actual* highest set bit - a loop bound that depends on the secret scalar's magnitude.
125    ///   This starts from `(Infinity, P)` instead (`Z = 0` represents infinity in this coordinate
126    ///   system - doubling or adding into it stays at `Z = 0`, verified algebraically against the
127    ///   same formulas below) and always runs the full 163 iterations, so leading zero bits of a
128    ///   smaller scalar cost nothing extra and leak nothing about where the top bit actually is.
129    /// - Each iteration's `if k_i == 1 {...} else {...}` (textbook step 2.1/2.2) is two formulas
130    ///   that are identical in shape, differing only in which of the two (X, Z) pairs plays which
131    ///   role. Implemented here as: conditionally swap the pair (branchless, via a XOR/mask swap)
132    ///   based on the bit, always run the "`k_i == 1`" formula, then swap back - the same
133    ///   operations execute on every iteration regardless of the bit's value.
134    ///
135    /// The final X-only-to-affine recovery needs `kP` and `(k+1)P` to both be finite points -
136    /// undefined at the two boundary values where that fails (`k == 0`/`k == ord(self)`, and
137    /// `k == ord(self) - 1` respectively). `docs/DECISIONS.md` D-110 (T-152) found and fixed a real
138    /// bug here: the recovery silently used `invert(ZERO) == ZERO` instead of detecting either
139    /// case, producing a wrong-but-plausible-looking result rather than failing loudly or
140    /// computing the right answer. See the `z1 == ZERO`/`z2 == ZERO` handling in the body below.
141    #[must_use]
142    pub fn scalar_multiply(self, k: &[u8; 21]) -> Self {
143        match self {
144            Point::Infinity => Point::Infinity,
145            Point::Affine(x, y) => {
146                let mut x1 = FieldElement::ONE;
147                let mut z1 = FieldElement::ZERO; // (x1 : z1) = Infinity
148                let mut x2 = x;
149                let mut z2 = FieldElement::ONE; // (x2 : z2) = P
150
151                for i in (0..163u32).rev() {
152                    let bit = bit_at(k, i);
153                    let swap = bit ^ 1;
154                    cswap(swap, &mut x1, &mut x2);
155                    cswap(swap, &mut z1, &mut z2);
156
157                    let t1 = z1;
158                    z1 = (x1.multiply(z2) + x2.multiply(z1)).square();
159                    x1 = x.multiply(z1) + x1.multiply(x2).multiply(t1).multiply(z2);
160                    let t2 = x2;
161                    x2 = x2.square().square() + b().multiply(z2.square().square());
162                    z2 = t2.square().multiply(z2.square());
163
164                    cswap(swap, &mut x1, &mut x2);
165                    cswap(swap, &mut z1, &mut z2);
166                }
167
168                // `z1 == ZERO` means `kP == Infinity` (`k == 0`, or `k == ord(self)` - only
169                // reachable at or past the edge of this function's documented `k < n` domain for a
170                // full-order point). `Point::Infinity` is a different enum variant from
171                // `Point::Affine`, not a same-shape value a branchless mask can select between
172                // (unlike the `z2 == ZERO` case below) - this needs an explicit branch.
173                // `docs/DECISIONS.md` D-110 (T-152): this only fires for `k == 0` or `k >=
174                // ord(self)`, never for an in-range secret scalar DSTU 4145 sign/verify actually
175                // constructs, so it is not a fresh timing side channel for real callers - a
176                // deliberate, named exception to this function's otherwise branchless posture,
177                // not left as silently-wrong output the way it was before this fix. The zero
178                // *test* itself still uses `is_zero_mask` (fixed-shape, no secret-dependent `==`)
179                // rather than `FieldElement`'s derived `PartialEq` - only the resulting branch on
180                // that mask is data-dependent, which is unavoidable given `Point::Infinity` is a
181                // different enum variant (`docs/SECURITY.md`'s "never `==` on secret data" rule).
182                if is_zero_mask(z1) != 0 {
183                    return Point::Infinity;
184                }
185
186                let x1_affine = x1.multiply(z1.invert());
187                let x2_affine = x2.multiply(z2.invert());
188                let t1 = x1_affine + x;
189                let t2 = x2_affine + x;
190                let inner = t1.multiply(t2) + x.square() + y;
191                let y1_affine_formula = x.invert().multiply(t1).multiply(inner) + y;
192
193                // `z2 == ZERO` means `(k+1)P == Infinity`, i.e. `k == ord(self) - 1` - genuinely
194                // inside the documented `k < n` contract (`docs/DECISIONS.md` D-110, T-152). The
195                // y-recovery formula above needs `(k+1)P`'s affine x-coordinate, which doesn't
196                // exist for infinity; `x2.multiply(z2.invert())` silently reads `invert(ZERO) ==
197                // ZERO` instead, corrupting `y1_affine_formula`. The correct `kP` in that case is
198                // exactly `-self = (x, x + y)` (`Point::negate`) - `x1_affine` already comes out
199                // right either way (`self`/`-self` share an x-coordinate on this curve family), so
200                // only `y` needs correcting. Selected branchlessly (same mask-and-XOR shape as
201                // `cswap` above, not `==`/`if`) since `z2` is secret-scalar-derived and both
202                // candidate values are same-shape field elements.
203                let y1_affine = select(is_zero_mask(z2), x + y, y1_affine_formula);
204
205                Point::Affine(x1_affine, y1_affine)
206            }
207        }
208    }
209}

211impl core::ops::Add for Point {
212    type Output = Self;
213
214    /// Affine point addition. Same public-data-only caveat as `double` - see the module doc.
215    fn add(self, other: Self) -> Self {
216        match (self, other) {
217            (Point::Infinity, q) => q,
218            (p, Point::Infinity) => p,
219            (Point::Affine(x1, y1), Point::Affine(x2, y2)) => {
220                if x1 == x2 {
221                    if y1 == y2 {
222                        return self.double();
223                    }
224                    // x1 == x2 and y1 != y2: the only other point sharing x1 is -P = (x1, x1+y1)
225                    // (char-2 curve of this form), so `other` must be `-self`.
226                    return Point::Infinity;
227                }
228                let lambda = (y1 + y2).multiply((x1 + x2).invert());
229                let x3 = lambda.square() + lambda + x1 + x2 + FieldElement::ONE;
230                let y3 = lambda.multiply(x1 + x3) + x3 + y1;
231                Point::Affine(x3, y3)
232            }
233        }
234    }
235}

237fn bit_at(bytes: &[u8; 21], i: u32) -> u64 {
238    let byte_index = 20 - (i / 8) as usize;
239    let bit_in_byte = i % 8;
240    u64::from((bytes[byte_index] >> bit_in_byte) & 1)
241}

243/// Constant-time conditional swap: swaps `a` and `b` in place when `swap == 1`, leaves both
244/// unchanged when `swap == 0` - no branch, same word operations either way.
245fn cswap(swap: u64, a: &mut FieldElement, b: &mut FieldElement) {
246    let mask = 0u64.wrapping_sub(swap);
247    for i in 0..3 {
248        let t = mask & (a.0[i] ^ b.0[i]);
249        a.0[i] ^= t;
250        b.0[i] ^= t;
251    }
252}

254/// Constant-time "is this field element zero" test: an all-ones mask if so, all-zeros otherwise.
255/// The standard `x | wrapping_neg(x)` top-bit branchless zero test (for any nonzero `x`, either
256/// `x` or `-x` has its sign bit set in two's-complement; for `x == 0` neither does), applied to
257/// the OR of all 3 limbs (any single nonzero limb makes the whole element nonzero) - `docs/
258/// DECISIONS.md` D-110 (T-152).
259fn is_zero_mask(a: FieldElement) -> u64 {
260    let combined = a.0[0] | a.0[1] | a.0[2];
261    let is_nonzero = (combined | combined.wrapping_neg()) >> 63;
262    0u64.wrapping_sub(1 ^ is_nonzero)
263}

265/// Constant-time select: returns `if_mask` when `mask` is all-ones, `otherwise` when all-zeros -
266/// same XOR-and-mask shape as `cswap`'s swap above, specialized to a one-sided select
267/// (`docs/DECISIONS.md` D-110, T-152).
268fn select(mask: u64, if_mask: FieldElement, otherwise: FieldElement) -> FieldElement {
269    let mut out = [0u64; 3];
270    for ((out_limb, a), b) in out.iter_mut().zip(if_mask.0.iter()).zip(otherwise.0.iter()) {
271        *out_limb = b ^ (mask & (a ^ b));
272    }
273    FieldElement(out)
274}

276/// López-Dahab projective point: `(X:Y:Z)` represents affine `(x, y) = (X/Z, Y/Z^2)` - **note the
277/// asymmetric scaling** (`Y` by `Z^2`, not `Z`, unlike Jacobian coordinates). `Z == FieldElement::
278/// ZERO` represents the point at infinity (any `X`/`Y` at that point are unconstrained - `to_affine`
279/// checks `Z` first and never reads them). Used only by `verify_combine`'s default-profile fast
280/// path below (`docs/DECISIONS.md` D-108) - public-data-only, never constant-time, never for a
281/// secret scalar. `double`/`mixed_add` below are the explicit formulas "dbl-2005-dl"/"madd-2005-dl"
282/// from the Bernstein/Lange Explicit-Formulas Database (`hyperelliptic.org/EFD/g12o/auto-shortw-
283/// lopezdahab.html`), specialized to this curve's `a2 = 1` (the module's `a = 1` convention) -
284/// their stated costs (4M+5S doubling, 8M+5S mixed addition) were independently re-derived by
285/// listing every `multiply`/`square` call below and matched the database's own count exactly,
286/// which is the actual correctness check for the transcription (this codebase has no local copy of
287/// a citable textbook derivation to check against page numbers, only the differential proptest in
288/// `tests/dstu4145_curve.rs` and this cost cross-check).
289#[cfg(not(feature = "small-tables"))]
290#[derive(Clone, Copy)]
291struct ProjectivePoint {
292    x: FieldElement,
293    y: FieldElement,
294    z: FieldElement,
295}
296
297#[cfg(not(feature = "small-tables"))]
298impl ProjectivePoint {
299    fn from_affine(p: Point) -> Self {
300        match p {
301            Point::Infinity => ProjectivePoint {
302                x: FieldElement::ONE,
303                y: FieldElement::ONE,
304                z: FieldElement::ZERO,
305            },
306            Point::Affine(x, y) => ProjectivePoint {
307                x,
308                y,
309                z: FieldElement::ONE,
310            },
311        }
312    }
313
314    /// "dbl-2005-dl", `a2 = 1`: `A = Z1^2; B = b*A^2; C = X1^2; Z3 = A*C; X3 = C^2+B;
315    /// Y3 = (Y1^2+Z3+B)*X3 + Z3*B` (the database's `a2*Z3` term is just `Z3` here, since `a2=1`).
316    /// A point with affine `x = 0` (order-2 point) doubles to infinity automatically through this
317    /// formula (`C = X1^2 = 0` forces `Z3 = A*C = 0`), matching the affine `double`'s explicit
318    /// `x1 == ZERO` check above without needing a separate branch here.
319    fn double(self) -> Self {
320        if self.z == FieldElement::ZERO {
321            return self; // doubling infinity is infinity
322        }
323        let a = self.z.square();
324        let big_b = b().multiply(a.square());
325        let c = self.x.square();
326        let z3 = a.multiply(c);
327        let x3 = c.square() + big_b;
328        let y3 = (self.y.square() + z3 + big_b).multiply(x3) + z3.multiply(big_b);
329        ProjectivePoint {
330            x: x3,
331            y: y3,
332            z: z3,
333        }
334    }
335
336    /// "madd-2005-dl", `a2 = 1`, `other` affine (`Z2 = 1`): `A = Y1+Y2*Z1^2; B = X1+X2*Z1;
337    /// C = B*Z1; Z3 = C^2; D = X2*Z3; X3 = A^2+C*(A+B^2+C); Y3 = (D+X3)*(A*C+Z3)+(Y2+X2)*Z3^2`.
338    /// Guarded ahead of the database's generic-case formula (which assumes neither operand is
339    /// infinity and the two affine `x`-coordinates differ) for totality, not for attack
340    /// resistance - `verify`'s final `r' == r` check (`signature.rs`) already closes off any
341    /// benefit an attacker could get from steering the accumulator here; the guards exist because
342    /// the formula is otherwise partial, on public data where an honest partial sum can
343    /// structurally (if very rarely, ~2^-163) coincide with infinity or with the table point.
344    fn mixed_add(self, other: Point) -> Self {
345        let (x2, y2) = match other {
346            Point::Infinity => return self,
347            Point::Affine(x, y) => (x, y),
348        };
349        if self.z == FieldElement::ZERO {
350            return ProjectivePoint::from_affine(other);
351        }
352        let z1_sq = self.z.square();
353        let a = self.y + y2.multiply(z1_sq);
354        let b_val = self.x + x2.multiply(self.z);
355        if b_val == FieldElement::ZERO {
356            return if a == FieldElement::ZERO {
357                self.double() // same point (B=0 means matching affine x; A=0 means matching y too)
358            } else {
359                // matching x, differing y: char-2 negation is (x, x+y) (see `Point::negate`), so
360                // this is `-self` - the sum is infinity.
361                ProjectivePoint {
362                    x: FieldElement::ONE,
363                    y: FieldElement::ONE,
364                    z: FieldElement::ZERO,
365                }
366            };
367        }
368        let c = b_val.multiply(self.z);
369        let z3 = c.square();
370        let d = x2.multiply(z3);
371        let b_sq = b_val.square();
372        let x3 = a.square() + c.multiply(a + b_sq + c);
373        let y3 = (d + x3).multiply(a.multiply(c) + z3) + (y2 + x2).multiply(z3.square());
374        ProjectivePoint {
375            x: x3,
376            y: y3,
377            z: z3,
378        }
379    }
380
381    /// The one deferred inversion for the whole `verify_combine` computation: `x = X*Z^-1`,
382    /// `y = Y*(Z^-1)^2` (matching the `(X/Z, Y/Z^2)` convention stated on the type itself).
383    fn to_affine(self) -> Point {
384        if self.z == FieldElement::ZERO {
385            return Point::Infinity;
386        }
387        let z_inv = self.z.invert();
388        let x = self.x.multiply(z_inv);
389        let y = self.y.multiply(z_inv.square());
390        Point::Affine(x, y)
391    }
392}

394/// `s*g + r*q` via Shamir's trick (simultaneous double-and-add over both scalars, one shared
395/// doubling per bit position) in projective coordinates, deferring every inversion to the single
396/// one inside `to_affine` at the end - `docs/DECISIONS.md` D-108. Public data only (`g`, `s`, `q`,
397/// `r` are all public in DSTU 4145 verification, per this module's own doc comment); safe to
398/// branch on the scalars' actual bit lengths (skip leading zero bit-pairs) precisely because this
399/// is not a secret-scalar code path, unlike `scalar_multiply`'s fixed 163 iterations.
400#[cfg(not(feature = "small-tables"))]
401fn shamir_double_scalar_multiply(g: Point, s: &[u8; 21], q: Point, r: &[u8; 21]) -> Point {
402    let g_plus_q = g + q; // existing trusted affine `Point::add` - one inversion, paid once here
403    let table = [Point::Infinity, q, g, g_plus_q]; // index (bit_s << 1) | bit_r
404
405    let top = (0..163u32)
406        .rev()
407        .find(|&i| bit_at(s, i) != 0 || bit_at(r, i) != 0);
408    let Some(top) = top else {
409        return Point::Infinity; // s == r == 0 - not reachable via `verify`'s own guards
410    };
411
412    let entry_at = |i: u32| -> Point {
413        // `bit_at` only ever returns 0 or 1, so this index is provably 0..=3 - no truncation
414        // possible regardless of pointer width.
415        #[allow(clippy::cast_possible_truncation)]
416        let index = ((bit_at(s, i) << 1) | bit_at(r, i)) as usize;
417        table[index]
418    };
419
420    let mut acc = ProjectivePoint::from_affine(entry_at(top));
421    for i in (0..top).rev() {
422        acc = acc.double();
423        let entry = entry_at(i);
424        if entry != Point::Infinity {
425            acc = acc.mixed_add(entry);
426        }
427    }
428    acc.to_affine()
429}

431/// `verify`'s `s*g + r*q` combine step - one function, two bodies, matching
432/// `hazmat::tables::apply_forward_matrix`'s `small-tables` idiom exactly (no `#[cfg]` at the call
433/// site in `signature.rs`). Default: the projective/Shamir fast path above. `small-tables`: today's
434/// unchanged double `scalar_multiply` call - see `docs/DECISIONS.md` D-108 for why this reuses the
435/// same feature Kalyna/Kupyna/Strumok use for their own big-table/small-table split, even though
436/// this particular case is a code-size/audit-surface tradeoff, not a flash-`const`-table one (no
437/// new `const` table is added; `{Infinity, g, q, g+q}` above is computed fresh per call).
438#[cfg(not(feature = "small-tables"))]
439#[must_use]
440pub fn verify_combine(g: Point, s: &[u8; 21], q: Point, r: &[u8; 21]) -> Point {
441    shamir_double_scalar_multiply(g, s, q, r)
442}

444#[cfg(feature = "small-tables")]
