1//! GF(2^257) field arithmetic, reduced modulo the trinomial `x^257 + x^12 + 1` - the reduction
2//! polynomial of DSTU 4145-2002's `m=257` curve. Domain parameters extracted from two independent
3//! real DSTU 4145 certificates (a `czo.gov.ua` test-CA certificate and a real Diia-issued
4//! production certificate), byte-reversed per the certificates' own "DSTU 4145-2002 little endian"
5//! OID convention, confirmed against Bouncy Castle's `DSTU4145NamedCurves.java` `curves[6]`
6//! (`ECCurve.F2m(257, 12, ZERO, ...)`) - see `docs/DECISIONS.md` D-185/D-186 for the full
7//! provenance and the byte-order pitfall found deriving it.
8//!
9//! **Correctness-first, not the word-offset-optimized closed form `gf2m163::reduce` uses** for its
10//! pentanomial - that hand-derivation is specific to `(m, W)` and doesn't generalize without
11//! redoing it from scratch per field size, the same posture already established for
12//! `gf2m_wide`'s generic multi-size reduction (`docs/DECISIONS.md`). `reduce` below implements the
13//! trinomial identity `x^(257+t) = x^(12+t) + x^t` directly via two fixed folding passes (see its
14//! own doc comment for the bit-count proof that two passes are necessary and sufficient) - a future
15//! perf task may hand-derive a closed form the way `gf2m163::reduce` has one, mirroring `T-45`'s
16//! own "not scheduled, sketched only" precedent for `gf2m163::multiply`.
17//!
18//! **Branchless by construction** (`docs/DECISIONS.md` D-25), same discipline as `gf2m163`: no
19//! secret-dependent branching or array indexing anywhere below. `multiply`'s software path selects
20//! each shifted operand via an all-ones/all-zeros mask; `reduce`'s folding passes run unconditionally
21//! regardless of operand values; `invert`'s only "branch" is on the fixed, public addition chain for
22//! the exponent `2^257 - 2`, identical on every call regardless of the secret operand.
23
24/// An element of GF(2^257): 5 little-endian 64-bit limbs. Bits 257..320 (the unused top 63 bits of
25/// the last limb) are always zero - every constructor and operation below maintains this.
26#[derive(Clone, Copy, Debug, PartialEq, Eq)]
27pub struct FieldElement(pub(crate) [u64; 5]);
28
29impl core::ops::Add for FieldElement {
30    type Output = Self;

32    /// GF(2^257) addition is bitwise XOR - no carry, no reduction needed.
33    fn add(self, other: Self) -> Self {
34        FieldElement([
35            self.0[0] ^ other.0[0],
36            self.0[1] ^ other.0[1],
37            self.0[2] ^ other.0[2],
38            self.0[3] ^ other.0[3],
39            self.0[4] ^ other.0[4],
40        ])
41    }
42}

44impl FieldElement {
45    pub const ZERO: FieldElement = FieldElement([0, 0, 0, 0, 0]);
46    pub const ONE: FieldElement = FieldElement([1, 0, 0, 0, 0]);

48    /// Builds a field element from a big-endian byte slice (up to 33 bytes / 257 bits). The
49    /// caller must ensure the value is already less than `2^257` - this does not reduce.
50    #[must_use]
51    pub fn from_be_bytes(bytes: &[u8]) -> Self {
52        let mut limbs = [0u64; 5];
53        for (i, &byte) in bytes.iter().rev().enumerate() {
54            let limb = i / 8;
55            let shift = (i % 8) * 8;
56            limbs[limb] |= u64::from(byte) << shift;
57        }
58        FieldElement(limbs)
59    }
60
61    /// Big-endian encoding, fixed at 33 bytes (257 bits, rounded up to a whole byte count).
62    #[must_use]
63    #[allow(clippy::cast_possible_truncation)] // deliberate: extracting one byte from a shifted limb
64    pub fn to_be_bytes(self) -> [u8; 33] {
65        let mut out = [0u8; 33];
66        for (i, byte) in out.iter_mut().rev().enumerate() {
67            let limb = i / 8;
68            let shift = (i % 8) * 8;
69            *byte = (self.0[limb] >> shift) as u8;
70        }
71        out
72    }

74    /// Dispatches to the hardware-`clmul` `poly_mul_wide_hw` below when the CPU supports it
75    /// (`std`-gated runtime detection, same shape as `gf2m163`/`gf2m_wide`'s own dispatch,
76    /// `docs/TASKS.md` T-198/T-199, `docs/DECISIONS.md` D-184/D-186 Decision 4) and falls back to
77    /// the portable bit-serial `poly_mul_wide` otherwise. Landed together with the software path
78    /// from this module's first commit, not as a later follow-up task - the dispatch design was
79    /// already proven on `gf2m163`/`gf2m_wide`.
80    #[must_use]
81    pub fn multiply(self, other: Self) -> Self {
82        #[cfg(all(
83            feature = "std",
84            not(kani),
85            any(target_arch = "x86_64", target_arch = "aarch64")
86        ))]
87        if crate::hazmat::gf2m_wide::clmul_native::feature_available() {
88            // Safety: `feature_available()` just confirmed the CPU supports the target feature
89            // `poly_mul_wide_hw` requires.
90            let wide = unsafe { poly_mul_wide_hw(&self.0, &other.0) };
91            return reduce(wide);
92        }
93        reduce(poly_mul_wide(&self.0, &other.0))
94    }
95
96    #[must_use]
97    pub fn square(self) -> Self {
98        reduce(square_wide(&self.0))
99    }

101    /// `self^-1 = self^(2^257 - 2)`, by Fermat's little theorem for `GF(2^257)*`. Undefined for
102    /// `self == ZERO` - this returns `ZERO` in that case (Fermat's formula gives `0^k = 0` for any
103    /// positive `k`), not a panic, but that value is not a meaningful inverse.
104    ///
105    /// `2^257 - 2 = 2*(2^256 - 1)`, so this computes `(self^(2^256 - 1))^2`. Since `256 = 2^8` is
106    /// itself a power of two, the addition chain for `self^(2^256-1)` is the simplest possible
107    /// shape - repeated doubling, `T_(2k) = T_k^(2^k) * T_k` - needing no odd steps the way
108    /// `gf2m163::invert`'s chain for the non-power-of-two exponent 163 does:
109    /// `1 -> 2 -> 4 -> 8 -> 16 -> 32 -> 64 -> 128 -> 256`, 8 combine steps (one multiply each, plus
110    /// the fixed number of squarings the `2^k` part costs), 255 total squarings either way (matches
111    /// `gf2m163::invert`'s own "squaring does not become free" note - `docs/DECISIONS.md` D-109).
112    /// Verified against a direct (non-addition-chain) oracle by differential test, not by the
113    /// chain's derivation alone - see `invert_direct`/`invert_matches_invert_direct` in `tests`.
114    #[must_use]
115    pub fn invert(self) -> Self {
116        // `sq_n(x, n) = x^(2^n)`, i.e. `n` repeated squarings.
117        let sq_n = |mut x: Self, n: u32| -> Self {
118            for _ in 0..n {
119                x = x.square();
120            }
121            x
122        };
123
124        let t1 = self; // self^(2^1 - 1)
125        let t2 = sq_n(t1, 1).multiply(t1); // self^(2^2 - 1)
126        let t4 = sq_n(t2, 2).multiply(t2); // self^(2^4 - 1)
127        let t8 = sq_n(t4, 4).multiply(t4); // self^(2^8 - 1)
128        let t16 = sq_n(t8, 8).multiply(t8); // self^(2^16 - 1)
129        let t32 = sq_n(t16, 16).multiply(t16); // self^(2^32 - 1)
130        let t64 = sq_n(t32, 32).multiply(t32); // self^(2^64 - 1)
131        let t128 = sq_n(t64, 64).multiply(t64); // self^(2^128 - 1)
132        let t256 = sq_n(t128, 128).multiply(t128); // self^(2^256 - 1)
133
134        t256.square() // self^(2^257 - 2)
135    }
136}

138/// Binary-polynomial (carry-less) multiplication of two 257-bit operands into a 10-limb
139/// (640-bit capacity, up to 513 significant bits) product - the same right-to-left shift-and-add
140/// method `gf2m163::poly_mul_wide` uses, widened to 5/10 limbs.
141fn poly_mul_wide(a: &[u64; 5], b: &[u64; 5]) -> [u64; 10] {
142    let mut acc = [0u64; 10];
143    let mut shifted = [b[0], b[1], b[2], b[3], b[4], 0, 0, 0, 0, 0];
144
145    for bit_index in 0..257u32 {
146        let limb = (bit_index / 64) as usize;
147        let bit = bit_index % 64;
148        let bit_value = (a[limb] >> bit) & 1;
149        let mask = 0u64.wrapping_sub(bit_value); // all-ones if the bit is 1, all-zeros otherwise
150        for i in 0..10 {
151            acc[i] ^= shifted[i] & mask;
152        }
153        shl1(&mut shifted);
154    }
155
156    acc
157}

159/// Hardware carry-less-multiply `poly_mul_wide` replacement (`docs/TASKS.md` T-199,
160/// `docs/DECISIONS.md` D-186 Decision 4) - schoolbook combination of 25 pairwise 64x64->128
161/// hardware `clmul`s, the same limb-placement identity `poly_mul_wide` computes a different way.
162/// See `gf2m163::poly_mul_wide_hw`'s own doc comment for why this shape (whole loop inlined inside
163/// one `#[target_feature]` function, no separate call boundary) was chosen over a software
164/// comb-method rewrite - identical reasoning applies here, this module inherits the same
165/// secret-scalar exposure via `curve257::scalar_multiply` once that module exists.
166#[cfg(all(feature = "std", target_arch = "x86_64"))]
167#[target_feature(enable = "pclmulqdq")]
168#[allow(
169    clippy::cast_possible_wrap,
170    clippy::cast_possible_truncation,
171    clippy::cast_sign_loss
172)]
173// deliberate: `_mm_set_epi64x`/`_mm_cvtsi128_si64` use `i64` purely as a bit container (no signed
174// interpretation anywhere below) - the casts are the intended reinterpret, not a value-changing
175// truncation.
176unsafe fn poly_mul_wide_hw(a: &[u64; 5], b: &[u64; 5]) -> [u64; 10] {
177    use std::arch::x86_64::{
178        _mm_clmulepi64_si128, _mm_cvtsi128_si64, _mm_set_epi64x, _mm_srli_si128,
179    };
180    let mut out = [0u64; 10];
181    for i in 0..5 {
182        for j in 0..5 {
183            // Safety: this function itself requires `pclmulqdq` (target_feature, callers gate on
184            // `clmul_native::feature_available()` first).
185            let ma = _mm_set_epi64x(0, a[i] as i64);
186            let mb = _mm_set_epi64x(0, b[j] as i64);
187            let prod = _mm_clmulepi64_si128(ma, mb, 0x00);
188            let lo = _mm_cvtsi128_si64(prod) as u64;
189            let hi = _mm_cvtsi128_si64(_mm_srli_si128::<8>(prod)) as u64;
190            out[i + j] ^= lo;
191            out[i + j + 1] ^= hi;
192        }
193    }
194    out
195}

197/// `aarch64` sibling of the `x86_64` `poly_mul_wide_hw` above - see its own doc comment.
198#[cfg(all(feature = "std", target_arch = "aarch64"))]
199#[target_feature(enable = "aes")]
200unsafe fn poly_mul_wide_hw(a: &[u64; 5], b: &[u64; 5]) -> [u64; 10] {
201    use std::arch::aarch64::vmull_p64;
202    let mut out = [0u64; 10];
203    for i in 0..5 {
204        for j in 0..5 {
205            // Safety: this function itself requires `aes`/`PMULL` (target_feature, callers gate
206            // on `clmul_native::feature_available()` first).
207            let prod: u128 = vmull_p64(a[i], b[j]);
208            out[i + j] ^= prod as u64;
209            out[i + j + 1] ^= (prod >> 64) as u64;
210        }
211    }
212    out
213}

225/// Spreads the low 32 bits of `x` across the low 64 bits of the result - see
226/// `gf2m163::spread32to64`'s own doc comment for the full derivation (identical technique, this is
227/// the same function, not re-derived).
228fn spread32to64(x: u32) -> u64 {
229    let mut x = u64::from(x);
230    x = (x | (x << 16)) & 0x0000_FFFF_0000_FFFF;
231    x = (x | (x << 8)) & 0x00FF_00FF_00FF_00FF;
232    x = (x | (x << 4)) & 0x0F0F_0F0F_0F0F_0F0F;
233    x = (x | (x << 2)) & 0x3333_3333_3333_3333;
234    x = (x | (x << 1)) & 0x5555_5555_5555_5555;
235    x
236}

238/// Squares a 257-bit operand into its 10-limb (up to 513-bit) wide product, via `spread32to64`
239/// instead of a full carry-less self-multiplication - same technique and per-limb placement as
240/// `gf2m163::square_wide`, widened to 5 input / 10 output limbs.
241fn square_wide(a: &[u64; 5]) -> [u64; 10] {
242    let mut out = [0u64; 10];
243    for i in 0..5 {
244        #[allow(clippy::cast_possible_truncation)]
245        // deliberate: splitting a limb into its two halves
246        let lo = a[i] as u32;
247        #[allow(clippy::cast_possible_truncation)]
248        // deliberate: splitting a limb into its two halves
249        let hi = (a[i] >> 32) as u32;
250        out[2 * i] = spread32to64(lo);
251        out[2 * i + 1] = spread32to64(hi);
252    }
253    out
254}

256/// Left-shifts a 10-limb little-endian array by exactly 1 bit, in place.
257fn shl1(x: &mut [u64; 10]) {
258    let mut carry = 0u64;
259    for limb in x.iter_mut() {
260        let next_carry = *limb >> 63;
261        *limb = (*limb << 1) | carry;
262        carry = next_carry;
263    }
264}

266/// Reduces a 10-limb (up to 576-bit capacity, up to 513 significant bits) product modulo
267/// `x^257 + x^12 + 1`, producing a fully-reduced 5-limb field element.
268///
269/// Implements the trinomial identity `x^(257+t) = x^(12+t) + x^t` directly, i.e. the "excess"
270/// portion `h = c >> 257` folds back in as `h XOR (h << 12)`, applied twice:
271///
272/// **Pass 1.** Input `c` has at most 513 significant bits (two <=257-bit operands multiply to at
273/// most 513 bits). `h1 = c >> 257` therefore has at most `513 - 257 = 256` bits - exactly 4 limbs,
274/// computed here into a 5-limb buffer (limb 4 providably always zero, not runtime-checked, matching
275/// this module's branchless posture). `h1 << 12` then has at most `256 + 12 = 268` bits - 5 limbs
276/// (limb-4-max bit index `267 - 256 = 11`), still fits the same 5-limb buffer with no 6th limb ever
277/// needed. `pass1 = lo1 XOR h1 XOR (h1 << 12)`, where `lo1` is `c`'s low 257 bits - `pass1` can
278/// therefore still exceed 257 bits, by at most `268 - 257 = 11` bits.
279///
280/// **Pass 2 (cleanup).** `h2 = pass1 >> 257` has at most 11 bits by the bound above - small enough
281/// to live entirely in a single word (`pass1[4] >> 1`, no cross-limb carry). `h2 << 12` then has at
282/// most `11 + 12 = 23` bits, also a single word, `XORed` directly into the result's limb 0 (23 < 64,
283/// no spread across limbs needed). The result is `lo2 XOR h2 XOR (h2 << 12)`, all now within 257
284/// bits - **exactly two passes are necessary and sufficient**, proven by this bit-count bound, not
285/// discovered by iterating until convergence (branchless per `docs/DECISIONS.md` D-25 - same
286/// "provably sufficient, run the fixed count" posture `gf2m163::reduce`'s own cleanup pass uses).
287fn reduce(c: [u64; 10]) -> FieldElement {
288    // Pass 1: split c into its low 257 bits (lo1) and everything above (h1 = c >> 257), then fold
289    // h1 back in via the trinomial identity.
290    let mut h1 = [0u64; 5];
291    for i in 0..5 {
292        h1[i] = (c[i + 4] >> 1) | (c[i + 5] << 63);
293    }
294    let lo1 = [c[0], c[1], c[2], c[3], c[4] & 1];
295
296    let mut h1_shifted = [0u64; 5];
297    h1_shifted[0] = h1[0] << 12;
298    for i in 1..5 {
299        h1_shifted[i] = (h1[i] << 12) | (h1[i - 1] >> 52);
300    }
301
302    let mut pass1 = [0u64; 5];
303    for i in 0..5 {
304        pass1[i] = lo1[i] ^ h1[i] ^ h1_shifted[i];
305    }
306
307    // Pass 2 (cleanup): pass1 has at most 268 bits, so its own excess (h2 = pass1 >> 257) has at
308    // most 11 bits - small enough that h2 and h2 << 12 (<= 23 bits) both live entirely in limb 0,
309    // no multi-limb shifting needed.
310    let h2 = pass1[4] >> 1;
311    let h2_shifted = h2 << 12;
312
313    let mut result = [pass1[0], pass1[1], pass1[2], pass1[3], pass1[4] & 1];
314    result[0] ^= h2 ^ h2_shifted;
315
316    FieldElement(result)
317}

360    /// `reduce`'s two-pass folding, cross-checked at the wide-product level for values sitting
361    /// exactly at the pass-1/pass-2 boundary this module's own doc comment derives (bit 267, the
362    /// highest bit `h1 << 12` can ever set) against a direct, unoptimized bit-at-a-time reduction -
363    /// an independent second algorithm, not a restatement of `reduce`'s own folding steps.
364    fn reduce_naive(c: [u64; 10]) -> FieldElement {
365        // Direct polynomial long division against x^257 + x^12 + 1: for each bit from the top
366        // down, if set, XOR the shifted trinomial pattern (x^12 + 1, i.e. bits {12, 0}) in at that
367        // position and clear the source bit - the textbook definition of reduction mod f(x).
368        let mut c = c;
369        for bit in (257..513).rev() {
370            let limb = bit / 64;
371            let shift = bit % 64;
372            let is_set = (c[limb] >> shift) & 1 == 1;
373            if is_set {
374                c[limb] ^= 1u64 << shift;
375                let t = bit - 257;
376                let t12 = t + 12;
377                c[t12 / 64] ^= 1u64 << (t12 % 64);
378                c[t / 64] ^= 1u64 << (t % 64);
379            }
380        }
381        FieldElement([c[0], c[1], c[2], c[3], c[4]])
382    }
