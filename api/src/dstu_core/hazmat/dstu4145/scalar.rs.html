1//! Scalar (mod `n`) integer arithmetic for DSTU 4145 signing - the curve's group order
2//! (`curve163::order()`), unrelated to `gf2m163::FieldElement`'s `GF(2^163)` polynomial
3//! arithmetic. Kept as a **distinct type** specifically so the two can never be mixed up by
4//! accident: both are 3-limb `[u64; 3]` internally, but `FieldElement::add` is XOR and
5//! `FieldElement::multiply` is carryless, while `Scalar::add`/`Scalar::multiply` are ordinary
6//! carrying integer arithmetic reduced mod `n` (`docs/DECISIONS.md` D-25's follow-up note - this was
7//! flagged as the layer's single biggest silent-correctness risk).
8//!
9//! Both operations are branchless throughout: `Scalar` carries the private key `d` and the
10//! ephemeral nonce `e` during signing, both secret. Reduction mod `n` uses a fixed-iteration
11//! restoring-division pass (double-and-conditionally-subtract, one pass per product bit, always
12//! run in full) rather than Barrett or any early-exit scheme - correctness-first, same posture as
13//! `gf2m163`'s field reduction.
14
15use super::curve163;
16use zeroize::Zeroize;
17
18// Not `ZeroizeOnDrop`: `Scalar` is `Copy` and used by-value pervasively (arithmetic ops, hazmat
19// `sign`/`verify` signatures) - `ZeroizeOnDrop`'s `Drop` impl is incompatible with `Copy` (E0184).
20// Callers holding a `Scalar` they want zeroized on drop (e.g. `crypto_sign::SigningKey`) call
21// `.zeroize()` explicitly instead.
22#[derive(Clone, Copy, Debug, PartialEq, Eq, Zeroize)]
23pub struct Scalar([u64; 3]);
24
25impl Scalar {
26    /// Builds a scalar from a big-endian byte slice (up to 21 bytes). The caller must ensure the
27    /// value is already less than `n` - this does not reduce (same convention as
28    /// `FieldElement::from_be_bytes`).
29    #[must_use]
30    pub fn from_be_bytes(bytes: &[u8]) -> Self {
31        Scalar(limbs_from_be_bytes(bytes))
32    }
33
34    #[must_use]
35    #[allow(clippy::cast_possible_truncation)] // deliberate: extracting one byte from a shifted limb
36    pub fn to_be_bytes(self) -> [u8; 21] {
37        let mut out = [0u8; 21];
38        for (i, byte) in out.iter_mut().rev().enumerate() {
39            let limb = i / 8;
40            let shift = (i % 8) * 8;
41            *byte = (self.0[limb] >> shift) as u8;
42        }
43        out
44    }
45
46    #[must_use]
47    pub fn is_zero(self) -> bool {
48        self.0 == [0, 0, 0]
49    }
50
51    fn n() -> [u64; 3] {
52        limbs_from_be_bytes(&curve163::order())
53    }
54
55    /// Builds a scalar from a big-endian 21-byte candidate, but only if it lies in `[1, n)` - the
56    /// valid private-key range. Used by `crypto_sign::SigningKey::generate`'s rejection-sampling
57    /// loop (`docs/TASKS.md` T-122): the comparison against `n` goes through the same constant-time
58    /// subtract-and-select primitive (`sub3`'s borrow flag) the rest of this module already uses
59    /// for secret arithmetic, rather than a branching `>=`, so evaluating one candidate doesn't
60    /// add a data-dependent-branch timing signal beyond the "how many draws until one was
61    /// accepted" every rejection-sampling scheme inherently has.
62    ///
63    /// Gated the same way its only caller (`crypto_sign::SigningKey::generate`) is: needs
64    /// `crate::randombytes` to draw candidates in the first place (`std` or the narrower
65    /// `getrandom` feature, `docs/TASKS.md` T-123/`docs/DECISIONS.md` D-74) - a build with neither has no way
66    /// to call this, so it would otherwise be dead code there.
67    #[cfg(any(feature = "std", feature = "getrandom"))]
68    #[must_use]
69    pub(crate) fn from_candidate_bytes(bytes: &[u8; 21]) -> Option<Self> {
70        let limbs = limbs_from_be_bytes(bytes);
71        let (_, borrow) = sub3(limbs, Self::n());
72        let in_range = borrow == 1; // borrow == 1 <=> limbs < n
73        let nonzero = limbs != [0, 0, 0];
74        (in_range && nonzero).then_some(Scalar(limbs))
75    }
76
77    #[must_use]
78    pub fn multiply(self, other: Self) -> Self {
79        Scalar(reduce_mod_n(mul3(self.0, other.0)))
80    }
81
82    /// Reduces an arbitrary-length big-endian byte string mod `n`, via the same bit-serial
83    /// restoring reduction as `reduce_mod_n` (constant-time, always processes every input bit
84    /// regardless of value) - used by `crypto_sign`'s deterministic nonce derivation to fold a
85    /// wider KMAC output (32+ bytes) into a valid scalar, the same role `reduce_mod_n` plays for a
86    /// multiplication product.
87    #[must_use]
88    pub(crate) fn reduce_wide_bytes(bytes: &[u8]) -> Self {
89        let n = Self::n();
90        let mut r = [0u64; 3];
91        for &byte in bytes {
92            for bit in (0..8).rev() {
93                let bit_val = u64::from((byte >> bit) & 1);
94                r = shl1_or(r, bit_val);
95                r = cond_sub_if_ge(r, n);
96            }
97        }
98        Scalar(r)
99    }
100}

102impl core::ops::Add for Scalar {
103    type Output = Self;
104
105    /// Ordinary integer addition mod `n` (not XOR - see the module doc).
106    fn add(self, other: Self) -> Self {
107        let (sum, _carry) = add3(self.0, other.0);
108        Scalar(cond_sub_if_ge(sum, Self::n()))
109    }
110}

112fn limbs_from_be_bytes(bytes: &[u8]) -> [u64; 3] {
113    let mut limbs = [0u64; 3];
114    for (i, &byte) in bytes.iter().rev().enumerate() {
115        let limb = i / 8;
116        let shift = (i % 8) * 8;
117        limbs[limb] |= u64::from(byte) << shift;
118    }
119    limbs
120}
121
122/// 3-limb add with carry-out.
123fn add3(a: [u64; 3], b: [u64; 3]) -> ([u64; 3], u64) {
124    let mut out = [0u64; 3];
125    let mut carry = 0u64;
126    for i in 0..3 {
127        let (s1, c1) = a[i].overflowing_add(b[i]);
128        let (s2, c2) = s1.overflowing_add(carry);
129        out[i] = s2;
130        carry = u64::from(c1) + u64::from(c2);
131    }
132    (out, carry)
133}
134
135/// 3-limb subtract with borrow-out (`1` if `a < b`).
136fn sub3(a: [u64; 3], b: [u64; 3]) -> ([u64; 3], u64) {
137    let mut out = [0u64; 3];
138    let mut borrow = 0u64;
139    for i in 0..3 {
140        let (d1, b1) = a[i].overflowing_sub(b[i]);
141        let (d2, b2) = d1.overflowing_sub(borrow);
142        out[i] = d2;
143        borrow = u64::from(b1) + u64::from(b2);
144    }
145    (out, borrow)
146}
147
148/// Returns `a - b` if `a >= b`, otherwise `a` unchanged - via a constant-time select on the
149/// subtraction's borrow flag (`borrow == 0` means `a >= b`), never a branch on the comparison.
150fn cond_sub_if_ge(a: [u64; 3], b: [u64; 3]) -> [u64; 3] {
151    let (diff, borrow) = sub3(a, b);
152    let mask = borrow.wrapping_sub(1); // borrow == 0 (a >= b) -> all-ones; borrow == 1 -> all-zeros
153    let mut out = [0u64; 3];
154    for i in 0..3 {
155        out[i] = a[i] ^ (mask & (a[i] ^ diff[i]));
156    }
157    out
158}

160/// 3-limb by 3-limb schoolbook multiplication into 6 limbs (real carrying arithmetic, unlike
161/// `gf2m163`'s carryless `poly_mul_wide`).
162#[allow(clippy::cast_possible_truncation)] // deliberate: low 64 bits of a u128 partial product
163fn mul3(a: [u64; 3], b: [u64; 3]) -> [u64; 6] {
164    let mut out = [0u64; 6];
165    for i in 0..3 {
166        let mut carry = 0u128;
167        for j in 0..3 {
168            let product = u128::from(a[i]) * u128::from(b[j]) + u128::from(out[i + j]) + carry;
169            out[i + j] = product as u64;
170            carry = product >> 64;
171        }
172        out[i + 3] = carry as u64;
173    }
174    out
175}

177/// Reduces a 6-limb product mod `n` via restoring division: process every bit of `product` from
178/// the most significant down, doubling the running remainder and folding in each bit, then always
179/// running the conditional subtract - the same fixed number of passes (`6 * 64`) regardless of
180/// `product`'s actual value.
181fn reduce_mod_n(product: [u64; 6]) -> [u64; 3] {
182    let n = Scalar::n();
183    let mut r = [0u64; 3];
184    for limb_idx in (0..6).rev() {
185        for bit in (0..64).rev() {
186            let bit_val = (product[limb_idx] >> bit) & 1;
187            r = shl1_or(r, bit_val);
188            r = cond_sub_if_ge(r, n);
189        }
190    }
191    r
192}
193
194/// Left-shifts a 3-limb value by 1 bit, OR-ing `bit` into the vacated low bit.
195fn shl1_or(x: [u64; 3], bit: u64) -> [u64; 3] {
196    let mut out = [0u64; 3];
197    let mut carry = bit;
198    for i in 0..3 {
199        let next_carry = x[i] >> 63;
200        out[i] = (x[i] << 1) | carry;
201        carry = next_carry;
202    }
203    out
204}

206#[cfg(all(test, any(feature = "std", feature = "getrandom")))]
207mod from_candidate_bytes_tests {
208    use super::{curve163, Scalar};
209
210    #[test]
211    fn rejects_zero() {
212        assert!(Scalar::from_candidate_bytes(&[0u8; 21]).is_none());
213    }
214
215    #[test]
216    fn rejects_n_itself() {
217        assert!(Scalar::from_candidate_bytes(&curve163::order()).is_none());
218    }
219
220    #[test]
221    fn rejects_above_n() {
222        let mut above_n = curve163::order();
223        above_n[20] += 1; // n's low byte is 0x4D, room to increment without carrying
224        assert!(Scalar::from_candidate_bytes(&above_n).is_none());
225    }
226
227    #[test]
228    fn accepts_n_minus_one() {
229        let mut n_minus_one = curve163::order();
230        n_minus_one[20] -= 1;
231        let scalar = Scalar::from_candidate_bytes(&n_minus_one).expect("n - 1 is in [1, n)");
232        assert_eq!(scalar.to_be_bytes(), n_minus_one);
233    }
234
235    #[test]
236    fn accepts_one() {
237        let mut one = [0u8; 21];
238        one[20] = 1;
239        let scalar = Scalar::from_candidate_bytes(&one).expect("1 is in [1, n)");
240        assert_eq!(scalar.to_be_bytes(), one);
241    }
242}
