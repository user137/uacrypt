1//! GF(2^163) field arithmetic, reduced modulo the pentanomial `x^163 + x^7 + x^6 + x^3 + 1` - the
2//! reduction polynomial of the DSTU 4145-2002 curve dual-sourced in
3//! `tests/vectors/dstu4145/gf2m163.json` (`docs/DECISIONS.md` D-14), matching
4//! `oracles/bouncycastle-java/.../DSTU4145NamedCurves.java`'s `ECCurve.F2m(163, 3, 6, 7, ...)`
5//! constructor (`k1, k2, k3 = 3, 6, 7`).
6//!
7//! **Branchless by construction** (`docs/DECISIONS.md` D-25): every operation below runs the same
8//! sequence of word ops regardless of the operand values. `multiply` selects each shifted operand
9//! via an all-ones/all-zeros mask derived from a single bit rather than an `if`, and `reduce`'s
10//! word-reduction and final cleanup pass run unconditionally rather than skipping zero words or
11//! looping until convergence (both real optimizations in the reference this was adapted from,
12//! OpenSSL's `BN_GF2m_mod_arr`, and both would branch on secret-dependent data here). `invert`'s
13//! only "branch" is on the fixed, public inversion exponent `2^163 - 2` - identical on every call
14//! regardless of the secret operand, so it carries no timing signal about that operand.
15//!
16//! `square` is a pure bit-spread (`spread32to64`/`square_wide`), not a full multiplication - GF(2)
17//! squaring satisfies `a(x)^2 = a(x^2)` since char-2 cross terms vanish, needing only fixed
18//! shift/AND/OR ops and no array indexing at all (`docs/DECISIONS.md` D-109, `docs/TASKS.md` T-153).
19//! `invert` still runs the same fixed 162-squaring chain either way; only the number of
20//! multiplies between squarings differs (D-109's Itoh-Tsujii-style addition chain vs. this file's
21//! prior direct form) - both are fixed, public, data-independent sequences.
22
23/// An element of GF(2^163): 3 little-endian 64-bit limbs. Bits 163..192 (the unused top 29 bits
24/// of the last limb) are always zero - every constructor and operation below maintains this.
25#[derive(Clone, Copy, Debug, PartialEq, Eq)]
26pub struct FieldElement(pub(crate) [u64; 3]);
27
28impl core::ops::Add for FieldElement {
29    type Output = Self;

31    /// GF(2^163) addition is bitwise XOR - no carry, no reduction needed (the XOR of two values
32    /// already below `2^163` stays below `2^163`).
33    fn add(self, other: Self) -> Self {
34        FieldElement([
35            self.0[0] ^ other.0[0],
36            self.0[1] ^ other.0[1],
37            self.0[2] ^ other.0[2],
38        ])
39    }
40}

42impl FieldElement {
43    pub const ZERO: FieldElement = FieldElement([0, 0, 0]);
44    pub const ONE: FieldElement = FieldElement([1, 0, 0]);

46    /// Builds a field element from a big-endian byte slice (up to 21 bytes / 163 bits). The
47    /// caller must ensure the value is already less than `2^163` - this does not reduce.
48    #[must_use]
49    pub fn from_be_bytes(bytes: &[u8]) -> Self {
50        let mut limbs = [0u64; 3];
51        for (i, &byte) in bytes.iter().rev().enumerate() {
52            let limb = i / 8;
53            let shift = (i % 8) * 8;
54            limbs[limb] |= u64::from(byte) << shift;
55        }
56        FieldElement(limbs)
57    }
58
59    /// Big-endian encoding, fixed at 21 bytes (163 bits, rounded up to a whole byte count).
60    #[must_use]
61    #[allow(clippy::cast_possible_truncation)] // deliberate: extracting one byte from a shifted limb
62    pub fn to_be_bytes(self) -> [u8; 21] {
63        let mut out = [0u8; 21];
64        for (i, byte) in out.iter_mut().rev().enumerate() {
65            let limb = i / 8;
66            let shift = (i % 8) * 8;
67            *byte = (self.0[limb] >> shift) as u8;
68        }
69        out
70    }

72    /// Dispatches to the hardware-`clmul` `poly_mul_wide_hw` below when the CPU supports it
73    /// (`std`-gated runtime detection, `docs/TASKS.md` T-198, same shape as `gf2m_wide`'s own
74    /// dispatch) and falls back to the portable bit-serial `poly_mul_wide` otherwise -
75    /// `no_std`/embedded builds, CPUs without the instruction, and (`not(kani)`) CBMC's own
76    /// symbolic execution all take this same unconditional software path, unchanged from before
77    /// this task. **This is the function `curve163::scalar_multiply` calls on its own
78    /// secret-scalar intermediates** (the signing nonce, the private key) - the hardware path was
79    /// chosen over a software comb-method rewrite specifically because it has no secret-indexed
80    /// memory access at all (see `poly_mul_wide_hw`'s own doc comment and T-196's revert), so this
81    /// dispatch does not reopen that question.
82    #[must_use]
83    pub fn multiply(self, other: Self) -> Self {
84        #[cfg(all(
85            feature = "std",
86            not(kani),
87            any(target_arch = "x86_64", target_arch = "aarch64")
88        ))]
89        if crate::hazmat::gf2m_wide::clmul_native::feature_available() {
90            // Safety: `feature_available()` just confirmed the CPU supports the target feature
91            // `poly_mul_wide_hw` requires.
92            let wide = unsafe { poly_mul_wide_hw(&self.0, &other.0) };
93            return reduce(wide);
94        }
95        reduce(poly_mul_wide(&self.0, &other.0))
96    }
97
98    #[must_use]
99    pub fn square(self) -> Self {
100        reduce(square_wide(&self.0))
101    }

103    /// `self^-1 = self^(2^163 - 2)`, by Fermat's little theorem for `GF(2^163)*`. Undefined for
104    /// `self == ZERO` - this returns `ZERO` in that case (Fermat's formula itself gives `0^k = 0`
105    /// for any positive `k`), not a panic, but that value is not a meaningful inverse and callers
106    /// must not treat it as one. Ordinarily callers should never invert zero at all (the DSTU 4145
107    /// sign/verify pseudocode's own retry loops exist precisely to avoid producing a zero value
108    /// that would need inverting) - the one documented exception is
109    /// `curve163::scalar_multiply`'s constant-time y-recovery step (`docs/DECISIONS.md` D-110,
110    /// T-152), which still calls this on a possibly-zero value (to stay branchless) but explicitly
111    /// masks the resulting corrupted output afterward rather than trusting it.
112    ///
113    /// `2^163 - 2 = 2*(2^162 - 1)`, so this computes `(self^(2^162 - 1))^2`. `self^(2^162 - 1)` is
114    /// built via an Itoh-Tsujii-style addition chain, using repeated application of
115    /// `T_(i+j) = T_i^(2^j) * T_j` (where `T_k` denotes `self^(2^k - 1)`) over the derived chain
116    /// `1 -> 2 -> 3 -> 6 -> 12 -> 24 -> 27 -> 54 -> 81 -> 162` (`162 = 2*81 = 2*(80+1)`) - **9**
117    /// combine steps (each one multiply, plus the fixed number of squarings the `2^j` part costs)
118    /// instead of the previous direct form's 162 multiplies, needing the same ~162 total squarings
119    /// either way (squaring does not become free in this polynomial-basis representation, only the
120    /// multiply count drops - `docs/DECISIONS.md` D-109, `docs/TASKS.md` T-153). The chain is a
121    /// fixed, public sequence over a fixed, public exponent, identical for every call regardless of
122    /// `self`'s value - the same constant-time argument that already justified the prior
123    /// fixed-iteration direct form. Verified against that direct form (kept as a test-only oracle,
124    /// `invert_direct` below) by differential test, not by the chain's derivation alone.
125    #[must_use]
126    #[allow(clippy::similar_names)] // deliberate: t12/t162 etc. are the standard T_k = self^(2^k-1) notation
127    pub fn invert(self) -> Self {
128        // `sq_n(x, n) = x^(2^n)`, i.e. `n` repeated squarings.
129        let sq_n = |mut x: Self, n: u32| -> Self {
130            for _ in 0..n {
131                x = x.square();
132            }
133            x
134        };
135
136        let t1 = self; // self^(2^1 - 1)
137        let t2 = t1.square().multiply(t1); // self^(2^2 - 1)
138        let t3 = sq_n(t2, 1).multiply(t1); // self^(2^3 - 1)
139        let t6 = sq_n(t3, 3).multiply(t3); // self^(2^6 - 1)
140        let t12 = sq_n(t6, 6).multiply(t6); // self^(2^12 - 1)
141        let t24 = sq_n(t12, 12).multiply(t12); // self^(2^24 - 1)
142        let t27 = sq_n(t24, 3).multiply(t3); // self^(2^27 - 1)
143        let t54 = sq_n(t27, 27).multiply(t27); // self^(2^54 - 1)
144        let t81 = sq_n(t54, 27).multiply(t27); // self^(2^81 - 1)
145        let t162 = sq_n(t81, 81).multiply(t81); // self^(2^162 - 1)
146
147        t162.square() // self^(2^163 - 2)
148    }
149}

151/// Binary-polynomial (carry-less) multiplication of two 163-bit operands into a 6-limb (384-bit
152/// capacity, up to 325 significant bits) product - the right-to-left shift-and-add method
153/// (`Guide to Elliptic Curve Cryptography`, Hankerson/Menezes/Vanstone, Algorithm 2.33), written
154/// with a branchless bit-select (`mask`) in place of the algorithm's `if a_i = 1` step.
155fn poly_mul_wide(a: &[u64; 3], b: &[u64; 3]) -> [u64; 6] {
156    let mut acc = [0u64; 6];
157    let mut shifted = [b[0], b[1], b[2], 0u64, 0u64, 0u64];
158
159    for bit_index in 0..163u32 {
160        let limb = (bit_index / 64) as usize;
161        let bit = bit_index % 64;
162        let bit_value = (a[limb] >> bit) & 1;
163        let mask = 0u64.wrapping_sub(bit_value); // all-ones if the bit is 1, all-zeros otherwise
164        for i in 0..6 {
165            acc[i] ^= shifted[i] & mask;
166        }
167        shl1(&mut shifted);
168    }
169
170    acc
171}

173/// Hardware carry-less-multiply `poly_mul_wide` replacement (`docs/TASKS.md` T-198,
174/// `docs/DECISIONS.md` D-184) - schoolbook combination of 9 pairwise 64x64->128 hardware `clmul`s,
175/// the same limb-placement identity the bit-serial method above computes a different way
176/// (cross-checked directly in `hw_dispatch_tests` below). **Chosen over a software comb-method
177/// rewrite of `poly_mul_wide` itself** (implemented, tested, and reverted in T-196): a comb method
178/// needs a secret-indexed `T[nibble]` table lookup, which this module's own "Branchless by
179/// construction... no array indexing at all" design principle exists to rule out, since this
180/// function runs on `curve163::scalar_multiply`'s secret-scalar intermediates. Schoolbook hardware
181/// `clmul` has no such cost - `pclmulqdq`/`vmull_p64` is called for a fixed 9 `(i, j)` pairs
182/// unconditionally, and the instruction's own latency does not depend on its operand bits. Every
183/// intrinsic call sits directly inside this single `#[target_feature]`-attributed function (not
184/// behind `gf2m_wide::clmul_native::clmul64`'s separately-attributed wrapper, which the T-196 spike
185/// used and which is a real non-inlinable call boundary at every one of the 9 pairs,
186/// `advisor()`-flagged before landing this) - the spike's own ~64-65x/~42x numbers are a floor for
187/// this production shape, not a target.
188#[cfg(all(feature = "std", target_arch = "x86_64"))]
189#[target_feature(enable = "pclmulqdq")]
190#[allow(
191    clippy::cast_possible_wrap,
192    clippy::cast_possible_truncation,
193    clippy::cast_sign_loss
194)]
195// deliberate: `_mm_set_epi64x`/`_mm_cvtsi128_si64` use `i64` purely as a bit container (no signed
196// interpretation anywhere below) - the casts are the intended reinterpret, not a value-changing
197// truncation.
198unsafe fn poly_mul_wide_hw(a: &[u64; 3], b: &[u64; 3]) -> [u64; 6] {
199    use std::arch::x86_64::{
200        _mm_clmulepi64_si128, _mm_cvtsi128_si64, _mm_set_epi64x, _mm_srli_si128,
201    };
202    let mut out = [0u64; 6];
203    for i in 0..3 {
204        for j in 0..3 {
205            // Safety: this function itself requires `pclmulqdq` (target_feature, callers gate on
206            // `clmul_native::feature_available()` first). Extracting both 64-bit halves via
207            // `_mm_cvtsi128_si64`/`_mm_srli_si128` (both SSE2, already required for `__m128i` to
208            // exist on this target) avoids the unaligned-pointer-cast `_mm_storeu_si128` into a
209            // stack byte array would otherwise need.
210            let ma = _mm_set_epi64x(0, a[i] as i64);
211            let mb = _mm_set_epi64x(0, b[j] as i64);
212            let prod = _mm_clmulepi64_si128(ma, mb, 0x00);
213            let lo = _mm_cvtsi128_si64(prod) as u64;
214            let hi = _mm_cvtsi128_si64(_mm_srli_si128::<8>(prod)) as u64;
215            out[i + j] ^= lo;
216            out[i + j + 1] ^= hi;
217        }
218    }
219    out
220}

222/// `aarch64` sibling of the `x86_64` `poly_mul_wide_hw` above - see its own doc comment for the
223/// rationale (`docs/TASKS.md` T-198).
224#[cfg(all(feature = "std", target_arch = "aarch64"))]
225#[target_feature(enable = "aes")]
226unsafe fn poly_mul_wide_hw(a: &[u64; 3], b: &[u64; 3]) -> [u64; 6] {
227    use std::arch::aarch64::vmull_p64;
228    let mut out = [0u64; 6];
229    for i in 0..3 {
230        for j in 0..3 {
231            // Safety: this function itself requires `aes`/`PMULL` (target_feature, callers gate
232            // on `clmul_native::feature_available()` first).
233            let prod: u128 = vmull_p64(a[i], b[j]);
234            out[i + j] ^= prod as u64;
235            out[i + j + 1] ^= (prod >> 64) as u64;
236        }
237    }
238    out
239}

256/// Spreads the low 32 bits of `x` across the low 64 bits of the result, each bit `i` of `x`
257/// landing at bit `2*i` of the result with a zero bit inserted between every pair - the
258/// "interleave bits by binary magic numbers" technique (Sean Eron Anderson's Bit Twiddling Hacks,
259/// `graphics.stanford.edu/~seander/bithacks.html#InterleaveBMN`), widened from its usual
260/// 16-bit-to-32-bit form to 32-bit-to-64-bit by doubling every mask/shift constant the same way.
261/// This computes exactly the GF(2) squaring identity `a(x)^2 = a(x^2)` (char-2 cross terms vanish:
262/// `(a_i*x^i)^2 = a_i*x^(2i)` since `a_i` is 0 or 1), applied 32 bits at a time - a pure fixed
263/// shift/AND/OR bit-spread, no array indexing anywhere, so no D-19-adjacent secret-indexing
264/// question exists at all (`docs/DECISIONS.md` D-25's "branchless by construction" posture,
265/// extended to a fresh operation). Verified against the `poly_mul_wide` self-multiply oracle by
266/// the `square_wide_matches_multiply_wide_*` tests below, at both every single-bit position and
267/// the all-bits-set case, not derived from the citation alone.
268fn spread32to64(x: u32) -> u64 {
269    let mut x = u64::from(x);
270    x = (x | (x << 16)) & 0x0000_FFFF_0000_FFFF;
271    x = (x | (x << 8)) & 0x00FF_00FF_00FF_00FF;
272    x = (x | (x << 4)) & 0x0F0F_0F0F_0F0F_0F0F;
273    x = (x | (x << 2)) & 0x3333_3333_3333_3333;
274    x = (x | (x << 1)) & 0x5555_5555_5555_5555;
275    x
276}

278/// Squares a 163-bit operand into its 6-limb (up to 325-bit) wide product, via `spread32to64`
279/// instead of a full carry-less `poly_mul_wide` self-multiplication. Each input limb's low/high
280/// 32-bit halves spread independently into exactly one 64-bit output limb apiece, with no
281/// cross-limb interaction needed: limb `i`'s low 32 bits (global bit range `[64i, 64i+31]`) spread
282/// to global bit range `[128i, 128i+62]`, landing entirely within output limb `2i`; its high 32
283/// bits (global bit range `[64i+32, 64i+63]`) spread to `[128i+64, 128i+126]`, landing entirely
284/// within output limb `2i+1` - both fit with no shift needed at placement time.
285fn square_wide(a: &[u64; 3]) -> [u64; 6] {
286    let mut out = [0u64; 6];
287    for i in 0..3 {
288        #[allow(clippy::cast_possible_truncation)]
289        // deliberate: splitting a limb into its two halves
290        let lo = a[i] as u32;
291        #[allow(clippy::cast_possible_truncation)]
292        // deliberate: splitting a limb into its two halves
293        let hi = (a[i] >> 32) as u32;
294        out[2 * i] = spread32to64(lo);
295        out[2 * i + 1] = spread32to64(hi);
296    }
297    out
298}

300/// Left-shifts a 6-limb little-endian array by exactly 1 bit, in place.
301fn shl1(x: &mut [u64; 6]) {
302    let mut carry = 0u64;
303    for limb in x.iter_mut() {
304        let next_carry = *limb >> 63;
305        *limb = (*limb << 1) | carry;
306        carry = next_carry;
307    }
308}

310/// Reduces a 6-limb (up to 325-bit) product modulo `x^163 + x^7 + x^6 + x^3 + 1`, producing a
311/// fully-reduced 3-limb field element. Adapted from OpenSSL's generic `BN_GF2m_mod_arr`
312/// (`crypto/bn/bn_gf2m.c`) specialized to `m = 163`, `W = 64` (so `dN = m / W = 2`,
313/// `d0 = m % W = 35`, `d1 = W - d0 = 29`) - with its two data-dependent shortcuts removed for
314/// constant-time behavior (`docs/DECISIONS.md` D-25):
315/// - the source removes a source word entirely if it happens to be zero; this always processes
316///   every word.
317/// - the source loops the final cleanup step until the overflow is zero; this always runs it a
318///   fixed 2 times, which is provably sufficient (see below) and a harmless no-op past that point,
319///   since re-XORing an already-zero overflow changes nothing.
320fn reduce(mut c: [u64; 6]) -> FieldElement {
321    // Main pass: reduce words 5, 4, 3 (each covering exponents >= 163) down into words 0..=3.
322    // Each source word `zz` contributes its middle-term (7, 6, 3) and constant-term (0)
323    // reductions, all of which land split across exactly two destination words `j-2`/`j-3` for
324    // this specific (m, W) pair - see docs/DECISIONS.md D-25 for the derivation of the shift amounts
325    // (28/36, 29/35, 32/32 for the three middle terms, 35/29 for the constant term).
326    for j in (3..=5).rev() {
327        let zz = c[j];
328        c[j] = 0;
329        c[j - 2] ^= (zz >> 28) ^ (zz >> 29) ^ (zz >> 32) ^ (zz >> 35);
330        c[j - 3] ^= (zz << 36) ^ (zz << 35) ^ (zz << 32) ^ (zz << 29);
331    }
332
333    // Final cleanup: word 2 (index dN = 2) may still have bits set above position 34 (global
334    // exponent 163+), left over from the main pass above. Each pass extracts those overflow bits
335    // (at most 29 of them, since a single 64-bit word can only overflow bit 35 by 64-35=29 bits),
336    // masks them out of c[2], and folds them into c[0] via the same z^163 = z^7+z^6+z^3+1
337    // identity. One pass is provably enough (c[2] is fully masked by the end of it, so a repeat
338    // pass reads an all-zero overflow and changes nothing) - run twice anyway as cheap insurance
339    // against a subtle off-by-one in that argument, per docs/DECISIONS.md D-25.
340    for _ in 0..2 {
341        let overflow = c[2] >> 35;
342        c[2] = (c[2] << 29) >> 29;
343        c[0] ^= overflow ^ (overflow << 3) ^ (overflow << 6) ^ (overflow << 7);
344    }
345
346    FieldElement([c[0], c[1], c[2]])
347}

455/// Kani pilot (see `docs/TASKS.md`/`docs/DECISIONS.md` for the tracked outcome): `reduce`'s doc
456/// comment above makes two claims that were previously only hand-argued, never machine-checked -
457/// "one pass is provably enough" and the implicit claim that the closed-form word-shift reduction
458/// actually computes the same result as the polynomial identity it's derived from. This proves
459/// both, for every one of the 2^384 possible 6-limb inputs, not just the fixed vectors and hand-
460/// picked property tests `dstu4145_gf2m.rs` already covers.
461#[cfg(kani)]
462mod kani_proofs {
463    use super::*;
464
465    /// Bit-at-a-time reference reduction, written directly from the polynomial identity
466    /// `x^163 = x^7 + x^6 + x^3 + 1` (mod the field polynomial) with no word-level shortcuts -
467    /// the same technique `gf2m_wide::reduce` uses in production, here kept only as an
468    /// independent oracle for this proof, not as a second implementation to maintain.
469    fn naive_reduce(mut c: [u64; 6]) -> FieldElement {
470        for degree in (163u32..384).rev() {
471            let limb = (degree / 64) as usize;
472            let bit = degree % 64;
473            if (c[limb] >> bit) & 1 == 1 {
474                c[limb] ^= 1u64 << bit;
475                let shift = degree - 163;
476                for term in [7u32, 6, 3, 0] {
477                    let d = shift + term;
478                    let l = (d / 64) as usize;
479                    let b = d % 64;
480                    c[l] ^= 1u64 << b;
481                }
482            }
483        }
484        let mut out = [0u64; 3];
485        out.copy_from_slice(&c[..3]);
486        FieldElement(out)
487    }
488
489    /// Cheaper of the two proofs: `reduce`'s output must always have its top 29 bits (of the
490    /// 35-bit-wide word 2) clear - i.e. genuinely `< 2^163`, not just "probably" per the doc
491    /// comment's hand-argued overflow bound.
492    #[kani::proof]
493    fn reduce_output_is_fully_reduced() {
494        let c: [u64; 6] = kani::any();
495        let r = reduce(c);
496        assert_eq!(r.0[2] >> 35, 0);
497    }
498
499    /// The expensive proof: `reduce` and `naive_reduce` must agree on every possible input, not
500    /// just the ones `dstu4145_gf2m.rs`'s fixed vectors happen to exercise.
501    #[kani::proof]
502    fn reduce_matches_naive_bit_loop() {
503        let c: [u64; 6] = kani::any();
504        assert_eq!(reduce(c), naive_reduce(c));
505    }
506
507    /// `spread32to64`'s exact bit-doubling specification (D-109/T-153, corrected by D-112):
508    /// every bit `i` of a symbolic 32-bit input lands at bit `2*i` of the output, and every other
509    /// output bit is zero - checked for all `2^32` possible inputs, not just the enumerated
510    /// single-bit/all-ones cases the external unit tests cover
511    /// (`spread32to64_places_each_bit_at_double_position`/`_of_zero_and_all_ones`).
512    ///
513    /// **This replaces an earlier, different proof that CI found intractable** (`docs/DECISIONS.md`
514    /// D-112): the original plan compared `square_wide(&a)` against `poly_mul_wide(&a, &a)` for a
515    /// symbolic `a` - i.e. proving two different multiplication constructions agree on the *same*
516    /// symbolic operand squared against itself. That is a multiplier-equivalence-checking problem
517    /// (SAT reasoning over products of two copies of the same symbolic bits), a fundamentally
518    /// harder class than `reduce`'s two proofs above or this one - both of which are pure fixed
519    /// shift/AND/OR/XOR over a symbolic input, with no product of two symbolic operands anywhere.
520    /// CBMC never finished checking that harness within the CI job's 20-minute budget (confirmed
521    /// from the job log, not assumed from the timeout alone: `Checking harness ...
522    /// square_wide_matches_poly_mul_wide_self...` was the last line before the runner killed it).
523    ///
524    /// This proof instead exhaustively covers the one genuinely novel piece of arithmetic
525    /// (`spread32to64`'s bit-interleave trick) directly against its own specification, with no
526    /// multiplication involved at all. `square_wide`'s limb-placement composition (which half of
527    /// which input limb lands at which output limb) is *not* re-proven exhaustively here - it's a
528    /// simple, inspectable placement of three `spread32to64` calls (see `square_wide`'s own doc
529    /// comment for the exact bit-range argument), verified by the existing limb-boundary unit tests
530    /// and the random-element proptest in `dstu4145_gf2m.rs` instead, the same "differential test is
531    /// the real proof, Kani only for the tractable subset" split this project already uses for
532    /// `invert()`'s own addition chain (never Kani-attempted, for the analogous reason: chaining
533    /// several multiplies is exactly the kind of computation this class of proof doesn't handle
534    /// cheaply).
535    #[kani::proof]
536    fn spread32to64_is_exact_bit_doubling() {
537        let x: u32 = kani::any();
538        let r = spread32to64(x);
539        for i in 0..32u32 {
540            let bit = u64::from((x >> i) & 1);
541            let placed = (r >> (2 * i)) & 1;
542            assert_eq!(placed, bit);
543            let odd = (r >> (2 * i + 1)) & 1;
544            assert_eq!(odd, 0);
545        }
546    }
547}

550/// T-196 Tier 1 spike - see `poly_mul_wide_hw`'s own doc comment above `multiply()` for the
551/// comb-method-vs-`clmul` security reasoning). Uses `gf2m_wide::clmul_native::{clmul64,
552/// feature_available}` only for `feature_available()` and (test-only) the differential oracle
553/// `clmul64` provides - the production dispatch itself calls `poly_mul_wide_hw` directly, not
554/// through this module.
555#[cfg(all(test, any(target_arch = "x86_64", target_arch = "aarch64")))]
556mod clmul_spike {
557    use super::{multiply_sw, poly_mul_wide, FieldElement};
558    use crate::hazmat::gf2m_wide::clmul_native::{clmul64, feature_available};
559    use proptest::prelude::*;

561    /// Schoolbook (not Karatsuba - checkable limb-by-limb) combination of 9 pairwise hardware
562    /// 64x64->128 carry-less multiplies via the test-only `clmul64` oracle - same limb-placement
563    /// identity `poly_mul_wide`'s own shift-and-add computes a different way, and the same
564    /// identity the production `poly_mul_wide_hw` computes with its intrinsic calls inlined
565    /// instead of routed through `clmul64`. Cross-checked against both below.
566    fn schoolbook_clmul_poly_mul_wide(a: &[u64; 3], b: &[u64; 3]) -> [u64; 6] {
567        let mut out = [0u64; 6];
568        for i in 0..3 {
569            for j in 0..3 {
570                // Safety: every call site below checks `feature_available()` first.
571                let (lo, hi) = unsafe { clmul64(a[i], b[j]) };
572                out[i + j] ^= lo;
573                out[i + j + 1] ^= hi;
574            }
575        }
576        out
577    }
578
579    fn arb_narrow() -> impl Strategy<Value = [u64; 3]> {
580        prop::collection::vec(any::<u64>(), 3).prop_map(|v| {
581            let mut out = [0u64; 3];
582            out.copy_from_slice(&v);
583            // Stay below 2^163: constructors mask bit 163+, `poly_mul_wide` itself doesn't and
584            // relies on that invariant (see its own doc comment) - matches every real caller.
585            out[2] &= (1u64 << 35) - 1;
586            out
587        })
588    }
