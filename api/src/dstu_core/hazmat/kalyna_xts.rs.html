1//! Kalyna-XTS: DSTU 7624:2014 mode of operation #9 (indexed substitution / disk-sector mode) - the
2//! 10th and last official mode, closing out this crate's full DSTU 7624 mode-of-operation coverage
3//! (`docs/TASKS.md` T-96). Reuses [`super::gf2m_wide`]'s field arithmetic (same `f[]` reduction
4//! polynomials as [`super::kalyna_gcm`]/[`super::kalyna_gmac`], confirmed byte-for-byte identical
5//! in `dstu7624_init_xts` - no new field module needed). Cited to
6//! `oracles/uapki/library/uapkic/src/dstu7624.c`'s `encrypt_xts`/`decrypt_xts` (lines 3003-3141)
7//! and `dstu7624_init_xts` (lines 4089-4132). `docs/DECISIONS.md` D-58 has the full citation, the
8//! `InvalidLength` finding below, and the oracle-coverage breakdown.
9//!
10//! # Confidentiality only - and that is the *correct* choice here, not a compromise
11//!
12//! Unlike every confidentiality-only mode elsewhere in this crate (ECB/OFB/CBC/CFB/CTR), XTS's
13//! lack of built-in integrity is not a misuse trap to warn callers away from - it is the standard,
14//! recognized design for disk-sector encryption, where integrity is deliberately the filesystem
15//! layer's responsibility, not the cipher's (`docs/release-readiness.md`'s use-case table already
16//! states this explicitly for the "full-disk encryption" row). **Do not use this for a new design
17//! outside that specific case** without understanding why every other confidentiality-only mode in
18//! this crate carries a stronger warning than this one does.
19//!
20//! # Ciphertext stealing - transcribed as found, not simplified by analogy to textbook XTS-AES
21//!
22//! For a buffer whose length is not a multiple of the block size, the standard IEEE P1619/NIST SP
23//! 800-38E ciphertext-stealing trick applies: the last full block and the short final block are
24//! encrypted out of their "natural" order and swapped, so the final block's shortfall is padded
25//! with bytes borrowed from the other one instead of a padding scheme of its own. This module's
26//! `k`/`r`-indexed derivation (`k = buffer.len() / block_bytes`, `r = buffer.len() % block_bytes`)
27//! was re-derived by hand-tracing `encrypt_xts`/`decrypt_xts` against two different official
28//! vectors (`k = 1` and `k = 2`) to confirm it generalizes, rather than assumed from the textbook
29//! algorithm's shape - see D-58 for the full trace.
30//!
31//! # A real gap found in the reference, not ported: unchecked-length underflow
32//!
33//! `encrypt_xts`'s C source computes `loop_len = plain_size - block_len` using unsigned (`size_t`)
34//! arithmetic, with no guard for `plain_size < block_len` - such an input underflows to a huge
35//! value and the main loop would read/write far past the buffer. `decrypt_xts` has a partial guard
36//! (`plain_size < 2*block_len ? 0 : ...`) but at a different threshold, and does not save the
37//! encrypt side either. Matching this crate's established posture for this exact class of gap
38//! (`hazmat::kalyna_kw`'s non-aligned branch, D-55; `hazmat::kalyna_cfb`'s multi-call panic,
39//! resolved to a checked error, `docs/TASKS.md` T-101), `encrypt_in_place`/`decrypt_in_place` reject
40//! `buffer.len() < block_bytes` with [`XtsError::InvalidLength`] instead of inheriting the
41//! underflow - XTS's own ciphertext-stealing scheme requires at least one full block by
42//! construction, so this is not a scope cut relative to the construction's real domain, only a
43//! guard against feeding it something the construction was never meant to accept.
44
45use super::gf2m_wide::{Gf2m128, Gf2m256, Gf2m512};

47#[derive(Debug, Clone, Copy, PartialEq, Eq)]
48pub enum XtsError {
49    InvalidLength,
50}
51
52fn xor_block(buf: &mut [u8], other: &[u8]) {
53    for (b, o) in buf.iter_mut().zip(other) {
54        *b ^= *o;
55    }
56}

58macro_rules! kalyna_xts_variant {
59    ($name:ident, $expanded:ident, $key_bytes:literal, $block_bytes:literal, $gf:ty) => {
60        #[doc = concat!(
61            "XTS mode over [`super::kalyna::", stringify!($expanded), "`] - see the module doc ",
62            "comment for the citation, the ciphertext-stealing derivation, and the misuse note."
63        )]
64        pub struct $name {
65            key: super::kalyna::$expanded,
66        }
67
68        impl $name {
69            #[must_use]
70            pub fn new(key: &[u8; $key_bytes]) -> Self {
71                Self {
72                    key: super::kalyna::$expanded::new(key),
73                }
74            }
75
76            /// Encrypts `buffer` in place under `iv` (one block, the "data unit" tweak seed).
77            ///
78            /// # Errors
79            ///
80            /// Returns [`XtsError::InvalidLength`] if `buffer.len() < ` one block - XTS's
81            /// ciphertext-stealing scheme has no meaning below that length.
82            pub fn encrypt_in_place(
83                &self,
84                iv: &[u8; $block_bytes],
85                buffer: &mut [u8],
86            ) -> Result<(), XtsError> {
87                let n = buffer.len();
88                if n < $block_bytes {
89                    return Err(XtsError::InvalidLength);
90                }
91                let mut gamma = <$gf>::from_le_bytes(&self.key.encrypt_block(iv));
92
93                let k = n / $block_bytes;
94                let r = n % $block_bytes;
95
96                if r == 0 {
97                    let mut off = 0usize;
98                    while off < n {
99                        gamma = gamma.double();
100                        let block = &mut buffer[off..off + $block_bytes];
101                        xor_block(block, &gamma.to_le_bytes());
102                        let mut tmp = [0u8; $block_bytes];
103                        tmp.copy_from_slice(block);
104                        let enc = self.key.encrypt_block(&tmp);
105                        block.copy_from_slice(&enc);
106                        xor_block(block, &gamma.to_le_bytes());
107                        off += $block_bytes;
108                    }
109                } else {
110                    // Blocks 0..k-1 get sequential tweaks 1..k, encrypted normally in place.
111                    for i in 0..k {
112                        gamma = gamma.double();
113                        let off = i * $block_bytes;
114                        let block = &mut buffer[off..off + $block_bytes];
115                        xor_block(block, &gamma.to_le_bytes());
116                        let mut tmp = [0u8; $block_bytes];
117                        tmp.copy_from_slice(block);
118                        let enc = self.key.encrypt_block(&tmp);
119                        block.copy_from_slice(&enc);
120                        xor_block(block, &gamma.to_le_bytes());
121                    }
122
123                    let last_off = (k - 1) * $block_bytes;
124                    let tail_off = k * $block_bytes;
125
126                    let mut scratch = [0u8; $block_bytes];
127                    scratch.copy_from_slice(&buffer[last_off..last_off + $block_bytes]);
128
129                    let mut combined = [0u8; $block_bytes];
130                    combined[..r].copy_from_slice(&buffer[tail_off..tail_off + r]);
131                    combined[r..].copy_from_slice(&scratch[r..]);
132
133                    gamma = gamma.double();
134                    xor_block(&mut combined, &gamma.to_le_bytes());
135                    let enc = self.key.encrypt_block(&combined);
136                    combined = enc;
137                    xor_block(&mut combined, &gamma.to_le_bytes());
138
139                    buffer[last_off..last_off + $block_bytes].copy_from_slice(&combined);
140                    buffer[tail_off..tail_off + r].copy_from_slice(&scratch[..r]);
141                }
142
143                Ok(())
144            }
145
146            /// Decrypts `buffer` in place under `iv` - the exact inverse of [`Self::encrypt_in_place`].
147            ///
148            /// # Errors
149            ///
150            /// Returns [`XtsError::InvalidLength`] if `buffer.len() < ` one block.
151            pub fn decrypt_in_place(
152                &self,
153                iv: &[u8; $block_bytes],
154                buffer: &mut [u8],
155            ) -> Result<(), XtsError> {
156                let n = buffer.len();
157                if n < $block_bytes {
158                    return Err(XtsError::InvalidLength);
159                }
160                let mut gamma = <$gf>::from_le_bytes(&self.key.encrypt_block(iv));
161
162                let k = n / $block_bytes;
163                let r = n % $block_bytes;
164
165                if r == 0 {
166                    let mut off = 0usize;
167                    while off < n {
168                        gamma = gamma.double();
169                        let block = &mut buffer[off..off + $block_bytes];
170                        xor_block(block, &gamma.to_le_bytes());
171                        let mut tmp = [0u8; $block_bytes];
172                        tmp.copy_from_slice(block);
173                        let dec = self.key.decrypt_block(&tmp);
174                        block.copy_from_slice(&dec);
175                        xor_block(block, &gamma.to_le_bytes());
176                        off += $block_bytes;
177                    }
178                } else {
179                    for i in 0..(k - 1) {
180                        gamma = gamma.double();
181                        let off = i * $block_bytes;
182                        let block = &mut buffer[off..off + $block_bytes];
183                        xor_block(block, &gamma.to_le_bytes());
184                        let mut tmp = [0u8; $block_bytes];
185                        tmp.copy_from_slice(block);
186                        let dec = self.key.decrypt_block(&tmp);
187                        block.copy_from_slice(&dec);
188                        xor_block(block, &gamma.to_le_bytes());
189                    }
190
191                    gamma = gamma.double(); // tweak_k
192                    let gamma_k = gamma;
193                    let gamma_k_plus_1 = gamma.double();
194
195                    let last_off = (k - 1) * $block_bytes;
196                    let tail_off = k * $block_bytes;
197
198                    let mut combined = [0u8; $block_bytes];
199                    combined.copy_from_slice(&buffer[last_off..last_off + $block_bytes]);
200                    xor_block(&mut combined, &gamma_k_plus_1.to_le_bytes());
201                    combined = self.key.decrypt_block(&combined);
202                    xor_block(&mut combined, &gamma_k_plus_1.to_le_bytes());
203
204                    let mut reconstructed = [0u8; $block_bytes];
205                    reconstructed[..r].copy_from_slice(&buffer[tail_off..tail_off + r]);
206                    reconstructed[r..].copy_from_slice(&combined[r..]);
207                    xor_block(&mut reconstructed, &gamma_k.to_le_bytes());
208                    reconstructed = self.key.decrypt_block(&reconstructed);
209                    xor_block(&mut reconstructed, &gamma_k.to_le_bytes());
210
211                    buffer[last_off..last_off + $block_bytes].copy_from_slice(&reconstructed);
212                    buffer[tail_off..tail_off + r].copy_from_slice(&combined[..r]);
213                }
214
215                Ok(())
216            }
217        }
218    };
219}
