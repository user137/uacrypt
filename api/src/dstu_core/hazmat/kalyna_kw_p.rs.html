1//! Kalyna-KW-p: DSTU 7624:2014 §15.4/15.5, key wrap with padding, for a message whose length is not
2//! a multiple of the block (T-253, `docs/DECISIONS.md` D-241/D-243). A block-aligned message is
3//! plain KW's case (§15.1), [`super::kalyna_kw`].
4//!
5//! Wrap (§15.4): `M || μ_{l/2}(|M|)` - the bit length as a half-block little-endian field - then
6//! annex Б.2's padding (a 1 bit, here the byte `0x80`, then zeros up to a block; Б.2 always pads,
7//! a full block when the data is already aligned), then KW (§15.2), which appends the all-zero
8//! checksum block. Unwrap (§15.5): KW, then Б.3 (drop the trailing zeros and the 1 bit), then the
9//! last half block must equal `μ_{l/2}` of the length that remains.
10//!
11//! # Deviations from the reference implementations, not from the text
12//!
13//! UAPKI's `padding()` pads only unaligned data, so for `|M| + l/2` a multiple of the block it
14//! produces a different ciphertext from §15.4 (this is the DSTU 9041 `l(p)=384` case, D-241); its
15//! `decrypt_kw` strips trailing zero bytes and never checks the length field. Bouncy Castle has no
16//! padded mode. This module follows the text.
17//!
18//! Unwrap is strict and constant-time over the recovered block: the 1 bit must be the byte `0x80`
19//! (this is a byte API), the length field must be exactly `μ_{l/2}(8·L)` with its unused top bytes
20//! zero, the padding must be the minimal Б.2 padding, and `L` must be a length `wrap` accepts
21//! (`L > l`, not a multiple of `l`). Any failure, the KW checksum's included, is the one
22//! [`KwError::ChecksumMismatch`], and `out` is zeroed: no padding oracle, one encoding per message.
23//!
24//! **Do not use this for new designs without a specific, understood reason** - the same raw key-wrap
25//! caveat as [`super::kalyna_kw`]. No heap allocation; inputs are bounded by KW's 20-block limit.
26

32/// Checks the padded block `p` (a multiple of `block` bytes, at least two blocks) against §15.4's
33/// layout without a branch on its bytes; returns whether it holds and the message length `L`.
34#[allow(clippy::cast_possible_truncation)] // every length here is <= MAX_R * 64 = 1280
35fn unpad(p: &[u8], block: usize, half: usize) -> (Choice, usize) {
36    let total = p.len() as u64;
37    let block = block as u64;
38    let half = half as u64;
39
40    // Б.3: the last nonzero byte holds the 1 bit.
41    let mut marker_at = 0u64;
42    let mut found = Choice::from(0);
43    for (i, &b) in p.iter().enumerate() {
44        let nonzero = !b.ct_eq(&0);
45        marker_at = u64::conditional_select(&marker_at, &(i as u64), nonzero);
46        found |= nonzero;
47    }
48    let mut marker = 0u8;
49    for (i, &b) in p.iter().enumerate() {
50        marker = u8::conditional_select(&marker, &b, (i as u64).ct_eq(&marker_at));
51    }
52    let len = marker_at.wrapping_sub(half);
53
54    let mut ok = found & marker.ct_eq(&0x80);
55    // L > l: the marker sits past the length field of a message longer than one block.
56    ok &= marker_at.ct_gt(&(half + block));
57    // L not a multiple of l (l is a power of two).
58    ok &= !(len & (block - 1)).ct_eq(&0);
59    // Minimal Б.2 padding: the marker is in the last block.
60    ok &= marker_at.ct_gt(&(total - block - 1));
61
62    // §15.5: the half block before the marker is μ_{l/2}(8·L), little-endian.
63    let expected_field = len.wrapping_mul(8).to_le_bytes();
64    for (i, &b) in p.iter().enumerate() {
65        let i = i as u64;
66        let in_field = !i.ct_lt(&len) & i.ct_lt(&marker_at);
67        let k = i.wrapping_sub(len);
68        let mut expected = 0u8;
69        for (j, &e) in expected_field.iter().enumerate() {
70            expected = u8::conditional_select(&expected, &e, k.ct_eq(&(j as u64)));
71        }
72        ok &= !in_field | b.ct_eq(&expected);
73    }
74    (ok, len as usize)
75}

77macro_rules! kalyna_kw_p_variant {
78    ($name:ident, $kw:ident, $expanded:ident, $key_bytes:literal, $block_bytes:literal, $half_bytes:literal) => {
79        #[doc = concat!(
80            "KW-p over [`super::kalyna::", stringify!($expanded), "`] - see the module doc for the ",
81            "citation, the strict unwrap and the misuse warning."
82        )]
83        #[derive(Debug)]
84        pub struct $name;
85
86        impl $name {
87            /// The wrapped length for a `plaintext_len`-byte message, or `None` if KW-p does not
88            /// take it: not longer than one block, a multiple of the block (plain KW's case), or
89            /// over KW's 20-block limit once padded.
90            #[must_use]
91            pub const fn wrapped_len(plaintext_len: usize) -> Option<usize> {
92                if plaintext_len <= $block_bytes || plaintext_len % $block_bytes == 0 {
93                    return None;
94                }
95                let padded = (plaintext_len + $half_bytes + 1).div_ceil($block_bytes) * $block_bytes;
96                if padded / $block_bytes > MAX_R {
97                    return None;
98                }
99                Some(padded + $block_bytes)
100            }
101
102            /// Wraps `plaintext` into `out`, which must be exactly
103            /// [`Self::wrapped_len`]`(plaintext.len())` long - expands the key schedule on every
104            /// call; prefer [`Self::wrap_with_cipher`] to wrap more than once under one key.
105            ///
106            /// # Errors
107            ///
108            /// [`KwError::InvalidLength`] if `wrapped_len` is `None` or `out` has another length.
109            pub fn wrap(
110                key: &[u8; $key_bytes],
111                plaintext: &[u8],
112                out: &mut [u8],
113            ) -> Result<(), KwError> {
114                let cipher = super::kalyna::$expanded::new(key);
115                Self::wrap_with_cipher(&cipher, plaintext, out)
116            }
117
118            /// [`Self::wrap`] with an already-expanded key schedule.
119            ///
120            /// # Errors
121            ///
122            /// Same cases as [`Self::wrap`].
123            pub fn wrap_with_cipher(
124                cipher: &super::kalyna::$expanded,
125                plaintext: &[u8],
126                out: &mut [u8],
127            ) -> Result<(), KwError> {
128                if Self::wrapped_len(plaintext.len()) != Some(out.len()) {
129                    return Err(KwError::InvalidLength);
130                }
131                let padded_len = out.len() - $block_bytes;
132                // Holds the key material being wrapped; wiped on every return path.
133                let mut padded = zeroize::Zeroizing::new([0u8; MAX_R * $block_bytes]);
134                padded[..plaintext.len()].copy_from_slice(plaintext);
135                let bits = (plaintext.len() as u64) * 8;
136                padded[plaintext.len()..plaintext.len() + 8].copy_from_slice(&bits.to_le_bytes());
137                padded[plaintext.len() + $half_bytes] = 0x80;
138                super::kalyna_kw::$kw::wrap_with_cipher(cipher, &padded[..padded_len], out)
139            }
140
141            /// Unwraps `ciphertext` into `out`, which must be exactly one block shorter than
142            /// `ciphertext`, and returns the message length `n`: the message is `out[..n]`, the
143            /// rest of `out` is zero. Expands the key schedule on every call; prefer
144            /// [`Self::unwrap_with_cipher`] to unwrap more than once under one key.
145            ///
146            /// # Errors
147            ///
148            /// [`KwError::InvalidLength`] if `ciphertext` is not a whole number of blocks, is under
149            /// three blocks or over KW's limit, or `out` has another length;
150            /// [`KwError::ChecksumMismatch`] for any integrity failure, checksum or padding alike,
151            /// with `out` zeroed.
152            pub fn unwrap(
153                key: &[u8; $key_bytes],
154                ciphertext: &[u8],
155                out: &mut [u8],
156            ) -> Result<usize, KwError> {
157                let cipher = super::kalyna::$expanded::new(key);
158                Self::unwrap_with_cipher(&cipher, ciphertext, out)
159            }
160
161            /// [`Self::unwrap`] with an already-expanded key schedule.
162            ///
163            /// # Errors
164            ///
165            /// Same cases as [`Self::unwrap`].
166            pub fn unwrap_with_cipher(
167                cipher: &super::kalyna::$expanded,
168                ciphertext: &[u8],
169                out: &mut [u8],
170            ) -> Result<usize, KwError> {
171                if ciphertext.len() < 3 * $block_bytes
172                    || !ciphertext.len().is_multiple_of($block_bytes)
173                    || ciphertext.len() / $block_bytes - 1 > MAX_R
174                    || out.len() != ciphertext.len() - $block_bytes
175                {
176                    return Err(KwError::InvalidLength);
177                }
178                let padded_len = out.len();
179                let mut padded = zeroize::Zeroizing::new([0u8; MAX_R * $block_bytes]);
180                if let Err(e) = super::kalyna_kw::$kw::unwrap_with_cipher(
181                    cipher,
182                    ciphertext,
183                    &mut padded[..padded_len],
184                ) {
185                    out.fill(0);
186                    return Err(e);
187                }
188                let (ok, len) = unpad(&padded[..padded_len], $block_bytes, $half_bytes);
189                for (i, (o, &p)) in out.iter_mut().zip(padded[..padded_len].iter()).enumerate() {
190                    *o = u8::conditional_select(&0, &p, (i as u64).ct_lt(&(len as u64)) & ok);
191                }
192                if bool::from(ok) {
193                    Ok(len)
194                } else {
195                    Err(KwError::ChecksumMismatch)
196                }
197            }
198        }
199    };
200}
