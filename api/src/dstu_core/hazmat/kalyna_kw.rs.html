1//! Kalyna-KW: DSTU 7624:2014 mode of operation #10 (key wrap) - a half-block Feistel-like network
2//! over `B` (one half-block accumulator) and a shifting queue of the remaining half-blocks, plus
3//! one appended all-zero "checksum" block. Cited to
4//! `oracles/uapki/library/uapkic/src/dstu7624.c`'s `encrypt_kw` (lines 3672-3755), `decrypt_kw`
5//! (lines 3812-3884), and `dstu7624_init_kw` (lines 3955-3969), cross-read against
6//! `oracles/bouncycastle-java/.../engines/DSTU7624WrapEngine.java` and
7//! `oracles/bouncycastle-dotnet/.../engines/Dstu7624WrapEngine.cs`. `docs/DECISIONS.md` D-55 has the
8//! full citation and two deliberate deviations from `dstu7624.c`, summarized here:
9//!
10//! # Deviation 1: block-aligned input only, no padding branch
11//!
12//! `dstu7624.c`'s `encrypt_kw` supports non-block-aligned input via an appended length field plus
13//! `0x80`-style padding, and `decrypt_kw` recovers it by scanning backward for the last nonzero
14//! byte through the appended checksum block - a heuristic that depends on the real plaintext's own
15//! last byte being nonzero, and can silently over-consume real data otherwise (a latent fragility
16//! in the reference C itself, not a transcription risk here - see D-55). Both Bouncy Castle ports
17//! reject non-aligned input outright (`DataLengthException`/`ArgumentException`). This module
18//! follows Bouncy Castle's restriction, matching every other raw mode in this crate
19//! ([`super::kalyna_cbc`], [`super::kalyna_cfb`]): **`wrap`/`unwrap` require block-aligned
20//! input and return [`KwError::InvalidLength`] otherwise - no padding scheme of this module's own.**
21//!
22//! # Deviation 2: added checksum verification on `unwrap`
23//!
24//! `dstu7624.c`'s `decrypt_kw` never checks that the recovered trailing block is actually
25//! all-zero; it returns whatever bytes result. Both Bouncy Castle ports do check (KW's only
26//! tamper-evidence mechanism) and reject otherwise. This module adds that check too
27//! (`subtle::ConstantTimeEq`, per `docs/SECURITY.md`'s constant-time-comparison rule - the checksum
28//! block is a function of secret key material through the whole Feistel network) - a deliberate,
29//! cited safety addition, not an omission.
30//!
31//! # The round-counter width fork - resolved by bounding input size, not by picking a side
32//!
33//! `dstu7624.c` XORs only the low byte of the round counter into the tweak position (`size_t i`
34//! implicitly truncated by assignment into a `uint8_t` slot); both Bouncy Castle ports XOR a full
35//! 4-byte little-endian encoding. The two conventions are **provably identical whenever the
36//! largest round counter used (`v`) is `<= 255`** (the encoding's upper 3 bytes are zero in that
37//! range, so `XOR`ing them is a no-op either way) - and genuinely unresolved above that, since no
38//! DSTU 7624:2014 primary text exists in this repo to break the tie, and Bouncy Castle's Java and
39//! .NET ports are the same lineage, not two independent readings (D-55). This module implements the
40//! 4-byte little-endian tweak and **hard-bounds input so `v` can never exceed 255**
41//! (`r <= MAX_R = 20`, since `v = 12r + 6`) - the fork is unreachable through this API by
42//! construction, not resolved by primary-source proof. `r <= 20` is generous for key-wrapping's
43//! actual purpose (up to 320/640/1280 bytes of key material depending on block size).
44//!
45//! # No integrity beyond the checksum, and no heap allocation
46//!
47//! **Do not use this for new designs without a specific, understood reason.** This is a raw
48//! key-wrap primitive, not a general-purpose AEAD - its only tamper-evidence is the checksum block
49//! above, with no associated data support. Operates entirely on caller-supplied buffers with
50//! fixed-size stack arrays (bounded by `MAX_R` above) - no `Vec`/`alloc`, matching
51//! [`super::kalyna_ccm`]'s no-heap-allocation precedent, the only other multi-block-buffer hazmat
52//! module in this crate.
53
54use subtle::ConstantTimeEq;
55
56#[derive(Debug, Clone, Copy, PartialEq, Eq)]
57pub enum KwError {
58    InvalidLength,
59    ChecksumMismatch,
60}
61
62const MAX_R: usize = 20;
63const MAX_B_SLOTS: usize = 2 * MAX_R + 1;
64
65macro_rules! kalyna_kw_variant {
66    ($name:ident, $expanded:ident, $key_bytes:literal, $block_bytes:literal, $half_bytes:literal) => {
67        #[doc = concat!(
68            "KW mode over [`super::kalyna::", stringify!($expanded), "`] - see the module doc ",
69            "comment for the citation, the two deliberate deviations from `dstu7624.c`, and the ",
70            "misuse warning."
71        )]
72        pub struct $name;
73
74        impl $name {
75            /// Wraps `plaintext` (block-aligned, `1..=MAX_R` blocks) into `out`, which must be
76            /// exactly one block longer than `plaintext` - expands the key schedule fresh on every
77            /// call. Prefer [`Self::wrap_with_cipher`] for a caller that wraps more than one
78            /// message under the same key.
79            ///
80            /// # Errors
81            ///
82            /// Returns [`KwError::InvalidLength`] if `plaintext` is empty, not a multiple of the
83            /// block size, longer than `MAX_R` blocks, or `out`'s length doesn't match.
84            pub fn wrap(
85                key: &[u8; $key_bytes],
86                plaintext: &[u8],
87                out: &mut [u8],
88            ) -> Result<(), KwError> {
89                let cipher = super::kalyna::$expanded::new(key);
90                Self::wrap_with_cipher(&cipher, plaintext, out)
91            }
92
93            /// Wraps `plaintext` using an already-expanded key schedule - the cached-schedule
94            /// counterpart to [`Self::wrap`]. `docs/DECISIONS.md` D-76 / `docs/TASKS.md` T-127: `wrap`
95            /// re-derives the full Kalyna round-key schedule on every call, which for KW's typically
96            /// small (`1..=MAX_R`-block) inputs is not amortized the way it is for a large message -
97            /// an avoidable cost for any caller wrapping more than one key under the same
98            /// wrapping key.
99            ///
100            /// # Errors
101            ///
102            /// Same cases as [`Self::wrap`].
103            pub fn wrap_with_cipher(
104                cipher: &super::kalyna::$expanded,
105                plaintext: &[u8],
106                out: &mut [u8],
107            ) -> Result<(), KwError> {
108                if plaintext.is_empty()
109                    || !plaintext.len().is_multiple_of($block_bytes)
110                    || plaintext.len() / $block_bytes > MAX_R
111                    || out.len() != plaintext.len() + $block_bytes
112                {
113                    return Err(KwError::InvalidLength);
114                }
115
116                let r = plaintext.len() / $block_bytes;
117                let n = 2 * (r + 1);
118                let v = (n - 1) * 6;
119
120                let mut big_b = [0u8; $half_bytes];
121                big_b.copy_from_slice(&plaintext[..$half_bytes]);
122
123                let mut b = [[0u8; $half_bytes]; MAX_B_SLOTS];
124                let mut off = $half_bytes;
125                let mut idx = 0;
126                while off < plaintext.len() {
127                    b[idx].copy_from_slice(&plaintext[off..off + $half_bytes]);
128                    idx += 1;
129                    off += $half_bytes;
130                }
131                // Remaining slots up to index n - 2 stay zero - the appended checksum block.
132
133                for i in 1..=v {
134                    let mut block = [0u8; $block_bytes];
135                    block[..$half_bytes].copy_from_slice(&big_b);
136                    block[$half_bytes..].copy_from_slice(&b[0]);
137                    let r_block = cipher.encrypt_block(&block);
138
139                    let mut tweaked = r_block;
140                    #[allow(clippy::cast_possible_truncation)] // i <= v <= 246, fits u32 trivially
141                    let tweak = (i as u32).to_le_bytes();
142                    for k in 0..4 {
143                        tweaked[$half_bytes + k] ^= tweak[k];
144                    }
145                    big_b.copy_from_slice(&tweaked[$half_bytes..]);
146
147                    for j in 0..(n - 2) {
148                        b[j] = b[j + 1];
149                    }
150                    b[n - 2].copy_from_slice(&r_block[..$half_bytes]);
151                }
152
153                out[..$half_bytes].copy_from_slice(&big_b);
154                let mut off = $half_bytes;
155                for j in 0..(n - 1) {
156                    out[off..off + $half_bytes].copy_from_slice(&b[j]);
157                    off += $half_bytes;
158                }
159                Ok(())
160            }
161
162            /// Unwraps `ciphertext` (block-aligned, at least 2 blocks) into `out`, which must be
163            /// exactly one block shorter than `ciphertext` - expands the key schedule fresh on
164            /// every call. Prefer [`Self::unwrap_with_cipher`] for a caller that unwraps more than
165            /// one message under the same key.
166            ///
167            /// # Errors
168            ///
169            /// Returns [`KwError::InvalidLength`] under the same-shaped conditions as
170            /// [`Self::wrap`] (mirrored for `ciphertext`/`out`), or
171            /// [`KwError::ChecksumMismatch`] if the recovered trailing checksum block isn't
172            /// all-zero (see the module doc comment's "Deviation 2").
173            pub fn unwrap(
174                key: &[u8; $key_bytes],
175                ciphertext: &[u8],
176                out: &mut [u8],
177            ) -> Result<(), KwError> {
178                let cipher = super::kalyna::$expanded::new(key);
179                Self::unwrap_with_cipher(&cipher, ciphertext, out)
180            }
181
182            /// Unwraps `ciphertext` using an already-expanded key schedule - the cached-schedule
183            /// counterpart to [`Self::unwrap`]. Same rationale as [`Self::wrap_with_cipher`].
184            ///
185            /// # Errors
186            ///
187            /// Same cases as [`Self::unwrap`].
188            pub fn unwrap_with_cipher(
189                cipher: &super::kalyna::$expanded,
190                ciphertext: &[u8],
191                out: &mut [u8],
192            ) -> Result<(), KwError> {
193                if ciphertext.len() < 2 * $block_bytes
194                    || !ciphertext.len().is_multiple_of($block_bytes)
195                    || (ciphertext.len() / $block_bytes - 1) > MAX_R
196                    || out.len() != ciphertext.len() - $block_bytes
197                {
198                    return Err(KwError::InvalidLength);
199                }
200
201                let r = ciphertext.len() / $block_bytes - 1;
202                let n = 2 * (r + 1);
203                let v = (n - 1) * 6;
204
205                let mut big_b = [0u8; $half_bytes];
206                big_b.copy_from_slice(&ciphertext[..$half_bytes]);
207
208                let mut b = [[0u8; $half_bytes]; MAX_B_SLOTS];
209                let mut off = $half_bytes;
210                for idx in 0..(n - 1) {
211                    b[idx].copy_from_slice(&ciphertext[off..off + $half_bytes]);
212                    off += $half_bytes;
213                }
214
215                for i in (1..=v).rev() {
216                    let last = b[n - 2];
217                    let mut block = [0u8; $block_bytes];
218                    block[..$half_bytes].copy_from_slice(&last);
219
220                    let mut tweaked_b = big_b;
221                    #[allow(clippy::cast_possible_truncation)] // i <= v <= 246, fits u32 trivially
222                    let tweak = (i as u32).to_le_bytes();
223                    for k in 0..4 {
224                        tweaked_b[k] ^= tweak[k];
225                    }
226                    block[$half_bytes..].copy_from_slice(&tweaked_b);
227
228                    let d = cipher.decrypt_block(&block);
229                    big_b.copy_from_slice(&d[..$half_bytes]);
230
231                    for j in (1..=(n - 2)).rev() {
232                        b[j] = b[j - 1];
233                    }
234                    b[0].copy_from_slice(&d[$half_bytes..]);
235                }
236
237                // Last two slots are the appended checksum block - verify all-zero before
238                // releasing anything into `out` (Deviation 2, module doc comment).
239                let zero = [0u8; $half_bytes];
240                let checksum_ok = b[n - 3].ct_eq(&zero) & b[n - 2].ct_eq(&zero);
241                if !bool::from(checksum_ok) {
242                    return Err(KwError::ChecksumMismatch);
243                }
244
245                out[..$half_bytes].copy_from_slice(&big_b);
246                let mut off = $half_bytes;
247                for j in 0..(n - 3) {
248                    out[off..off + $half_bytes].copy_from_slice(&b[j]);
249                    off += $half_bytes;
250                }
251                Ok(())
252            }
253        }
254    };
255}
