1//! Strumok stream cipher (DSTU 8845:2019), 256- and 512-bit key variants.
2//!
3//! Ported from `docs/pseudocode/strumok.md` (itself transcribed from the designers' paper,
4//! `docs/papers/Strumok.pdf` Sections 2-9) and structurally cross-checked against
5//! `oracles/strumok-dstu8845/strumok.c` (outspace, unofficial, no license) and
6//! `oracles/uapki/library/uapkic/src/dstu8845.c` (UAPKI, BSD-2-Clause, state-expertise pedigree -
7//! see `docs/ORACLES.md`). Citation and verification status: `docs/DECISIONS.md` D-18.
8//!
9//! The `T` nonlinear substitution (Section 7) is exactly one Kalyna/Kupyna round's `eta` (S-box)
10//! composed with `tau` (the MDS linear layer) applied to a single 64-bit word treated as an
11//! 8-byte column. Originally computed at runtime via `hazmat::tables::{SBOXES, MDS_MATRIX,
12//! apply_matrix}` (confirmed, by computing it that way and diffing all 2048 entries byte-for-byte
13//! against both oracles' precomputed `T0..T7` tables, that the two are identical). **Switched
14//! 2026-07-22 (`docs/DECISIONS.md` D-26) to using those same precomputed `T0..T7` tables directly**
15//! (transcribed from `oracles/strumok-dstu8845/strumok.c`, the same byte-for-byte cross-check
16//! already covers them) - a large, measured throughput win (`docs/PERFORMANCE.md`), since it replaces
17//! 8 S-box lookups plus a full `GF(2^8)` matrix-multiply with 8 table lookups XOR-ed together.
18//! **The `small-tables` feature (`docs/DECISIONS.md` D-35/D-38) reverts `t_function` to exactly this
19//! pre-D-26 runtime form** - `T0..T7`'s 16 KB isn't compiled at all in that profile.
20//!
21//! `MUL_ALPHA`/`MUL_ALPHA_INV` (Sections 8-9, multiplication by the LFSR feedback polynomial's
22//! generator in GF(2^64)) are **not** derivable from the Kalyna/Kupyna tables - they belong to a
23//! different field construction specific to Strumok's LFSR. Transcribed here from
24//! `oracles/uapki/.../dstu8845.c`'s `mul_T`/`invmul_T` (cross-checked byte-identical against
25//! `oracles/strumok-dstu8845/strumok.c`'s `strumok_alpha_mul`/`strumok_alphainv_mul` - same
26//! lineage per `docs/DECISIONS.md` D-15, so this confirms transcription accuracy, not independence).
27//!
28//! The state-transition function (`Next`, Section 4) originally shifted the 16-word state
29//! register on every step (`s.copy_within(1..16, 0)`) - mechanically checkable against the spec
30//! text's `Next(S_i, mode)` description, at the cost of a real 120-byte move every step. **Switched
31//! 2026-07-22 (D-26) to a ring buffer**: a `head` index into the same fixed-size array, advanced
32//! modulo 16 each step, with the one slot that logically "falls off" (`S[0]`) overwritten in place
33//! to become the new `S[15]` - the same technique `oracles/strumok-dstu8845/strumok.c`'s
34//! `next_stream()` uses (there, fully unrolled instead of index-based). No data movement at all,
35//! same output for the same input - see the module's existing test suite, unchanged by this.
36//!
37//! **Switched 2026-07-27 (`docs/TASKS.md` T-135, `docs/DECISIONS.md` D-86) to also batch-generate a full
38//! 128-byte (16-word) block per call** in `Core::apply_keystream`'s bulk phase, fusing the input
39//! XOR into the same pass at `u64` (word) granularity instead of one byte at a time - closing most
40//! of the remaining gap to `next_stream_full_crypt`'s equivalent path left open after D-26. See
41//! `next_block`'s own doc for why this uses a one-time array rotation instead of `next_step`'s
42//! per-step masked `head` indexing, and why not the 16-way const-generic dispatch `hazmat::kalyna`/
43//! `kupyna` use for their own analogous unrolling (`T-128`/`T-134`).
44//!
45//! Only raw keystream generation/XOR is provided here - no key/IV management beyond what
46//! `Init` (Section 3) requires, no higher-level nonce or AEAD construction (that is
47//! `crypto_stream`'s job, see `docs/dstu-crypto-project.md` "Concrete API shape").
48//!
49//! # Warning: never reuse the same key+IV pair for two different messages
50//!
51//! This is inherent to every stream cipher, not a bug here: two messages `XOR`-ed with the *same*
52//! keystream (same key, same IV) let an attacker recover `plaintext_a XOR plaintext_b` directly
53//! from `ciphertext_a XOR ciphertext_b`, with no key material needed at all (the classic "two-time
54//! pad" break). `apply_keystream`/`Strumok256`/`Strumok512` have no state to detect or prevent
55//! this - a fresh IV per message is entirely the caller's responsibility. See
56//! `tests/strumok.rs`'s `reusing_key_and_iv_leaks_plaintext_xor` for the property pinned directly,
57//! the same pattern `hazmat::kalyna_gcm`'s nonce-coverage warning uses (D-56/D-63).

765/// `mul_alpha(w) = (w << 8) xor MUL_ALPHA[top byte of w]` (`docs/pseudocode/strumok.md` Sections

791/// the shared Kalyna/Kupyna S-box + MDS (`super::tables`) applied to `w` treated as one 8-byte
792/// column - exactly `T`'s original pre-D-26 form (see module doc), before it got its own
793/// precomputed tables.

799    }
800    super::tables::apply_forward_matrix(core::slice::from_mut(&mut column));

802}

862/// `Next(S_i, mode)` (`docs/pseudocode/strumok.md` Section 4), implemented as a ring buffer - see
863/// module doc. `head` is the physical index of logical `S[0]`; logical `S[k]` lives at physical
864/// index `(head + k) & 15`.
865fn next_step(s: &mut [u64; 16], head: &mut usize, r0: &mut u64, r1: &mut u64, init_mode: bool) {
866    let s0 = s[*head];
867    let s11 = s[(*head + 11) & 15];
868    let s13 = s[(*head + 13) & 15];
869    let s15 = s[(*head + 15) & 15];
870
871    let new_r1 = t_function(*r0);
872    let new_r0 = r1.wrapping_add(s13);
873
874    let mut feedback = mul_alpha(s0) ^ mul_alpha_inv(s11) ^ s13;
875    if init_mode {
876        feedback ^= fsm(s15, *r0, *r1);
877    }
878
879    // The slot holding old S[0] is exactly the slot that becomes new S[15] once head advances -
880    // no other slot moves.
881    s[*head] = feedback;
882    *head = (*head + 1) & 15;
883    *r0 = new_r0;
884    *r1 = new_r1;
885}

892/// Batched, fixed-index equivalent of 16 back-to-back `strm`+`next_step(init_mode: false)` calls
893/// starting from `head == 0`, fused with the input XOR - `docs/TASKS.md` T-135, `docs/DECISIONS.md` D-86.
894///
895/// **Requires `head == 0` on entry** (the caller rotates `s` once via `[u64; 16]::rotate_left`
896/// before any run of batch calls - see `Core::apply_keystream`'s bulk phase; a full 16-step batch
897/// is always a net-zero rotation, so `head` stays `0` for every subsequent batch in the same run).
898/// Chosen over a 16-way const-generic dispatch on `head` (the `hazmat::kalyna`/`kupyna`
899/// `T-128`/`T-134` pattern) specifically for code size: 16 fully-unrolled monomorphizations of a
900/// 16-step function is a different order of magnitude than those modules' 2-3 `const N`
901/// instantiations, and this project budgets flash down to 16-64 KB STM32 parts (`docs/resource-
902/// profiles.md`) and ships a whole `small-tables` feature to save 16 KB - see D-86 for the full
903/// tradeoff writeup.
904///
905/// Each unrolled step `k` (`0..16`) is mechanically derived from `strm`+`next_step` at
906/// `head = k`, **not** transcribed from `oracles/strumok-dstu8845/strumok.c`'s
907/// `next_stream_full_crypt` - that source computes its output *after* updating `S[i]`, using the
908/// already-advanced `r0`/`r1`; this project's `apply_keystream` calls `strm` (pre-step state)
909/// *before* `next_step`, and the batch below preserves that same evaluation order:
910///
911/// | `k` | reads `s[prev]` at | `p11` (`(k+11)&15`) | `p13` (`(k+13)&15`) |
912/// |---|---|---|---|
913/// | 0..15 | `(k+15)&15` | `(k+11)&15` | `(k+13)&15` |
914///
915/// (the table collapses to that one formula; the `step!` invocations below spell out each `k`'s
916/// three literal indices so they're auditable without re-deriving the modular arithmetic by hand,
917/// the same "kept diffable" rationale the module's table constants use). Correctness is not
918/// asserted from this derivation alone - `tests/strumok.rs`'s `batched_path_matches_scalar_*`
919/// proptest is the actual gate, checked against a kept-unchanged, purely-scalar reference copy of
920/// the pre-T-135 `apply_keystream`.
921fn next_block(s: &mut [u64; 16], r0: &mut u64, r1: &mut u64, input: &[u64; 16]) -> [u64; 16] {
922    let mut out = [0u64; 16];
923
924    macro_rules! step {
925        ($k:expr, $prev:expr, $p11:expr, $p13:expr) => {{
926            let sk = s[$k];
927            out[$k] = input[$k] ^ fsm(s[$prev], *r0, *r1) ^ sk;
928            let new_r1 = t_function(*r0);
929            let new_r0 = r1.wrapping_add(s[$p13]);
930            let feedback = mul_alpha(sk) ^ mul_alpha_inv(s[$p11]) ^ s[$p13];
931            s[$k] = feedback;
932            *r0 = new_r0;
933            *r1 = new_r1;
934        }};
935    }
936
937    step!(0, 15, 11, 13);
938    step!(1, 0, 12, 14);
939    step!(2, 1, 13, 15);
940    step!(3, 2, 14, 0);
941    step!(4, 3, 15, 1);
942    step!(5, 4, 0, 2);
943    step!(6, 5, 1, 3);
944    step!(7, 6, 2, 4);
945    step!(8, 7, 3, 5);
946    step!(9, 8, 4, 6);
947    step!(10, 9, 5, 7);
948    step!(11, 10, 6, 8);
949    step!(12, 11, 7, 9);
950    step!(13, 12, 8, 10);
951    step!(14, 13, 9, 11);
952    step!(15, 14, 10, 12);
953
954    out
955}

976    fn new(key: &[u8], iv: &[u8; 32]) -> Self {
977        let mut s = init_state(key, iv);
978        let mut head = 0usize;
979        let (mut r0, mut r1) = (0u64, 0u64);
980        for _ in 0..32 {
981            next_step(&mut s, &mut head, &mut r0, &mut r1, true);
982        }
983        next_step(&mut s, &mut head, &mut r0, &mut r1, false);
984        Self {
985            s,
986            head,
987            r0,
988            r1,
989            block: [0u8; 8],
990            block_pos: 8,
991        }
992    }
993
994    /// XORs the keystream into `data` in place (applying it to an all-zero buffer yields the raw
995    /// keystream, matching `crates/dstu-core/tests/vectors/strumok/keystream-{256,512}.json`).
996    ///
997    /// Three phases (`docs/TASKS.md` T-135, `docs/DECISIONS.md` D-86), mirroring
998    /// `oracles/strumok-dstu8845/strumok.c`'s `dstu8845_crypt` own `>=128`-bytes/remainder split,
999    /// without widening `block`/`block_pos` - so arbitrary chunk sizes and cross-call alignment
1000    /// (the `crypto_stream`/`uacrypt strumok-crypt` streaming use case) still work exactly as
1001    /// before. Split into one method per phase (`docs/DECISIONS.md` D-94) purely to bring each one's
1002    /// own Cognitive Complexity under `SonarCloud`'s threshold - same three phases, same math, not
1003    /// a behavior change; `drain`/`bulk`/`remainder` are single-caller private methods, expected
1004    /// (and verified, D-94) to still inline into one function the same way the un-split version
1005    /// did.
1006    fn apply_keystream(&mut self, data: &mut [u8]) {
1007        let pos = self.drain(data, 0);
1008        let pos = self.bulk(data, pos);
1009        self.remainder(data, pos);
1010    }
1011
1012    /// Phase 1: whatever's left in `block` from a previous call, byte-at-a-time, until the buffer
1013    /// is empty or `data` runs out.
1014    fn drain(&mut self, data: &mut [u8], mut pos: usize) -> usize {
1015        while pos < data.len() && self.block_pos < 8 {
1016            data[pos] ^= self.block[self.block_pos];
1017            self.block_pos += 1;
1018            pos += 1;
1019        }
1020        pos
1021    }
1022
1023    /// Phase 2, only once the block buffer is empty: rotate `s` so `head == 0` (at most once per
1024    /// call - see `next_block`'s doc), then run `next_block` over each full 128-byte chunk,
1025    /// XOR-ing a whole `u64` word at a time.
1026    fn bulk(&mut self, data: &mut [u8], mut pos: usize) -> usize {
1027        if self.block_pos != 8 || data.len() - pos < 128 {
1028            return pos;
1029        }
1030        if self.head != 0 {
1031            self.s.rotate_left(self.head);
1032            self.head = 0;
1033        }
1034        while data.len() - pos >= 128 {
1035            let mut input = [0u64; 16];
1036            for (word, chunk) in input.iter_mut().zip(data[pos..pos + 128].chunks_exact(8)) {
1037                let mut bytes = [0u8; 8];
1038                bytes.copy_from_slice(chunk);
1039                *word = u64::from_le_bytes(bytes);
1040            }
1041            let out = next_block(&mut self.s, &mut self.r0, &mut self.r1, &input);
1042            for (chunk, word) in data[pos..pos + 128].chunks_exact_mut(8).zip(out.iter()) {
1043                chunk.copy_from_slice(&word.to_le_bytes());
1044            }
1045            pos += 128;
1046        }
1047        pos
1048    }
1049
1050    /// Phase 3: the original per-byte scalar path for whatever's left (`< 128` bytes), refilling
1051    /// `block`/`block_pos` exactly as before.
1052    fn remainder(&mut self, data: &mut [u8], mut pos: usize) {
1053        while pos < data.len() {
1054            if self.block_pos == 8 {
1055                let z = strm(&self.s, self.head, self.r0, self.r1);
1056                self.block = z.to_le_bytes();
1057                next_step(
1058                    &mut self.s,
1059                    &mut self.head,
1060                    &mut self.r0,
1061                    &mut self.r1,
1062                    false,
1063                );
1064                self.block_pos = 0;
1065            }
1066            data[pos] ^= self.block[self.block_pos];
1067            self.block_pos += 1;
1068            pos += 1;
1069        }
1070    }
1071}

1146            let end = (pos + want).min(len);

1148            pos = end;
1149            idx += 1;
1150        }
