1//! Kalyna-GMAC: DSTU 7624:2014 mode of operation #7's MAC-only sibling - a single Horner-style
2//! GHASH-like accumulation over one message stream (no AAD/ciphertext split, no encryption), built
3//! on the same [`super::gf2m_wide`] field arithmetic [`super::kalyna_gcm`] uses. Cited to
4//! `oracles/uapki/library/uapkic/src/dstu7624.c`'s `encrypt_gmac` (lines 3572-3620) and
5//! `dstu7624_init_gmac` (lines 4015-4063). `docs/DECISIONS.md` D-57 has the full citation, the found
6//! reference bug (below), and the oracle-coverage breakdown.
7//!
8//! # Ported from `encrypt_gmac`, not `gmac_update`/`gmac_final` - a real bug found, not assumed
9//!
10//! `dstu7624.c` has **two** GMAC code paths, and they disagree on multi-block input given in one
11//! call. `encrypt_gmac`'s loop is a textbook Horner chain: for each padded block,
12//! `B = (B XOR block) * H`. The streaming pair `gmac_update`/`gmac_final` (reachable via
13//! `dstu7624_update_mac`/`dstu7624_final_mac`, what the self-test actually calls) has a stale-index
14//! bug instead: `kalyna_xor(&data_buf[i], B, block_len, B)` inside the post-multiply loop reuses the
15//! *current* loop index `i`, not `i + block_len` - so on a single call carrying more than one full
16//! block, later blocks' bytes are never read into the accumulator at all (confirmed by hand-tracing
17//! a 2-block input: block 2's content drops out of the final `B` entirely). Fed one block per
18//! `update` call instead, the same streaming code *does* reduce to the correct Horner chain
19//! (traced: first call leaves `B = block1*H`; second leaves `B = (block1*H XOR block2)*H`, matching
20//! `encrypt_gmac` exactly) - so this is a genuine single-call bug in the streaming path, not an
21//! intended alternate construction, and not something this module reproduces. This module ports
22//! `encrypt_gmac`'s coherent one-shot loop; see D-57 for why none of the 5 official vectors can
23//! settle this by themselves (every one is exactly one block long - `advisor()` caught this before
24//! any code was written).
25//!
26//! # One-shot, not streaming
27//!
28//! Same rationale as [`super::kalyna_cmac`]: nothing in this crate consumes an incremental MAC yet,
29//! and the only streaming code path available to transcribe from is the buggy one above.
30//!
31//! # Padding, and a length-block placement distinct from GCM's
32//!
33//! Non-block-aligned messages get the same `0x80`-then-zeros marker [`super::kalyna_cmac`]/
34//! [`super::kalyna_kw`]/[`super::kalyna_gcm`] already use (`padding()` in the C source) - unlike
35//! [`super::kalyna_kw`]'s block-aligned-only restriction, `encrypt_gmac`'s padding step is a
36//! self-contained one-shot allocation with no analogous OOB risk, so this module supports arbitrary
37//! lengths. The final length block holds the **padded** message bit-length (little-endian `u64`)
38//! always at a fixed low-8-byte offset, with every other byte zero - **not**
39//! [`super::kalyna_gcm`]'s two-value, half-block-offset-scaled layout (there is only one stream
40//! here, no AAD/ciphertext split to keep separate).
41//!
42//! # No key separation from any encryption key
43//!
44//! **Do not use this for new designs without a specific, understood reason.** Same misuse warning
45//! as [`super::kalyna_cmac`]: no key derivation or domain separation is provided by this API.
46
47use subtle::ConstantTimeEq;
48
49use super::gf2m_wide::{Gf2m128, Gf2m256, Gf2m512};
50
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub enum GmacError {
53    InvalidLength,
54    TagMismatch,
55}
56
57macro_rules! kalyna_gmac_variant {
58    ($name:ident, $expanded:ident, $key_bytes:literal, $block_bytes:literal, $gf:ty) => {
59        #[doc = concat!(
60            "GMAC over [`super::kalyna::", stringify!($expanded), "`] - see the module doc ",
61            "comment for the citation, the found reference bug, and the misuse warning."
62        )]
63        pub struct $name;
64
65        impl $name {
66            /// Computes the full-block-length GMAC tag of `message` under `key` - expands the key
67            /// schedule fresh on every call. Prefer [`Self::mac_with_cipher`] for a caller that
68            /// computes a MAC for more than one message under the same key. Callers truncate to their chosen `q`
69            /// (between 8 and one full block) themselves, per the source construction - matching
70            /// [`super::kalyna_gcm`]'s own tag-truncation convention.
71            #[must_use]
72            pub fn mac(key: &[u8; $key_bytes], message: &[u8]) -> [u8; $block_bytes] {
73                let cipher = super::kalyna::$expanded::new(key);
74                Self::mac_with_cipher(&cipher, message)
75            }
76
77            /// Computes the full-block-length GMAC tag of `message` using an already-expanded key
78            /// schedule - the cached-schedule counterpart to [`Self::mac`]. Same rationale as
79            /// [`super::kalyna_cmac`]'s own `mac_with_cipher` (`docs/DECISIONS.md` D-76 / `docs/TASKS.md`
80            /// T-127): `mac` re-derives the full Kalyna round-key schedule on every call, an
81            /// avoidable cost for any caller computing a MAC for more than one message under the same key.
82            #[must_use]
83            pub fn mac_with_cipher(
84                cipher: &super::kalyna::$expanded,
85                message: &[u8],
86            ) -> [u8; $block_bytes] {
87                let h_key = <$gf>::from_le_bytes(&cipher.encrypt_block(&[0u8; $block_bytes]));
88
89                let msg_len = message.len();
90                let rem = msg_len % $block_bytes;
91                let padded_len = if rem == 0 {
92                    msg_len
93                } else {
94                    msg_len + ($block_bytes - rem)
95                };
96
97                let mut acc = <$gf>::ZERO;
98                let mut off = 0usize;
99                while off < padded_len {
100                    let end = (off + $block_bytes).min(msg_len);
101                    let mut block = [0u8; $block_bytes];
102                    if end > off {
103                        block[..end - off].copy_from_slice(&message[off..end]);
104                    }
105                    if rem != 0 && msg_len >= off && msg_len < off + $block_bytes {
106                        block[msg_len - off] = 0x80;
107                    }
108                    acc = acc.add(<$gf>::from_le_bytes(&block)).multiply(h_key);
109                    off += $block_bytes;
110                }
111
112                let mut length_block = [0u8; $block_bytes];
113                #[allow(clippy::cast_possible_truncation)] // realistic lengths fit u64 trivially
114                let padded_len_bits = (padded_len as u64) * 8;
115                length_block[..8].copy_from_slice(&padded_len_bits.to_le_bytes());
116
117                let acc_bytes = acc.to_le_bytes();
118                let mut combined = [0u8; $block_bytes];
119                for i in 0..$block_bytes {
120                    combined[i] = length_block[i] ^ acc_bytes[i];
121                }
122
123                cipher.encrypt_block(&combined)
124            }
125
126            /// Recomputes the GMAC tag and compares its first `tag.len()` bytes against `tag` in
127            /// constant time (`subtle::ConstantTimeEq`, per `docs/SECURITY.md`'s hard constraint on
128            /// secret comparisons) - matching [`super::kalyna_gcm`]'s tag-verify discipline, and,
129            /// like it, a deliberate departure from `dstu7624.c`'s own raw `memcmp` (`docs/DECISIONS.md`
130            /// D-57).
131            ///
132            /// # Errors
133            ///
134            /// Returns [`GmacError::InvalidLength`] if `tag.len()` is outside 8 bytes to one full
135            /// block, or [`GmacError::TagMismatch`] if authentication fails.
136            pub fn verify(
137                key: &[u8; $key_bytes],
138                message: &[u8],
139                tag: &[u8],
140            ) -> Result<(), GmacError> {
141                let cipher = super::kalyna::$expanded::new(key);
142                Self::verify_with_cipher(&cipher, message, tag)
143            }
144
145            /// The cached-schedule counterpart to [`Self::verify`] - see
146            /// [`Self::mac_with_cipher`]'s doc comment for why this exists.
147            ///
148            /// # Errors
149            ///
150            /// Returns [`GmacError::InvalidLength`] if `tag.len()` is outside 8 bytes to one full
151            /// block, or [`GmacError::TagMismatch`] if authentication fails.
152            pub fn verify_with_cipher(
153                cipher: &super::kalyna::$expanded,
154                message: &[u8],
155                tag: &[u8],
156            ) -> Result<(), GmacError> {
157                if !(8..=$block_bytes).contains(&tag.len()) {
158                    return Err(GmacError::InvalidLength);
159                }
160                let expected = Self::mac_with_cipher(cipher, message);
161                if bool::from(expected[..tag.len()].ct_eq(tag)) {
162                    Ok(())
163                } else {
164                    Err(GmacError::TagMismatch)
165                }
166            }
167        }
168    };
169}
