1//! The passphrase file (`docs/DECISIONS.md` D-219, moved from `uacrypt` into the core by T-269): a
2//! 28-byte header in front of an ordinary stream file (D-208), so `uacrypt encrypt --passphrase`
3//! and every binding read and write the same bytes.
4//!
5//! ```text
6//! 0x50 || 0x01 (container version) || 0x01 (Argon2id v1.3)
7//!   || m_cost KiB (u32 LE) || t_cost (u32 LE) || p_cost (u8) || salt (16) || <stream file, D-208>
8//! ```
9//!
10//! The stream key is [`derive_key`]`(passphrase, salt, Moderate)`. The header carries no tag of
11//! its own: a changed salt gives another key and the first record fails, and a changed marker,
12//! version, KDF id or cost is refused by name. Only the costs [`seal_stream`] writes are accepted,
13//! and that check happens in [`Header::parse`], before any Argon2 memory is allocated - a file
14//! cannot make its reader allocate 1 GiB. No oracle exists for the header: it is this project's
15//! own format; [`derive_key`] itself equals libsodium's `crypto_pwhash`.
16//!
17//! ```rust,no_run
18//! use dstu_core::crypto_pwhash::passphrase_file::{seal_stream, Header, HEADER_BYTES};
19//!
20//! let (mut enc, prefix) = seal_stream(b"correct horse battery staple")?;
21//! let mut file = prefix.to_vec();
22//! let mut record = [0u8; dstu_core::crypto_secretstream::MAX_RECORD_BYTES];
23//! let n = enc.seal_final(b"short message", &mut record).expect("fits in one record");
24//! file.extend_from_slice(&record[..n]);
25//!
26//! let header = Header::parse(file[..HEADER_BYTES].try_into().expect("28 bytes"))?;
27//! let decryptor = header.open_stream(b"correct horse battery staple")?;
28//! // feed `decryptor` everything after the header (the stream-file codec's reader contract)
29//! # let _ = decryptor;
30//! # Ok::<(), dstu_core::crypto_pwhash::passphrase_file::PassphraseFileError>(())
31//! ```

38/// First byte of a passphrase file. A key-encrypted stream file starts with its own format
39/// version (`2`) instead, so a reader can tell the two apart from byte 0.
40pub const CONTAINER_TYPE: u8 = 0x50;
41const CONTAINER_VERSION: u8 = 0x01;
42const KDF_ARGON2ID13: u8 = 0x01;
43const PREFIX_LEN: usize = 3 + 4 + 4 + 1;
44/// Length of the header in front of the stream file.
45pub const HEADER_BYTES: usize = PREFIX_LEN + SALT_BYTES;
46/// The bytes [`seal_stream`] returns to write first: the header, then the stream file's own
47/// version byte and header.
48pub const SEAL_STREAM_PREFIX_BYTES: usize = HEADER_BYTES + FILE_PREFIX_BYTES;
49const _: () = assert!(HEADER_BYTES == 28);
50
51/// The one preset written and accepted (D-219's F3: libsodium `MODERATE`, 256 MiB, t=3).
52const STRENGTH: Strength = Strength::Moderate;
53
54/// Why a passphrase file cannot be written or opened. A wrong passphrase is not here: it shows up
55/// as the stream's first record failing ([`FileError::Auth`]).
56#[derive(Debug)]
57pub enum PassphraseFileError {
58    /// Byte 0 is not [`CONTAINER_TYPE`] - for example a key-encrypted stream file.
59    NotPassphraseFile,
60    /// Another container version or key-derivation function than this build writes.
61    UnsupportedFormat { version: u8, kdf: u8 },
62    /// Argon2 costs other than the one preset this format uses; refused before Argon2 runs.
63    UnsupportedParams {
64        m_cost: u32,
65        t_cost: u32,
66        p_cost: u8,
67    },
68    /// An empty passphrase - refused on both sides.
69    EmptyPassphrase,
70    /// The salt's CSPRNG, or Argon2 itself (its memory), failed.
71    PwHash(PwHashError),
72}
73
74impl fmt::Display for PassphraseFileError {
75    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
76        match self {
77            PassphraseFileError::NotPassphraseFile => write!(f, "not a passphrase file"),
78            PassphraseFileError::UnsupportedFormat { version, kdf } => write!(
79                f,
80                "unsupported passphrase file format (container version {version}, KDF {kdf})"
81            ),
82            PassphraseFileError::UnsupportedParams {
83                m_cost,
84                t_cost,
85                p_cost,
86            } => write!(
87                f,
88                "unsupported Argon2 parameters (m={m_cost} KiB, t={t_cost}, p={p_cost})"
89            ),
90            PassphraseFileError::EmptyPassphrase => write!(f, "the passphrase is empty"),
91            PassphraseFileError::PwHash(e) => write!(f, "{e}"),
92        }
93    }
94}
95
96impl core::error::Error for PassphraseFileError {}

98fn stream_key(passphrase: &[u8], salt: &[u8; SALT_BYTES]) -> Result<Key, PassphraseFileError> {
99    if passphrase.is_empty() {
100        return Err(PassphraseFileError::EmptyPassphrase);
101    }
102    let key = derive_key(passphrase, salt, STRENGTH).map_err(PassphraseFileError::PwHash)?;
103    Ok(Key::from_bytes(*key))
104}

106/// Starts writing a passphrase file: draws a fresh salt, runs Argon2 (256 MiB, t=3), and returns
107/// the [`FileEncryptor`] for the records plus the [`SEAL_STREAM_PREFIX_BYTES`]-byte prefix to
108/// write first. Drive the encryptor as the stream-file codec documents.
109///
110/// # Errors
111///
112/// [`PassphraseFileError::EmptyPassphrase`], or [`PassphraseFileError::PwHash`] if the CSPRNG
113/// fails or Argon2's memory cannot be allocated.
114pub fn seal_stream(
115    passphrase: &[u8],
116) -> Result<(FileEncryptor, [u8; SEAL_STREAM_PREFIX_BYTES]), PassphraseFileError> {
117    let mut salt = [0u8; SALT_BYTES];
118    randombytes_buf(&mut salt).map_err(|e| PassphraseFileError::PwHash(PwHashError::Random(e)))?;
119    let key = stream_key(passphrase, &salt)?;
120    let (enc, file_prefix) = match FileEncryptor::init(&key) {
121        Ok(ok) => ok,
122        Err(FileError::Random(e)) => {
123            return Err(PassphraseFileError::PwHash(PwHashError::Random(e)))
124        }
125        // `FileEncryptor::init`'s only failure mode is the OS CSPRNG call for its header.
126        Err(_) => unreachable!("FileEncryptor::init only ever fails via FileError::Random"),
127    };
128    let (m_cost, t_cost, p_cost) = STRENGTH.cost();
129    let mut prefix = [0u8; SEAL_STREAM_PREFIX_BYTES];
130    prefix[..3].copy_from_slice(&[CONTAINER_TYPE, CONTAINER_VERSION, KDF_ARGON2ID13]);
131    prefix[3..7].copy_from_slice(&m_cost.to_le_bytes());
132    prefix[7..11].copy_from_slice(&t_cost.to_le_bytes());
133    // `Strength::cost`'s p_cost is always 1 (libsodium's fixed lane count).
134    prefix[11] = u8::try_from(p_cost).unwrap_or(u8::MAX);
135    prefix[PREFIX_LEN..HEADER_BYTES].copy_from_slice(&salt);
136    prefix[HEADER_BYTES..].copy_from_slice(&file_prefix);
137    Ok((enc, prefix))
138}

140/// A checked passphrase-file header: everything but the passphrase is known to be acceptable.
141#[derive(Debug, Clone, Copy, PartialEq, Eq)]
142pub struct Header {
143    salt: [u8; SALT_BYTES],
144}
145
146impl Header {
147    /// Checks a file's first [`HEADER_BYTES`] bytes without running Argon2, so a reader can refuse
148    /// a wrong file before asking for a passphrase.
149    ///
150    /// # Errors
151    ///
152    /// [`PassphraseFileError::NotPassphraseFile`], [`PassphraseFileError::UnsupportedFormat`], or
153    /// [`PassphraseFileError::UnsupportedParams`] for any cost but the one preset.
154    pub fn parse(header: &[u8; HEADER_BYTES]) -> Result<Self, PassphraseFileError> {
155        let [kind, version, kdf, m0, m1, m2, m3, t0, t1, t2, t3, p, salt @ ..] = *header;
156        if kind != CONTAINER_TYPE {
157            return Err(PassphraseFileError::NotPassphraseFile);
158        }
159        if version != CONTAINER_VERSION || kdf != KDF_ARGON2ID13 {
160            return Err(PassphraseFileError::UnsupportedFormat { version, kdf });
161        }
162        let m_cost = u32::from_le_bytes([m0, m1, m2, m3]);
163        let t_cost = u32::from_le_bytes([t0, t1, t2, t3]);
164        if Strength::from_cost(m_cost, t_cost, u32::from(p)) != Some(STRENGTH) {
165            return Err(PassphraseFileError::UnsupportedParams {
166                m_cost,
167                t_cost,
168                p_cost: p,
169            });
170        }
171        Ok(Header { salt })
172    }
173
174    /// Runs Argon2 (256 MiB, t=3) and returns the [`FileDecryptor`] for everything after the
175    /// header. A wrong passphrase is reported by the decryptor, at the first record.
176    ///
177    /// # Errors
178    ///
179    /// [`PassphraseFileError::EmptyPassphrase`], or [`PassphraseFileError::PwHash`] if Argon2's
180    /// memory cannot be allocated.
181    pub fn open_stream(&self, passphrase: &[u8]) -> Result<FileDecryptor, PassphraseFileError> {
182        let key = stream_key(passphrase, &self.salt)?;
183        Ok(FileDecryptor::new(&key))
184    }
185}
